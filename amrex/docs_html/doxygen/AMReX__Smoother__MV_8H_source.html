198 void operator() (AlgVector<T>& xvec, AlgVector<T> const& bvec,
199 bool with_initial_guess = false)
200 {
201 BL_PROFILE("ChebyshevSmoother");
202 AMREX_ASSERT(xvec.partition() == m_A->partition());
203 T const lmax = lambdaMax();
204 T const lmin = lmax / m_ratio;
205 T const theta = (lmax + lmin) / T(2);
206 T const delta = (lmax - lmin) / T(2);
207 T const sigma = theta / delta;
208
209 auto const& diag = m_jacobi.scaling();
210 if (!m_work || (*m_work)[0].partition() != xvec.partition()) {
211 m_work = std::make_shared<Array<AlgVector<T>,3>>();
212 for (auto& v : *m_work) { v.define(xvec.partition()); }
213 }
214 auto& r = (*m_work)[0];
215 auto& d = (*m_work)[1];
216 auto& ad = (*m_work)[2];
217 if ( ! with_initial_guess) { xvec.setValAsync(0); }
218
219 T const c0 = T(1) / theta;
220 for (int it = 0; it < m_niters; ++it) {
221 // r = b - A x; d = D^{-1} r / theta; x += d
222 if (it == 0 && ! with_initial_guess) {
223 ForEach(xvec, d, r, bvec, diag,
224 [=] AMREX_GPU_DEVICE (T& xi, T& di, T& ri, T const& bi,
225 T const& dd)
226 {
227 ri = bi;
228 di = (dd != T(0)) ? c0 * ri / dd : T(0);
229 xi += di;
230 });
231 } else {
232 SpMV(r, *m_A, xvec);
233 ForEach(xvec, d, r, bvec, diag,
234 [=] AMREX_GPU_DEVICE (T& xi, T& di, T& ri, T const& bi,
235 T const& dd)
236 {
237 ri = bi - ri;
238 di = (dd != T(0)) ? c0 * ri / dd : T(0);
239 xi += di;
240 });
241 }
242 T rho_old = T(1) / sigma;
243 for (int k = 1; k < m_degree; ++k) {
244 // r -= A d; rho = 1/(2 sigma - rho_old);
245 // d = rho rho_old d + (2 rho / delta) D^{-1} r; x += d
246 SpMV(ad, *m_A, d);
247 T const rho = T(1) / (T(2)*sigma - rho_old);
248 T const c1 = rho * rho_old;
249 T const c2 = T(2) * rho / delta;
250 ForEach(xvec, d, r, ad, diag,
251 [=] AMREX_GPU_DEVICE (T& xi, T& di, T& ri, T const& adi,
252 T const& dd)
253 {
254 ri -= adi;
255 di = c1 * di + ((dd != T(0)) ? c2 * ri / dd : T(0));
256 xi += di;
257 });
258 rho_old = rho;
259 }
260 }
261 Gpu::streamSynchronize();
262 }

277 void estimate ()
278 {

280 AlgVector<T> v(m_A->partition());
281 AlgVector<T> w(m_A->partition());
282 {
283 auto* pv = v.data();
284 ParallelForRNG(v.numLocalRows(), [=] AMREX_GPU_DEVICE (Long i, RandomEngine const& eng) noexcept
285 {
286 pv[i] = T(Random(eng)) - T(0.5);
287 });
288 }
289 T lambda = T(0);
290 for (int it = 0; it < m_eig_iters; ++it) {
291 T const nv = v.norm2();
292 if (nv == T(0)) { break; }
293 v.scaleAsync(T(1)/nv);
294 SpMV(w, *m_A, v);
295 ForEach(w, diag, [=] AMREX_GPU_DEVICE (T& wi, T const& dd)
296 { wi = (dd != T(0)) ? wi / dd : T(0); });
297 lambda = amrex::Dot(v, w);
298 std::swap(v, w);
299 }
300 m_lambda_max = lambda * T(1.1);
301 if (m_l1) { m_lambda_max = std::min(m_lambda_max, T(1)); }
302 if (m_lambda_max <= T(0)) { m_lambda_max = T(1); }
