146template <typename T, template <typename> class V, typename I>
147void CSR<T,V,I>::sort ()
148{
149 if (nnz <= 0) { return; }
150
151#ifdef AMREX_USE_GPU
152
153#if defined(AMREX_USE_CUDA) || defined(AMREX_USE_HIP)
154
155 // The function is synchronous. If that is no longer the case, we might
156 // need to update SpMatrix::define.
157
158 constexpr int nthreads = 256;
159 constexpr int nwarps_per_block = nthreads / Gpu::Device::warp_size;
160
161 AMREX_ALWAYS_ASSERT((nrows()+nwarps_per_block-1) < Long(std::numeric_limits<int>::max()));
162
163 auto nr = int(nrows());
164 int nblocks = (nr + nwarps_per_block-1) / nwarps_per_block;
165 auto const& stream = Gpu::gpuStream();
166
167 auto* pmat = mat.data();
168 auto* pcol = col_index.data();
169 auto* prow = row_offset.data();
170
171 // Short rows (the common case on fine AMG levels): one thread per row,
172 // insertion sort in registers. Longer rows are left to the warp kernel
173 // below. Both kernels skip rows that are already sorted.
174 // The cap keeps the per-thread arrays at ~256 bytes for wide value types.
175 constexpr int SHORT_ROW = std::clamp(int(256/(sizeof(I)+sizeof(T))), 4, 16);
176 ParallelFor(nr, [=] AMREX_GPU_DEVICE (int r) noexcept
177 {
178 Long const b = prow[r];
179 int const len = int(prow[r+1] - b);
180 if (len <= 1 || len > SHORT_ROW) { return; }
181 I keys[SHORT_ROW];
182 bool sorted = true;
183 for (int i = 0; i < len; ++i) {
184 keys[i] = pcol[b+i];
185 if (i > 0) { sorted = sorted && (keys[i-1] <= keys[i]); }
186 }
187 if (sorted) { return; }
188 T vals[SHORT_ROW];
189 for (int i = 0; i < len; ++i) {
190 vals[i] = pmat[b+i];
191 }
192 for (int i = 1; i < len; ++i) {
193 I const k = keys[i];
194 T const v = vals[i];
195 int j = i - 1;
196 while (j >= 0 && keys[j] > k) {
197 keys[j+1] = keys[j];
198 vals[j+1] = vals[j];
199 --j;
200 }
201 keys[j+1] = k;
202 vals[j+1] = v;
203 }
204 for (int i = 0; i < len; ++i) {
205 pcol[b+i] = keys[i];
206 pmat[b+i] = vals[i];
207 }
208 });
209
210 Gpu::Buffer<int> needs_fallback({0});
211 auto* d_needs_fallback = needs_fallback.data();
212
213 amrex::launch_global<nthreads><<<nblocks, nthreads, 0, stream>>>
214 ([=] AMREX_GPU_DEVICE () noexcept
215 {
216 int wid = int(threadIdx.x)/Gpu::Device::warp_size;
217 int r = int(blockIdx.x)*nwarps_per_block + wid;
218 if (r >= nr) return;
219
220 Long const b = prow[r];
221 Long const e = prow[r+1];
222 auto const len = int(e - b);
223
224 if (len <= 1) return;
225
226 int lane = int(threadIdx.x) - wid * Gpu::Device::warp_size;
227
228 bool sorted = true;
229 for (Long i = lane + 1; i < len; i += Gpu::Device::warp_size) {
230 sorted = sorted && (pcol[b+i-1] <= pcol[b+i]);
231 }
232#if defined(AMREX_USE_CUDA)
233 if (__all_sync(0xffffffff, sorted)) { return; }
234#else
235 if (__all(sorted)) { return; }
236#endif
237
238 constexpr int ITEMS_PER_THREAD = AMREX_HIP_OR_CUDA(2,4);
239 constexpr int ITEMS_PER_WARP = Gpu::Device::warp_size * ITEMS_PER_THREAD;
240
241 if (len <= ITEMS_PER_WARP)
242 {
243#if defined(AMREX_USE_CUDA)
244 using WarpSort = cub::WarpMergeSort<I, ITEMS_PER_THREAD, Gpu::Device::warp_size, T>;
245 __shared__ typename WarpSort::TempStorage temp_storage[nwarps_per_block];
246#elif defined(AMREX_USE_HIP)
247 using WarpSort = rocprim::warp_sort<I, Gpu::Device::warp_size, T>;
248 __shared__ typename WarpSort::storage_type temp_storage[nwarps_per_block];
249#endif
250
251 I keys[ITEMS_PER_THREAD];
252 T values[ITEMS_PER_THREAD];
253
254 #pragma unroll
255 for (int i = 0; i < ITEMS_PER_THREAD; ++i) {
256 int idx = lane * ITEMS_PER_THREAD + i;
257 if (idx < len) {
258 keys[i] = pcol[b + idx];
259 values[i] = pmat[b + idx];
260 } else {
261 keys[i] = std::numeric_limits<I>::max();
262 values[i] = T{};
263 }
264 }
265
266 AMREX_HIP_OR_CUDA(
267 WarpSort{}.sort(keys, values, temp_storage[wid]),
268 WarpSort(temp_storage[wid]).Sort(
269 keys, values, [](I x, I y) {return x < y;}));
270
271 #pragma unroll
272 for (int i = 0; i < ITEMS_PER_THREAD; ++i) {
273 int idx = lane * ITEMS_PER_THREAD + i;
274 if (idx < len) {
275 pcol[b + idx] = keys[i];
276 pmat[b + idx] = values[i];
277 }
278 }
279 } else {
280 if (lane == 0) {
281 Gpu::Atomic::AddNoRet(d_needs_fallback, 1);
282 }
283 }
284 });
285
286 auto* h_needs_fallback = needs_fallback.copyToHost();
287
288 if (*h_needs_fallback)
289 {
290 // The item count of the segmented sort below is 32-bit.
291 AMREX_ALWAYS_ASSERT(nnz < Long(std::numeric_limits<int>::max()));
292
293 V<I> col_index_out(col_index.size());
294 V<T> mat_out(mat.size());
295 auto* d_col_out = col_index_out.data();
296 auto* d_val_out = mat_out.data();
297
298 std::size_t temp_bytes = 0;
299
300 AMREX_GPU_SAFE_CALL(AMREX_HIP_OR_CUDA(
301 rocprim::segmented_radix_sort_pairs,
302 cub::DeviceSegmentedRadixSort::SortPairs)
303 (nullptr, temp_bytes, pcol, d_col_out, pmat, d_val_out,
304 nnz, nr, prow, prow+1, 0, int(sizeof(I)*CHAR_BIT),
305 stream));
306
307 auto* d_temp = (void*) The_Arena()->alloc(temp_bytes);
308
309 AMREX_GPU_SAFE_CALL(AMREX_HIP_OR_CUDA(
310 rocprim::segmented_radix_sort_pairs,
311 cub::DeviceSegmentedRadixSort::SortPairs)
312 (d_temp, temp_bytes, pcol, d_col_out, pmat, d_val_out,
313 nnz, nr, prow, prow+1, 0, int(sizeof(I)*CHAR_BIT),
314 stream));
315
316 std::swap(col_index, col_index_out);
317 std::swap(mat, mat_out);
318
319 Gpu::streamSynchronize();
320 The_Arena()->free(d_temp);
321 }
322
323 // let's test both by print matrix out to see if it's sorted.
324
325 AMREX_GPU_ERROR_CHECK();
326
327#elif defined(AMREX_USE_SYCL)
328
329 // xxxxx TODO SYCL: Let's not worry about performance for now.
330 CSR<T,Gpu::PinnedVector,I> h_csr;
331 duplicateCSR(Gpu::deviceToHost, h_csr, *this);
332 Gpu::streamSynchronize();

346template <typename T, template <typename> class V, typename I>
347void CSR<T,V,I>::sort_on_host ()
348{
349 if (nnz <= 0) { return; }
350
351 constexpr int SMALL = 128;
352
353 Long nr = nrows();
354
355#ifdef AMREX_USE_OMP
356#pragma omp parallel
357#endif
358 {
359 V<I> lcols;
360 V<T> lvals;
361 V<int> perm;
362
363 I scols[SMALL];
364 T svals[SMALL];
365
366#ifdef AMREX_USE_OMP
367#pragma omp for
368#endif
369 for (Long r = 0; r < nr; ++r) {
370 Long const b = row_offset[r ];
371 Long const e = row_offset[r+1];
372 auto const len = int(e - b);
373
374 if (len <= 1) { continue; }
375
376 bool sorted = true;
377 for (int i = 1; i < len; ++i) {
378 if (col_index[b+i-1] > col_index[b+i]) {
379 sorted = false;
380 break;
381 }
382 }
383 if (sorted) { continue; }
384
385 if (len <= SMALL) {
386 // Insertion sort using arrays on stack
387 for (int i = 0; i < len; ++i) {
388 scols[i] = col_index[b+i];
389 svals[i] = mat [b+i];
390 }
391 for (int i = 1; i < len; ++i) {
392 auto c = scols[i];
393 auto v = svals[i];
394 auto j = i;
395 while (j > 0 && scols[j-1] > c) {
396 scols[j] = scols[j-1];
397 svals[j] = svals[j-1];
398 --j;
399 }
400 scols[j] = c;
401 svals[j] = v;
402 }
403 for (int i = 0; i < len; ++i) {
404 col_index[b+i] = scols[i];
405 mat [b+i] = svals[i];
406 }
407 } else {
408 lcols.resize(len);
409 lvals.resize(len);
410 perm.resize(len);
411
412 for (int i = 0; i < len; ++i) {
413 lcols[i] = col_index[b+i];
414 lvals[i] = mat [b+i];
415 perm [i] = i;
416 }
417
418 std::ranges::sort(perm,
419 [&] (int i0, int i1) {
420 return lcols[i0] < lcols[i1];
421 });
422
423 for (int out = 0; out < len; ++out) {
424 auto const in = perm[out];
425 col_index[b+out] = lcols[in];
426 mat [b+out] = lvals[in];
427 }
428 }
429 }
430 }
