151template <typename T, template <typename> class V, typename Index>
152void CSR<T,V,Index>::sort ()
153{
154 if (nnz <= 0) { return; }
155
156#ifdef AMREX_USE_GPU
157
158 // The function is synchronous. If that is no longer the case, we might
159 // need to update SpMatrix::define.
160
161#if defined(AMREX_USE_CUDA) || defined(AMREX_USE_HIP)
162
163 constexpr int nthreads = 256;
164 constexpr int nwarps_per_block = nthreads / Gpu::Device::warp_size;
165
166 AMREX_ALWAYS_ASSERT((nrows()+nwarps_per_block-1) < Long(std::numeric_limits<int>::max()));
167
168 auto nr = int(nrows());
169 int nblocks = (nr + nwarps_per_block-1) / nwarps_per_block;
170 auto const& stream = Gpu::gpuStream();
171
172#else
173
174 AMREX_ALWAYS_ASSERT(nrows() < Long(std::numeric_limits<int>::max()));
175
176 auto nr = int(nrows());
177
178#endif
179
180 auto* pmat = mat.data();
181 auto* pcol = col_index.data();
182 auto* prow = row_offset.data();
183
184 // Short rows (the common case on fine AMG levels): one thread per row,
185 // insertion sort in registers. Longer rows are left to the backend-specific
186 // code below. All kernels skip rows that are already sorted.
187 // The cap keeps the per-thread arrays at ~256 bytes for wide value types.
188 constexpr int SHORT_ROW = std::clamp(int(256/(sizeof(Index)+sizeof(T))), 4, 16);
189 ParallelFor(nr, [=] AMREX_GPU_DEVICE (int r) noexcept
190 {
191 Long const b = prow[r];
192 int const len = int(prow[r+1] - b);
193 if (len <= 1 || len > SHORT_ROW) { return; }
194 Index keys[SHORT_ROW];
195 bool sorted = true;
196 for (int i = 0; i < len; ++i) {
197 keys[i] = pcol[b+i];
198 if (i > 0) { sorted = sorted && (keys[i-1] <= keys[i]); }
199 }
200 if (sorted) { return; }
201 T vals[SHORT_ROW];
202 for (int i = 0; i < len; ++i) {
203 vals[i] = pmat[b+i];
204 }
205 for (int i = 1; i < len; ++i) {
206 Index const k = keys[i];
207 T const v = vals[i];
208 int j = i - 1;
209 while (j >= 0 && keys[j] > k) {
210 keys[j+1] = keys[j];
211 vals[j+1] = vals[j];
212 --j;
213 }
214 keys[j+1] = k;
215 vals[j+1] = v;
216 }
217 for (int i = 0; i < len; ++i) {
218 pcol[b+i] = keys[i];
219 pmat[b+i] = vals[i];
220 }
221 });
222
223#if defined(AMREX_USE_CUDA) || defined(AMREX_USE_HIP)
224
225 Gpu::Buffer<int> needs_fallback({0});
226 auto* d_needs_fallback = needs_fallback.data();
227
228 amrex::launch_global<nthreads><<<nblocks, nthreads, 0, stream>>>
229 ([=] AMREX_GPU_DEVICE () noexcept
230 {
231 int wid = int(threadIdx.x)/Gpu::Device::warp_size;
232 int r = int(blockIdx.x)*nwarps_per_block + wid;
233 if (r >= nr) return;
234
235 Long const b = prow[r];
236 Long const e = prow[r+1];
237 auto const len = int(e - b);
238
239 if (len <= 1) return;
240
241 int lane = int(threadIdx.x) - wid * Gpu::Device::warp_size;
242
243 bool sorted = true;
244 for (Long i = lane + 1; i < len; i += Gpu::Device::warp_size) {
245 sorted = sorted && (pcol[b+i-1] <= pcol[b+i]);
246 }
247#if defined(AMREX_USE_CUDA)
248 if (__all_sync(0xffffffff, sorted)) { return; }
249#else
250 if (__all(sorted)) { return; }
251#endif
252
253 constexpr int ITEMS_PER_THREAD = AMREX_HIP_OR_CUDA(2,4);
254 constexpr int ITEMS_PER_WARP = Gpu::Device::warp_size * ITEMS_PER_THREAD;
255
256 if (len <= ITEMS_PER_WARP)
257 {
258#if defined(AMREX_USE_CUDA)
259 using WarpSort = cub::WarpMergeSort<Index, ITEMS_PER_THREAD, Gpu::Device::warp_size, T>;
260 __shared__ typename WarpSort::TempStorage temp_storage[nwarps_per_block];
261#elif defined(AMREX_USE_HIP)
262 using WarpSort = rocprim::warp_sort<Index, Gpu::Device::warp_size, T>;
263 __shared__ typename WarpSort::storage_type temp_storage[nwarps_per_block];
264#endif
265
266 Index keys[ITEMS_PER_THREAD];
267 T values[ITEMS_PER_THREAD];
268
269 #pragma unroll
270 for (int i = 0; i < ITEMS_PER_THREAD; ++i) {
271 int idx = lane * ITEMS_PER_THREAD + i;
272 if (idx < len) {
273 keys[i] = pcol[b + idx];
274 values[i] = pmat[b + idx];
275 } else {
276 keys[i] = std::numeric_limits<Index>::max();
277 values[i] = T{};
278 }
279 }
280
281 AMREX_HIP_OR_CUDA(
282 WarpSort{}.sort(keys, values, temp_storage[wid]),
283 WarpSort(temp_storage[wid]).Sort(
284 keys, values, [](Index x, Index y) {return x < y;}));
285
286 #pragma unroll
287 for (int i = 0; i < ITEMS_PER_THREAD; ++i) {
288 int idx = lane * ITEMS_PER_THREAD + i;
289 if (idx < len) {
290 pcol[b + idx] = keys[i];
291 pmat[b + idx] = values[i];
292 }
293 }
294 } else {
295 if (lane == 0) {
296 Gpu::Atomic::AddNoRet(d_needs_fallback, 1);
297 }
298 }
299 });
300
301 auto* h_needs_fallback = needs_fallback.copyToHost();
302
303 if (*h_needs_fallback)
304 {
305 // The item count of the segmented sort below is 32-bit.
306 AMREX_ALWAYS_ASSERT(nnz < Long(std::numeric_limits<int>::max()));
307
308 V<Index> col_index_out(col_index.size());
309 V<T> mat_out(mat.size());
310 auto* d_col_out = col_index_out.data();
311 auto* d_val_out = mat_out.data();
312
313 std::size_t temp_bytes = 0;
314
315 AMREX_GPU_SAFE_CALL(AMREX_HIP_OR_CUDA(
316 rocprim::segmented_radix_sort_pairs,
317 cub::DeviceSegmentedRadixSort::SortPairs)
318 (nullptr, temp_bytes, pcol, d_col_out, pmat, d_val_out,
319 nnz, nr, prow, prow+1, 0, int(sizeof(Index)*CHAR_BIT),
320 stream));
321
322 auto* d_temp = (void*) The_Arena()->alloc(temp_bytes);
323
324 AMREX_GPU_SAFE_CALL(AMREX_HIP_OR_CUDA(
325 rocprim::segmented_radix_sort_pairs,
326 cub::DeviceSegmentedRadixSort::SortPairs)
327 (d_temp, temp_bytes, pcol, d_col_out, pmat, d_val_out,
328 nnz, nr, prow, prow+1, 0, int(sizeof(Index)*CHAR_BIT),
329 stream));
330
331 std::swap(col_index, col_index_out);
332 std::swap(mat, mat_out);
333
334 Gpu::streamSynchronize();
335 The_Arena()->free(d_temp);
336 }
337
338 // let's test both by print matrix out to see if it's sorted.
339
340 AMREX_GPU_ERROR_CHECK();
341
342#elif defined(AMREX_USE_SYCL)
343
344 Gpu::Buffer<int> needs_fallback({0});
345 auto* d_needs_fallback = needs_fallback.data();
346
347 ParallelFor(nr, [=] AMREX_GPU_DEVICE (int r) noexcept
348 {
349 Long const b = prow[r];
350 Long const e = prow[r+1];
351 if (e - b <= SHORT_ROW) { return; }
352 for (Long i = b+1; i < e; ++i) {
353 if (pcol[i-1] > pcol[i]) {
354 Gpu::Atomic::AddNoRet(d_needs_fallback, 1);
355 return;
356 }
357 }
358 });
359
360 auto* h_needs_fallback = needs_fallback.copyToHost();
361
362 if (*h_needs_fallback)
363 {
364 // Longer unsorted rows: let oneMKL sort the whole matrix on the device.
365 Long const ncols = 1 + Long(Reduce::Max<Index>(nnz, pcol));
366 auto& q = Gpu::Device::streamQueue();
367 oneapi::mkl::sparse::matrix_handle_t handle{};
368 oneapi::mkl::sparse::init_matrix_handle(&handle);
369#if defined(INTEL_MKL_VERSION) && (INTEL_MKL_VERSION < 20250300)
370 oneapi::mkl::sparse::set_csr_data(q, handle, Index(nr), Index(ncols),
371 oneapi::mkl::index_base::zero, prow, pcol, pmat);
372#else
373 oneapi::mkl::sparse::set_csr_data(q, handle, Index(nr), Index(ncols), Index(nnz),
374 oneapi::mkl::index_base::zero, prow, pcol, pmat);
375#endif
376 oneapi::mkl::sparse::sort_matrix(q, handle, {}).wait();
377 oneapi::mkl::sparse::release_matrix_handle(q, &handle).wait();
378 }
379
380 Gpu::streamSynchronize();
381
382#endif
383
384#else
385
386 sort_on_host();
387
388#endif

391template <typename T, template <typename> class V, typename Index>
392void CSR<T,V,Index>::sort_on_host ()
393{
394 if (nnz <= 0) { return; }
395
396 constexpr int SMALL = 128;
397
398 Long nr = nrows();
399
400#ifdef AMREX_USE_OMP
401#pragma omp parallel
402#endif
403 {
404 V<Index> lcols;
405 V<T> lvals;
406 V<int> perm;
407
408 Index scols[SMALL];
409 T svals[SMALL];
410
411#ifdef AMREX_USE_OMP
412#pragma omp for
413#endif
414 for (Long r = 0; r < nr; ++r) {
415 Long const b = row_offset[r ];
416 Long const e = row_offset[r+1];
417 auto const len = int(e - b);
418
419 if (len <= 1) { continue; }
420
421 bool sorted = true;
422 for (int i = 1; i < len; ++i) {
423 if (col_index[b+i-1] > col_index[b+i]) {
424 sorted = false;
425 break;
426 }
427 }
428 if (sorted) { continue; }
429
430 if (len <= SMALL) {
431 // Insertion sort using arrays on stack
432 for (int i = 0; i < len; ++i) {
433 scols[i] = col_index[b+i];
434 svals[i] = mat [b+i];
435 }
436 for (int i = 1; i < len; ++i) {
437 auto c = scols[i];
438 auto v = svals[i];
439 auto j = i;
440 while (j > 0 && scols[j-1] > c) {
441 scols[j] = scols[j-1];
442 svals[j] = svals[j-1];
443 --j;
444 }
445 scols[j] = c;
446 svals[j] = v;
447 }
448 for (int i = 0; i < len; ++i) {
449 col_index[b+i] = scols[i];
450 mat [b+i] = svals[i];
451 }
452 } else {
453 lcols.resize(len);
454 lvals.resize(len);
455 perm.resize(len);
456
457 for (int i = 0; i < len; ++i) {
458 lcols[i] = col_index[b+i];
459 lvals[i] = mat [b+i];
460 perm [i] = i;
461 }
462
463 std::ranges::sort(perm,
464 [&] (int i0, int i1) {
465 return lcols[i0] < lcols[i1];
466 });
467
468 for (int out = 0; out < len; ++out) {
469 auto const in = perm[out];
470 col_index[b+out] = lcols[in];
471 mat [b+out] = lvals[in];
472 }
473 }
474 }
475 }

503 {
504 if (i < lnnz) { col_dst[i] = static_cast<Index>(col_src[i]); }
505 if (i < lnro) { row_dst[i] = static_cast<Index>(row_src[i]); }
506 });
507 }
