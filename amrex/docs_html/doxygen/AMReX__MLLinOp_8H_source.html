1077 static RT spDiagForScale (Vector<RT> const& diag) noexcept {

1079 RT dmax = RT(0.0);

1413void
1414MLLinOpT<MF>::defineGrids (const Vector<Geometry>& a_geom,
1415 const Vector<BoxArray>& a_grids,
1416 const Vector<DistributionMapping>& a_dmap,
1417 const Vector<FabFactory<FAB> const*>& a_factory)
1418{

1420
1421#ifdef AMREX_USE_EB
1422 if ( ! a_factory.empty() ) {
1423 auto const* ebf = dynamic_cast<EBFArrayBoxFactory const*>(a_factory[0]);
1424 if (ebf && !(ebf->isAllRegular())) { // Has non-trivial EB
1425 mg_domain_min_width = 4;
1426 }
1427 }
1428#endif
1429
1430 m_num_amr_levels = 0;
1431 for (int amrlev = 0; amrlev < std::ssize(a_geom); amrlev++) {
1432 if (!a_grids[amrlev].empty()) {
1433 m_num_amr_levels++;
1434 }
1435 }
1436
1437 m_amr_ref_ratio.resize(m_num_amr_levels);
1438 m_num_mg_levels.resize(m_num_amr_levels);
1439
1440 m_geom.clear();
1441 m_grids.clear();
1442 m_dmap.clear();
1443 m_factory.clear();
1444 m_domain_covered.clear();
1445 mg_coarsen_ratio_vec.clear();
1446
1447 m_geom.resize(m_num_amr_levels);
1448 m_grids.resize(m_num_amr_levels);
1449 m_dmap.resize(m_num_amr_levels);
1450 m_factory.resize(m_num_amr_levels);
1451
1452 m_default_comm = ParallelContext::CommunicatorSub();
1453
1454 const RealBox& rb = a_geom[0].ProbDomain();
1455 const int coord = a_geom[0].Coord();
1456 const Array<int,AMREX_SPACEDIM>& is_per = a_geom[0].isPeriodic();
1457
1458 IntVect mg_coarsen_ratio_v(mg_coarsen_ratio);
1459 if (hasHiddenDimension()) {
1460 AMREX_ASSERT_WITH_MESSAGE(AMREX_SPACEDIM == 3,
1461 "Hidden direction only supported for 3d");
1462 mg_coarsen_ratio_v[info.hidden_direction] = 1;
1463 }
1464
1465 // fine amr levels
1466 for (int amrlev = m_num_amr_levels-1; amrlev > 0; --amrlev)
1467 {
1468 m_num_mg_levels[amrlev] = 1;
1469 m_geom[amrlev].push_back(a_geom[amrlev]);
1470 m_grids[amrlev].push_back(a_grids[amrlev]);
1471 m_dmap[amrlev].push_back(a_dmap[amrlev]);
1472 if (amrlev < std::ssize(a_factory)) {
1473 m_factory[amrlev].emplace_back(a_factory[amrlev]->clone());
1474 } else {
1475 m_factory[amrlev].push_back(std::make_unique<DefaultFabFactory<FAB>>());
1476 }
1477
1478 IntVect rr = mg_coarsen_ratio_v;
1479 const Box& dom = a_geom[amrlev].Domain();
1480 for (int i = 0; i < 2; ++i)
1481 {
1482 if (!dom.coarsenable(rr)) { amrex::Abort("MLLinOp: Uncoarsenable domain"); }
1483
1484 const Box& cdom = amrex::coarsen(dom,rr);
1485 if (cdom == a_geom[amrlev-1].Domain()) { break; }
1486
1487 ++(m_num_mg_levels[amrlev]);
1488
1489 m_geom[amrlev].emplace_back(cdom, rb, coord, is_per);
1490
1491 m_grids[amrlev].push_back(a_grids[amrlev]);
1492 AMREX_ASSERT(m_grids[amrlev].back().coarsenable(rr));
1493 m_grids[amrlev].back().coarsen(rr);
1494
1495 m_dmap[amrlev].push_back(a_dmap[amrlev]);
1496
1497 rr *= mg_coarsen_ratio_v;
1498 }
1499
1500#if (AMREX_SPACEDIM > 1)
1501 if (hasHiddenDimension()) {
1502 m_amr_ref_ratio[amrlev-1] = rr[(info.hidden_direction+1) % AMREX_SPACEDIM];
1503 } else
1504#endif
1505 {
1506 m_amr_ref_ratio[amrlev-1] = rr[0];
1507 }
1508 }
1509
1510 // coarsest amr level
1511 m_num_mg_levels[0] = 1;
1512 m_geom[0].push_back(a_geom[0]);
1513 m_grids[0].push_back(a_grids[0]);
1514 m_dmap[0].push_back(a_dmap[0]);
1515 if (!a_factory.empty()) {
1516 m_factory[0].emplace_back(a_factory[0]->clone());
1517 } else {
1518 m_factory[0].push_back(std::make_unique<DefaultFabFactory<FAB>>());

1525 {

1528 compactify(m_geom[amrlev][0].Domain()).numPts());
1529 }
1530
1531 if (m_mg_deferred) {
1532 // MG levels > 0 of AMR level 0 are built by prepareMGHierarchy.
1533 m_bottom_comm = m_default_comm;
1534 m_do_agglomeration = false;
1535 m_do_consolidation = false;
1536 } else {
1537 defineCoarseMGLevels();
1538 }

1541 {

1543 {
1544 m_factory[amrlev].emplace_back(makeFactory(amrlev,mglev));
1545 }
1546 }

1555template <typename MF>
1556void
1557MLLinOpT<MF>::defineCoarseMGLevels ()
1558{
1559 BL_PROFILE("MLLinOp::defineCoarseMGLevels()");
1560
1561 AMREX_ASSERT(m_grids[0].size() == 1 && mg_coarsen_ratio_vec.empty());
1562
1563 // Copies, because m_geom[0], m_grids[0] and m_dmap[0] grow below.
1564 Geometry const geom0 = m_geom[0][0];
1565 BoxArray const ba0 = m_grids[0][0];
1566 DistributionMapping const dm0 = m_dmap[0][0];
1567
1568 const RealBox& rb = geom0.ProbDomain();
1569 const int coord = geom0.Coord();
1570 const Array<int,AMREX_SPACEDIM>& is_per = geom0.isPeriodic();
1571 auto const npts0 = ba0.numPts();
1572
1573 IntVect mg_coarsen_ratio_v(mg_coarsen_ratio);
1574 IntVect mg_box_min_width_v(mg_box_min_width);
1575 IntVect mg_domain_min_width_v(mg_domain_min_width);
1576 if (hasHiddenDimension()) {
1577 mg_coarsen_ratio_v[info.hidden_direction] = 1;
1578 mg_box_min_width_v[info.hidden_direction] = 0;
1579 mg_domain_min_width_v[info.hidden_direction] = 0;
1580 }
1581
1582 Box aggbox;
1583 bool aggable = false;
1584
1585 if (ba0.size() > 1 && info.do_agglomeration)
1586 {
1587 if (m_domain_covered[0])
1588 {
1589 aggbox = geom0.Domain();
1590 if (hasHiddenDimension()) {
1591 aggbox.makeSlab(hiddenDirection(), ba0[0].smallEnd(hiddenDirection()));
1592 }
1593 aggable = true;
1594 }
1595 else
1596 {
1597 aggbox = ba0.minimalBox();
1598 aggable = (aggbox.numPts() == npts0);
1599 }
1600 }
1601
1602 bool agged = false;
1603 bool coned = false;
1604 int agg_lev = 0, con_lev = 0;
1605
1606 AMREX_ALWAYS_ASSERT( ! (info.do_semicoarsening && info.hasHiddenDimension()
1607 && (info.semicoarsening_direction == -1 ||
1608 info.semicoarsening_direction == info.hidden_direction))
1609 && info.semicoarsening_direction >= -1
1610 && info.semicoarsening_direction < AMREX_SPACEDIM );
1611
1612 // Directions to coarsen for stretched cells: those shorter than 1.5
1613 // times the shortest. A fixed semicoarsening direction takes precedence.
1614 bool const aniso_coarsening = supportsAnisotropicCoarsening() && !hasHiddenDimension()
1615 && !(info.do_semicoarsening && info.semicoarsening_direction != -1);
1616 auto const dx0 = anisotropicCoarseningCellSize(geom0);
1617 auto base_ratio = [&] (IntVect const& accum) -> IntVect
1618 {
1619 IntVect rr = mg_coarsen_ratio_v;
1620 if (aniso_coarsening) {
1621 Real dxmin = std::numeric_limits<Real>::max();
1622 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1623 dxmin = std::min(dxmin, dx0[idim]*Real(accum[idim]));
1624 }
1625 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1626 if (dx0[idim]*Real(accum[idim]) >= Real(1.5)*dxmin) { rr[idim] = 1; }
1627 }
1628 }
1629 return rr;
1630 };
1631
1632 if (info.do_agglomeration && aggable)
1633 {
1634 Box dbx = geom0.Domain();
1635 Box bbx = aggbox;
1636 Real const nbxs = static_cast<Real>(ba0.size());
1637 Long const threshold_npts = AMREX_D_TERM(Long(info.agg_grid_size),
1638 *info.agg_grid_size,
1639 *info.agg_grid_size);
1640 Vector<Box> domainboxes{dbx};
1641 Vector<Box> boundboxes{bbx};
1642 Vector<int> agg_flag{false};
1643 Vector<IntVect> accum_coarsen_ratio{IntVect(1)};
1644 int numsclevs = 0;
1645
1646 for (int lev = 0; lev < info.max_coarsening_level; ++lev)
1647 {
1648 IntVect const rr_base = base_ratio(accum_coarsen_ratio.back());
1649 IntVect rr_level = rr_base;
1650 bool const do_semicoarsening_level = info.do_semicoarsening
1651 && numsclevs < info.max_semicoarsening_level;
1652 if (do_semicoarsening_level
1653 && info.semicoarsening_direction != -1)
1654 {
1655 rr_level[info.semicoarsening_direction] = 1;
1656 }
1657 IntVect is_coarsenable;
1658 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1659 IntVect rr_dir(1);
1660 rr_dir[idim] = rr_level[idim];
1661 // With odd ratios, the first ratio must also fit the original boxes.
1662 bool const fit_ba0 = mg_odd_coarsening && lev == 0;

1664 && bbx.coarsenable(rr_dir, mg_box_min_width_v)
1665 && (!fit_ba0 || ba0.coarsenable(rr_dir, mg_box_min_width_v));
1666 for (int r : {3, 5}) {
1667 if (is_coarsenable[idim] || !mg_odd_coarsening || rr_level[idim] != 2) {
1668 break;
1669 }
1670 rr_dir[idim] = r;
1671 if (dbx.coarsenable(rr_dir, mg_domain_min_width_v)
1672 && bbx.coarsenable(rr_dir, mg_box_min_width_v)
1673 && (!fit_ba0 || ba0.coarsenable(rr_dir, mg_box_min_width_v)))
1674 {
1675 is_coarsenable[idim] = true;
1676 rr_level[idim] = r;
1677 }
1678 }
1679 if (!is_coarsenable[idim] && do_semicoarsening_level
1680 && info.semicoarsening_direction == -1)
1681 {
1682 is_coarsenable[idim] = true;
1683 rr_level[idim] = 1;
1684 }
1685 }
1686 if (mg_independent_coarsening) {
1687 // A direction that cannot be coarsened gets ratio 1, and one whose
1688 // cells are 1.5 times longer than the shortest coarsenable waits.
1689 Real hmin = std::numeric_limits<Real>::max();
1690 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1691 if (is_coarsenable[idim] && rr_level[idim] > 1) {
1692 hmin = std::min(hmin, dx0[idim]*Real(accum_coarsen_ratio.back()[idim]));
1693 }
1694 }
1695 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1696 if (!is_coarsenable[idim] || dx0[idim]*Real(accum_coarsen_ratio.back()[idim]) >= Real(1.5)*hmin) {
1697 is_coarsenable[idim] = true;
1698 rr_level[idim] = 1;
1699 }
1700 }
1701 }
1702 if (is_coarsenable != IntVect(1) || rr_level == IntVect(1)) {
1703 break;
1704 }
1705 if (do_semicoarsening_level && info.semicoarsening_direction == -1) {
1706 // make sure there is at most one direction that is not coarsened
1707 int n_ones = AMREX_D_TERM( static_cast<int>(rr_level[0] == 1),
1708 + static_cast<int>(rr_level[1] == 1),
1709 + static_cast<int>(rr_level[2] == 1));
1710 if (n_ones > 1 && !aniso_coarsening) { break; }
1711 }
1712 bool semi_level = false; // odd ratios do not count
1713 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1714 semi_level = semi_level || (rr_level[idim] == 1 && rr_base[idim] != 1);
1715 }
1716 if (semi_level) {
1717 ++numsclevs;
1718 }
1719
1720 accum_coarsen_ratio.push_back(accum_coarsen_ratio.back()*rr_level);
1721 domainboxes.push_back(dbx.coarsen(rr_level));
1722 boundboxes.push_back(bbx.coarsen(rr_level));
1723 bool to_agg = (bbx.d_numPts() / nbxs)
1724 < Real(0.999)*static_cast<Real>(threshold_npts);
1725 agg_flag.push_back(to_agg);
1726 }
1727
1728 for (int lev = 1, nlevs = static_cast<int>(domainboxes.size()); lev < nlevs; ++lev) {
1729 if (!agged && !agg_flag[lev] &&
1730 ba0.coarsenable(accum_coarsen_ratio[lev], mg_box_min_width_v))
1731 {
1732 m_grids[0].push_back(amrex::coarsen(ba0, accum_coarsen_ratio[lev]));
1733 m_dmap[0].push_back(dm0);
1734 } else {
1735 IntVect cr = domainboxes[lev-1].length() / domainboxes[lev].length();
1736 if (!agged && lev > 1 && !m_grids[0].back().coarsenable(cr)) {
1737 // The boxes of level lev-1 cannot be coarsened further,
1738 // so agglomerate from that level instead.
1739 m_grids[0].pop_back();
1740 m_dmap[0].pop_back();
1741 m_geom[0].pop_back();
1742 --lev;
1743 cr = domainboxes[lev-1].length() / domainboxes[lev].length();
1744 }
1745 if (!m_grids[0].back().coarsenable(cr)) {
1746 break; // average_down would fail if fine boxarray is not coarsenable.
1747 }
1748 m_grids[0].emplace_back(boundboxes[lev]);
1749 Box const cell_box = amrex::enclosedCells(boundboxes[lev]);
1750 if (cell_box.numPts() > threshold_npts) {
1751 IntVect max_grid_size(info.agg_grid_size);
1752 if (info.do_semicoarsening && info.max_semicoarsening_level >= lev
1753 && info.semicoarsening_direction != -1)
1754 {
1755 IntVect blen = cell_box.size();
1756 AMREX_D_TERM(int mgs_0 = (max_grid_size[0]+blen[0]-1) / blen[0];,
1757 int mgs_1 = (max_grid_size[1]+blen[1]-1) / blen[1];,
1758 int mgs_2 = (max_grid_size[2]+blen[2]-1) / blen[2]);
1759 max_grid_size[info.semicoarsening_direction]
1760 *= AMREX_D_TERM(mgs_0, *mgs_1, *mgs_2);
1761 }
1762 if (mg_agg_no_split_direction >= 0 &&
1763 mg_agg_no_split_direction < AMREX_SPACEDIM)
1764 {
1765 int const d = mg_agg_no_split_direction;
1766 max_grid_size[d] = std::max(max_grid_size[d], cell_box.length(d));
1767 }
1768 // Chop in units of the next coarsening ratio, so that
1769 // the boxes can be coarsened.
1770 IntVect const rn = (lev+1 < nlevs)
1771 ? domainboxes[lev].length() / domainboxes[lev+1].length() : IntVect(1);
1772 BoxArray& cba = m_grids[0].back();
1773 cba.coarsen(rn);
1774 cba.maxSize((max_grid_size+rn-1)/rn);
1775 cba.refine(rn);
1776 }
1777 m_dmap[0].push_back(DistributionMapping());
1778 if (!agged) {
1779 agged = true;
1780 agg_lev = lev;
1781 }
1782 }
1783 m_geom[0].emplace_back(domainboxes[lev],rb,coord,is_per);
1784 }
1785 }
1786 else
1787 {
1788 Long consolidation_threshold = 0;
1789 Real avg_npts = 0.0;
1790 if (info.do_consolidation) {
1791 avg_npts = static_cast<Real>(ba0.d_numPts()) / static_cast<Real>(ParallelContext::NProcsSub());
1792 consolidation_threshold = AMREX_D_TERM(Long(info.con_grid_size),
1793 *info.con_grid_size,
1794 *info.con_grid_size);
1795 }
1796
1797 Box const& dom0 = geom0.Domain();
1798 IntVect rr_vec(1);
1799 int numsclevs = 0;
1800 for (int lev = 0; lev < info.max_coarsening_level; ++lev)
1801 {
1802 IntVect const rr_base = base_ratio(rr_vec);
1803 IntVect rr_level = rr_base;
1804 bool do_semicoarsening_level = info.do_semicoarsening
1805 && numsclevs < info.max_semicoarsening_level;
1806 if (do_semicoarsening_level
1807 && info.semicoarsening_direction != -1)
1808 {
1809 rr_level[info.semicoarsening_direction] = 1;
1810 }
1811 IntVect is_coarsenable;
1812 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1813 IntVect rr_dir(1);
1814 rr_dir[idim] = rr_vec[idim] * rr_level[idim];
1815 is_coarsenable[idim] = dom0.coarsenable(rr_dir, mg_domain_min_width_v)
1816 && ba0.coarsenable(rr_dir, mg_box_min_width_v);
1817 for (int r : {3, 5}) {
1818 if (is_coarsenable[idim] || !mg_odd_coarsening || rr_level[idim] != 2) {
1819 break;
1820 }
1821 rr_dir[idim] = rr_vec[idim] * r;
1822 if (dom0.coarsenable(rr_dir, mg_domain_min_width_v)
1823 && ba0.coarsenable(rr_dir, mg_box_min_width_v))
1824 {
1825 is_coarsenable[idim] = true;
1826 rr_level[idim] = r;
1827 }
1828 }
1829 if (!is_coarsenable[idim] && do_semicoarsening_level
1830 && info.semicoarsening_direction == -1)
1831 {
1832 is_coarsenable[idim] = true;
1833 rr_level[idim] = 1;
1834 }
1835 }
1836 if (mg_independent_coarsening) {
1837 // A direction that cannot be coarsened gets ratio 1, and one whose
1838 // cells are 1.5 times longer than the shortest coarsenable waits.
1839 Real hmin = std::numeric_limits<Real>::max();
1840 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1841 if (is_coarsenable[idim] && rr_level[idim] > 1) {
1842 hmin = std::min(hmin, dx0[idim]*Real(rr_vec[idim]));
1843 }
1844 }
1845 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1846 if (!is_coarsenable[idim] || dx0[idim]*Real(rr_vec[idim]) >= Real(1.5)*hmin) {
1847 is_coarsenable[idim] = true;
1848 rr_level[idim] = 1;
1849 }
1850 }
1851 }
1852 if (is_coarsenable != IntVect(1) || rr_level == IntVect(1)) {
1853 break;
1854 }
1855 if (do_semicoarsening_level && info.semicoarsening_direction == -1) {
1856 // make sure there is at most one direction that is not coarsened
1857 int n_ones = AMREX_D_TERM( static_cast<int>(rr_level[0] == 1),
1858 + static_cast<int>(rr_level[1] == 1),
1859 + static_cast<int>(rr_level[2] == 1));
1860 if (n_ones > 1 && !aniso_coarsening) { break; }
1861 }
1862 bool semi_level = false; // odd ratios do not count
1863 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
1864 semi_level = semi_level || (rr_level[idim] == 1 && rr_base[idim] != 1);
1865 }
1866 if (semi_level) {
1867 ++numsclevs;
1868 }
1869 rr_vec *= rr_level;
1870
1871 m_geom[0].emplace_back(amrex::coarsen(dom0, rr_vec), rb, coord, is_per);
1872 m_grids[0].push_back(amrex::coarsen(ba0, rr_vec));
1873
1874 if (info.do_consolidation)
1875 {
1876 if (avg_npts/static_cast<Real>(AMREX_D_TERM(rr_vec[0], *rr_vec[1], *rr_vec[2]))
1877 < Real(0.999)*static_cast<Real>(consolidation_threshold))
1878 {
1879 coned = true;
1880 con_lev = m_dmap[0].size();
1881 m_dmap[0].push_back(DistributionMapping());
1882 }
1883 else
1884 {
1885 m_dmap[0].push_back(m_dmap[0].back());
1886 }
1887 }
1888 else
1889 {
1890 m_dmap[0].push_back(dm0);
1891 }
1892 }
1893 }
1894
1895 m_num_mg_levels[0] = m_grids[0].size();
1896
1897 for (int mglev = 0; mglev < m_num_mg_levels[0] - 1; mglev++){
1898 const Box& fine_domain = m_geom[0][mglev].Domain();
1899 const Box& crse_domain = m_geom[0][mglev+1].Domain();
1900 mg_coarsen_ratio_vec.push_back(fine_domain.length()/crse_domain.length());
1901 }
1902
1903 for (int amrlev = 0; amrlev < m_num_amr_levels; ++amrlev) {
1904 if (AMRRefRatio(amrlev) == 4 && mg_coarsen_ratio_vec.empty()) {
1905 mg_coarsen_ratio_vec.push_back(IntVect(2));
1906 }
1907 }
1908
1909 if (agged)
1910 {
1911 makeAgglomeratedDMap(m_grids[0], m_dmap[0]);
1912 }
1913 else if (coned)
1914 {
1915 makeConsolidatedDMap(m_grids[0], m_dmap[0], info.con_ratio, info.con_strategy);
1916 }
1917
1918 if (agged || coned)
1919 {
1920 m_bottom_comm = makeSubCommunicator(m_dmap[0].back());
1921 }
1922 else
1923 {
1924 m_bottom_comm = m_default_comm;
1925 }
1926
1927 m_do_agglomeration = agged;
1928 m_do_consolidation = coned;
1929
1930 if (verbose > 1) {
1931 if (agged) {
1932 Print() << "MLLinOp::defineGrids(): agglomerated AMR level 0 starting at MG level "
1933 << agg_lev << " of " << m_num_mg_levels[0] << "\n";
1934 } else if (coned) {
1935 Print() << "MLLinOp::defineGrids(): consolidated AMR level 0 starting at MG level "
1936 << con_lev << " of " << m_num_mg_levels[0]
1937 << " (ratio = " << info.con_ratio << ")" << "\n";
1938 } else {
1939 Print() << "MLLinOp::defineGrids(): no agglomeration or consolidation of AMR level 0\n";
1940 }
1941 }
1942
1943 for (int mglev = 1; mglev < m_num_mg_levels[0]; ++mglev)
1944 {
1945 m_factory[0].emplace_back(makeFactory(0,mglev));
1946 }
1947}
