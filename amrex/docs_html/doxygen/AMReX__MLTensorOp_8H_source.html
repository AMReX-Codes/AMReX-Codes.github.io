20// Tensor solver for high Reynolds flows with small gradient in viscosity.
21// The system it solves is
22//
23// alpha a v - beta div dot tau = rhs
24//
25// where tau = eta [grad v + (grad v)^T] + (kappa-(2/3)eta) (div v) I.
26// Here eta and kappa are shear and bulk viscosity, and I is identity tensor.
27//
28// The user needs to provide `a` by `setACoeffs`, eta by `setShearViscosity`,
29// and kappa by `setBulkViscosity`. If `setBulkViscosity` is not called,
30// kappa is set to zero.
31//
32// The scalars alpha and beta can be set with `setScalars(Real, Real)`. If
33// they are not set, their default value is 1.
34

288template <FabArrayType AMF>

290void
291MLTensorOpT<MF>::setBulkViscosity (int amrlev, const Array<AMF const*,AMREX_SPACEDIM>& kappa)
292{
293 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
294 m_kappa[amrlev][0][idim].LocalCopy(*kappa[idim], 0, 0, 1, IntVect(0));
295 }
296 m_has_kappa = true;
297 m_needs_update = true;
298 this->template spForward<MLTensorOpT>([&] (auto* sp) { sp->setBulkViscosity(amrlev, kappa); });
299}

304void
305MLTensorOpT<MF>::setMappingFactors (int amrlev, const Array<AMF const*,AMREX_SPACEDIM>& fac)
306{
307 if (m_mapfac.empty()) { m_mapfac.resize(this->NAMRLevels()); }
308 for (int idim = 0; idim < AMREX_SPACEDIM; ++idim) {
309 AMREX_ALWAYS_ASSERT(fac[idim]->nComp() >= AMREX_SPACEDIM);
310 if (!m_mapfac[amrlev][idim].ok()) {
311 m_mapfac[amrlev][idim].define
312 (amrex::convert(this->m_grids[amrlev][0], IntVect::TheDimensionVector(idim)),
313 this->m_dmap[amrlev][0], AMREX_SPACEDIM, 0, MFInfo(), *this->m_factory[amrlev][0]);
314 }
315 m_mapfac[amrlev][idim].LocalCopy(*fac[idim], 0, 0, AMREX_SPACEDIM, IntVect(0));
316 }
317 m_use_mapped = true;
318 m_needs_update = true;
319 this->template spForward<MLTensorOpT>([&] (auto* sp) { sp->setMappingFactors(amrlev, fac); });
320}
