25// Unpack functor: ADD with plain += when the tags are thread safe, atomics
26// otherwise; COPY stores (benign races store the same value).
27template <typename BUF, typename T, bool ADD, bool ATOMIC>
28struct FBOnArrivalUnpack {
29 char const* dbase;
30 int dcomp;
31 int ncomp;
32
33 AMREX_GPU_DEVICE AMREX_FORCE_INLINE
34 void operator() (int icell, int ncells, int i, int j, int k,
35 CommRecvBufTag<T> const& tag) const noexcept
36 {
37 if (icell < ncells) {
38 Array4<BUF const> sfab{(BUF const*)(dbase + tag.poff),
39 amrex::begin(tag.bx), amrex::end(tag.bx), ncomp};
40 for (int n = 0; n < ncomp; ++n) {
41 if constexpr (ADD && ATOMIC) {
42 Gpu::Atomic::AddNoRet(&tag.dfab(i,j,k,n+dcomp), (T)sfab(i,j,k,n));
43 } else if constexpr (ADD) {
44 tag.dfab(i,j,k,n+dcomp) += (T)sfab(i,j,k,n);
45 } else {
46 tag.dfab(i,j,k,n+dcomp) = (T)sfab(i,j,k,n);
47 }
48 }
49 }
50 }
51};

70// Ranges of warps of a TagVector, passed to the kernel by value. The
71// messages completed by one Waitsome are unpacked by one launch over them.
72struct FBOnArrivalWarpRanges {
73 static constexpr int max_ranges = 16;
74 int begin[max_ranges] = {0}; // first warp of each range in the TagVector
75 int prefix[max_ranges+1] = {0}; // prefix sum of the range sizes
76 int n = 0;
77
78 // Returns false if the table is full and the range cannot be merged.
79 bool add (int b, int nw) noexcept {
80 if (n > 0 && begin[n-1] + (prefix[n]-prefix[n-1]) == b) {
81 prefix[n] += nw;
82 } else if (n < max_ranges) {
83 begin[n] = b;
84 prefix[n+1] = prefix[n] + nw;
85 ++n;
86 } else {
87 return false;
88 }
89 return true;
90 }
91};

93template <typename T, typename F>
94void fb_oa_unpack_ranges (TagVector<CommRecvBufTag<T>> const& tv,
95 FBOnArrivalWarpRanges const& r, F const& f)
96{
97 const int nwarps = r.prefix[r.n];
98 if (nwarps == 0) { return; }
99
100 constexpr int nthreads = TagVector<CommRecvBufTag<T>>::nthreads;
101 constexpr int nwarps_per_block = nthreads / Gpu::Device::warp_size;
102 const int nblocks = (nwarps + nwarps_per_block-1) / nwarps_per_block;
103 const auto d_tags = tv.d_tags;
104 const auto d_nwarps = tv.d_nwarps;
105 const auto ntags = tv.ntags;
106
107 amrex::launch<nthreads>(nblocks, Gpu::gpuStream(),
108#ifdef AMREX_USE_SYCL
109 [=] AMREX_GPU_DEVICE (sycl::nd_item<1> const& item) noexcept
110 [[sycl::reqd_work_group_size(nthreads)]]
111 [[sycl::reqd_sub_group_size(Gpu::Device::warp_size)]]
112#else
113 [=] AMREX_GPU_DEVICE () noexcept
114#endif
115 {
116#ifdef AMREX_USE_SYCL
117 std::size_t g_tid = item.get_global_id(0);
118#else
119 auto g_tid = std::size_t(blockDim.x)*blockIdx.x + threadIdx.x;
120#endif
121 auto l_wid = int(g_tid / Gpu::Device::warp_size);
122 if (l_wid >= nwarps) { return; }
123 int ir = amrex::bisect(r.prefix, 0, r.n, l_wid);
124 int g_wid = r.begin[ir] + (l_wid - r.prefix[ir]);
125 int tag_id = amrex::bisect(d_nwarps, 0, ntags, g_wid);
126 int b_wid = g_wid - d_nwarps[tag_id];
127#ifdef AMREX_USE_SYCL
128 int lane = item.get_local_id(0) % Gpu::Device::warp_size;
129#else
130 int lane = threadIdx.x % Gpu::Device::warp_size;
131#endif
132 int icell = b_wid*Gpu::Device::warp_size + lane;
133 tagparfor_call_f(icell, d_tags[tag_id], f);
134 });
135}

167 lo.offset.push_back(lo.nbytes);
168 lo.nbytes += nbytes;
169 }

173 make_layout(*TheFB.m_RcvTags, false, e->rcv);

177 int m = 0;

180 for (auto const& cct : kv.second) {

193 // range of warps in recv_tv (same warp count as TagVector::define).
194 Vector<CommRecvBufTag<value_type>> tags;

196 int m = 0;

198 auto off = static_cast<std::ptrdiff_t>(e->rcv.offset[m]);
199 int nwarps = e->rcv_warp_begin.back();
200 for (auto const& cct : kv.second) {
201 const int li = this->localindex(cct.dstIndex);
202 tags.push_back(CommRecvBufTag<value_type>{
203 .dfab = this->atLocalIdx(li).array(),
204 .poff = off, .bx = cct.dbox});
205 off += cct.dbox.numPts() * ncomp * sizeof(BUF);
206 nwarps += static_cast<int>((detail::get_tag_size(tags.back())
207 + Gpu::Device::warp_size-1)
208 / Gpu::Device::warp_size);
209 }
210 e->rcv_warp_begin.push_back(nwarps);
211 ++m;
212 }
213 e->recv_tv.define(tags);
214 AMREX_ALWAYS_ASSERT(e->rcv_warp_begin.back() == e->recv_tv.ntotwarps);
215 }
216
217 auto* r = e.get();
218 m_fb_on_arrival_meta[key] = std::move(e);
219 return r;

297template <class FAB>
298template <typename BUF>
299void
300FabArray<FAB>::FB_on_arrival_finish ()
301{
302 auto* e = fbd->oa_meta;
303 const FB* TheFB = fbd->fb;
304 const auto N_rcvs = static_cast<int>(e->rcv.rank.size());
305 const auto N_snds = static_cast<int>(e->snd.rank.size());
306
307 if (N_rcvs > 0) {
308 const bool sumboundary = TheFB->m_sb_snghost.allGE(0);
309 // Below the threshold, wait for all messages and unpack them in one
310 // launch (the Waitsome loop costs more than it hides for tiny messages).
311 const bool on_arrival = e->rcv.nbytes >=
312 static_cast<std::size_t>(std::max(Long(0), FabArrayBase::m_fb_on_arrival_min_bytes));
313 auto unpack = [&] (auto const& f) {
314 if (!on_arrival) {
315 {
316 BL_PROFILE("FB::wait_recvs");
317 ParallelDescriptor::Waitall(fbd->recv_reqs, fbd->recv_stat);
318 }
319 BL_PROFILE("FB::unpack");
320 detail::ParallelFor_doit(e->recv_tv, f);
321 return;
322 }
323 // The messages completed by each Waitsome are unpacked by one
324 // launch. Messages already completed by FillBoundary_test have
325 // null requests, which Waitsome never returns, so they go first.
326 detail::FBOnArrivalWarpRanges ranges;
327 auto flush = [&] () {
328 if (ranges.n > 0) {
329 BL_PROFILE("FB::unpack");
330 detail::fb_oa_unpack_ranges(e->recv_tv, ranges, f);
331 ranges.n = 0;
332 }
333 };
334 auto add = [&] (int m) {
335 const int b = e->rcv_warp_begin[m];
336 const int nw = e->rcv_warp_begin[m+1] - b;
337 if (nw > 0 && !ranges.add(b, nw)) {
338 flush();
339 ranges.add(b, nw);
340 }
341 };
342 for (int m = 0; m < N_rcvs; ++m) {
343 if (fbd->recv_reqs[m] == MPI_REQUEST_NULL) { add(m); }
344 }
345 flush();
346 Vector<int> indx(N_rcvs);
347 for (;;) {
348 int completed = 0;
349 {
350 BL_PROFILE("FB::wait_recvs");
351 ParallelDescriptor::Waitsome(fbd->recv_reqs, completed, indx, fbd->recv_stat);
352 }
353 if (completed == MPI_UNDEFINED || completed == 0) { break; }
354 std::sort(indx.begin(), indx.begin()+completed);
355 for (int c = 0; c < completed; ++c) { add(indx[c]); }
356 flush();
357 }
358 };
359
360 using T = value_type;
361 if (sumboundary) {
362 if constexpr (amrex::IsAddAssignable<T>::value) {
363 if (TheFB->m_threadsafe_rcv) {
364 unpack(detail::FBOnArrivalUnpack<BUF,T,true,false>
365 {fbd->the_recv_data, fbd->scomp, fbd->ncomp});
366 } else if constexpr (amrex::HasAtomicAdd<T>::value) {
367 unpack(detail::FBOnArrivalUnpack<BUF,T,true,true>
368 {fbd->the_recv_data, fbd->scomp, fbd->ncomp});
369 } else {
370 amrex::Abort("FB_on_arrival_finish: SumBoundary needs thread-safe tags or atomic add");
371 }
372 }
373 } else {
374 unpack(detail::FBOnArrivalUnpack<BUF,T,false,false>
375 {fbd->the_recv_data, fbd->scomp, fbd->ncomp});
376 }
377 }
378
379 // In a no-sync region the recv buffer is released in stream order.
380 const bool sync = !Gpu::inNoSyncRegion();
381 if (sync) {
382 BL_PROFILE("FB::unpack_sync");
383 Gpu::streamSynchronize();
384 }
385 if (fbd->the_recv_data) {
386 if (sync) {
387 The_Comms_Arena()->free(fbd->the_recv_data);
388 } else {
389 Gpu::freeAsync(The_Comms_Arena(), fbd->the_recv_data);
390 }
391 }
392
393 if (N_snds > 0) {
394 BL_PROFILE("FB::wait_sends");
395 Vector<MPI_Status> stats(fbd->send_reqs.size());
396 ParallelDescriptor::Waitall(fbd->send_reqs, stats);
397 }
398 // The pack kernel finished before the sends were posted.
399 if (fbd->the_send_data) { The_Comms_Arena()->free(fbd->the_send_data); }
400
401 fbd.reset();
402}
