30 requires (ET::value)
31 std::vector<std::pair<std::string,T>> getNewEnumNameValuePairs ()
32 {
33 auto tmp = amrex::split(std::string(ET::enum_names), ",");
34 std::vector<std::pair<std::string,T>> r;
35 r.reserve(tmp.size());
36 int next_value = 0;
37 for (auto const& item : tmp) {
38 auto const& kv = amrex::split(item, "=");
39 if (kv.size() == 1) {
40 r.emplace_back(amrex::trim(kv[0]), static_cast<T>(next_value));
41 ++next_value;
42 } else if (kv.size() == 2) {
43 auto const& value_string = amrex::trim(kv[1]);
44 // For k = v, find if v exists as a key. If not, v must be an int.
45 auto it = std::find_if(r.begin(), r.end(),
46 [&] (auto const& x) -> bool
47 { return x.first == value_string; });
48 T this_value;
49 if (it != r.end()) {
50 this_value = it->second;
51 } else { // v is an int literal
52 // The base of the integer is auto detected, so 0x/0 prefixes work.
53 auto s = value_string;
54 s.erase(std::remove(s.begin(), s.end(), '\''), s.end()); // digit separators
55 std::size_t pos = 0;
56 this_value = static_cast<T>(std::stoi(s, &pos, 0));
57 // Only an integer suffix may follow. Binary literals and
58 // expressions such as `1 << 2` are not supported.
59 if (s.find_first_not_of("uUlLzZ", pos) != std::string::npos) {
60 std::string error_msg("amrex::getEnumNameValuePairs: unsupported enumerator initializer `");
61 error_msg.append(value_string).append("` in AMREX_ENUM(")
62 .append(ET::class_name).append(", ").append(ET::enum_names)
63 .append(").");
64 throw std::runtime_error(error_msg);
65 }
66 }
67 r.emplace_back(amrex::trim(kv[0]), this_value);
68 next_value = static_cast<int>(this_value) + 1;
69 } else {
70 std::string error_msg("amrex::getEnumNameIntPairs: AMREX_ENUM(");
71 error_msg.append(ET::class_name).append(", ").append(ET::enum_names)
72 .append(").");
73 throw std::runtime_error(error_msg);
74 }
75 }
76 return r;
77 }
