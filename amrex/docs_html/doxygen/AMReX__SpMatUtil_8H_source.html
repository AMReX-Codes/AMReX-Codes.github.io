48template <typename T, template<typename> class Allocator>
49SpMatrix<T,Allocator> transpose (SpMatrix<T,Allocator> const& A,
50 AlgPartition const& col_partition)
51{
52 BL_PROFILE("SpMatrix::transpose");
53
54 auto& Am = const_cast<SpMatrix<T,Allocator>&>(A);
55 Am.setColumnPartition(col_partition);
56 Am.startComm_tr(col_partition);
57
58 // The transpose is built directly in split form: its local block is
59 // the transpose of A's local block (columns are A's local row
60 // indices), and its remote block comes from the other ranks.
61 SpMatrix<T,Allocator> AT;
62 AT.m_row_begin = col_partition[ParallelDescriptor::MyProc()];
63 AT.m_row_end = col_partition[ParallelDescriptor::MyProc()+1];
64 AT.m_partition = col_partition;
65 AT.m_nnz = A.m_csr_local.nnz;
66 AT.m_csr_local = transpose<int>(A.m_csr_local, AT.numLocalRows());
67 AT.m_col_partition = A.partition();
68 AT.m_col_begin = A.globalRowBegin();
69 AT.m_col_end = A.globalRowEnd();
70 AT.m_split = true;
71#ifdef AMREX_USE_MPI
72 AT.m_ri_ltor.resize(AT.numLocalRows(), -1);
73 AT.m_remote_cols_vv.resize(ParallelDescriptor::NProcs());
74#endif
75
76 Am.finishComm_tr(AT);
77
78 return AT;
79}

107template <typename T, template<typename> class Allocator = DefaultAllocator>
108SpMatrix<T,Allocator> RandomMatrix (AlgPartition partition, Long nrows, Long ncols,
109 Real lambda, int nnz_per_row_max)
110{
111 AMREX_ALWAYS_ASSERT(partition.numGlobalRows() == nrows);
112
113 auto nlrows = partition.numLocalRows();
114 Long lnnz = nlrows * nnz_per_row_max;
115 typename SpMatrix<T,Allocator>::template container_type<T> mat(lnnz);
116 typename SpMatrix<T,Allocator>::template container_type<Long> col_index(lnnz);
117 typename SpMatrix<T,Allocator>::template container_type<Long> row_offset(nlrows+1, 0);
118 auto* pmat = mat.data();
119 auto* pcol = col_index.data();
120 auto* prow = row_offset.data();
121
122 auto ncols_max = (unsigned int)
123 std::min(ncols, Long(std::numeric_limits<unsigned int>::max()));
124 // A row cannot have more distinct columns than ncols.
125 auto nnz_row_cap = int(std::min(Long(nnz_per_row_max), ncols));
126
127 ParallelForRNG(nlrows, [=] AMREX_GPU_DEVICE (Long lrow, RandomEngine const& eng)
128 {
129 auto os = lrow * nnz_per_row_max;
130 prow[lrow] = os;
131 if (lrow+1 == nlrows) {
132 prow[lrow+1] = lnnz;
133 }
134
135 auto nnz_row = int(std::min(Long(amrex::RandomPoisson(lambda, eng)),
136 Long(nnz_row_cap)));
137
138 for (int j = 0; j < nnz_row; ++j) {
139 pmat[os+j] = Random(eng);
140 while (true) {
141 auto c = Long(Random_int(ncols_max, eng));
142 bool new_c = true;
143 for (int jj = 0; jj < j; ++jj) {
144 new_c = new_c && (c != pcol[os+jj]);
145 }
146 if (new_c) {
147 pcol[os+j] = c;
148 break;
149 }
150 }
151 }
152
153 for (int j = nnz_row; j < nnz_per_row_max; ++j) {
154 pcol[os+j] = -1;
155 pmat[os+j] = T(0);
156 }
157 });
158
159 SpMatrix<T,Allocator> r;
160 r.define(std::move(partition), pmat, pcol, lnnz, prow,
161 CsrSorted{false}, CsrValid{false});
162
163 return r;
164}

188template <typename T, template<typename> class Allocator = DefaultAllocator>
189bool almostEqual (SpMatrix<T,Allocator> const& A, SpMatrix<T,Allocator> const& B,
190 int ulp = 2)
191{
192 auto const& col_partition_A = A.columnPartition();
193 auto const& col_partition_B = B.columnPartition();
194 if (col_partition_A.empty() && ! col_partition_B.empty()) {
195 const_cast<SpMatrix<T,Allocator>&>(A).setColumnPartition(col_partition_B);
196 } else if (! col_partition_A.empty() && col_partition_B.empty()) {
197 const_cast<SpMatrix<T,Allocator>&>(B).setColumnPartition(col_partition_A);
198 } else if (col_partition_A.empty() && col_partition_B.empty()) {
199 // Neither has been split: compare the unsplit CSR with global columns.
200 auto& a = const_cast<SpMatrix<T,Allocator>&>(A);
201 auto& b = const_cast<SpMatrix<T,Allocator>&>(B);
202 Long const nnz = a.numLocalNonZeros();
203 Long const nrows = a.numLocalRows();
204 if (a.globalRowBegin() != b.globalRowBegin() ||
205 nnz != b.numLocalNonZeros() || nrows != b.numLocalRows())
206 {
207 return false;
208 }
209 T const* amat = a.data();
210 T const* bmat = b.data();
211 Long const* acol = a.columnIndex();
212 Long const* bcol = b.columnIndex();
213 Long const* arow = a.rowOffset();
214 Long const* brow = b.rowOffset();
215 Long N = std::max(nnz, nrows);
216 auto r = Reduce::Min<int>(N, [=] AMREX_GPU_DEVICE (Long i) {
217 bool t = true;
218 if (i < nnz) {
219 t = amrex::almostEqual(amat[i], bmat[i], ulp) && acol[i] == bcol[i];
220 }
221 if (t && i < nrows) {
222 t = arow[i] == brow[i];
223 }
224 return int(t);
225 }, 1);
226 return r;
227 }
228
229 auto const& pca = A.const_parcsr();
230 auto const& pcb = B.const_parcsr();
231
232 if (pca.row_begin != pcb.row_begin ||
233 pca.col_begin != pcb.col_begin ||
234 pca.csr0.nnz != pcb.csr0.nnz ||
235 pca.csr0.nrows != pcb.csr0.nrows ||
236 pca.csr1.nnz != pcb.csr1.nnz ||
237 pca.csr1.nrows != pcb.csr1.nrows)
238 {
239 return false;
240 }
241
242 Long N = std::max({pca.csr0.nnz, pca.csr0.nrows, pca.csr1.nnz, pca.csr1.nrows});
243 auto r = Reduce::Min<int>(N, [=] AMREX_GPU_DEVICE (Long i) {
244 bool t = true;
245 if (i < pca.csr0.nnz) {
246 t = amrex::almostEqual(pca.csr0.mat[i], pcb.csr0.mat[i], ulp) &&
247 pca.csr0.col_index[i] == pcb.csr0.col_index[i];
248 }
249 if (t && i < pca.csr0.nrows) {
250 t = pca.csr0.row_offset[i] == pcb.csr0.row_offset[i];
251 // Which local rows have remote entries.
252 if (t && pca.csr1.nnz > 0) {
253 t = pca.row_map[i] == pcb.row_map[i];
254 }
255 }
256 if (t && i < pca.csr1.nnz) {
257 // Remote column indices are local; compare the global ones.
258 t = amrex::almostEqual(pca.csr1.mat[i], pcb.csr1.mat[i], ulp) &&
259 pca.col_map[pca.csr1.col_index[i]] == pcb.col_map[pcb.csr1.col_index[i]];
260 }
261 if (t && i < pca.csr1.nrows) {
262 t = pca.csr1.row_offset[i] == pcb.csr1.row_offset[i];
263 }
264 return int(t);
265 }, 1);
266
267 return r;
268}
