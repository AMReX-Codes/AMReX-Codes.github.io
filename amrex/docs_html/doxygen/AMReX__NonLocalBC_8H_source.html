48struct MultiBlockIndexMapping {
50 IntVect permutation{AMREX_D_DECL(0, 1, 2)};
52 IntVect offset{AMREX_D_DECL(0, 0, 0)};
54 IntVect sign{AMREX_D_DECL(1, 1, 1)};
55
62 [[nodiscard]] AMREX_GPU_HOST_DEVICE Dim3 operator()(Dim3 i) const noexcept {
63 int iv[3]{i.x, i.y, i.z};
64 int iv_new[3]{};
65 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
66 iv_new[d] = sign[d] * (iv[permutation[d]] - offset[d]);
67 }
68 return Dim3{.x = iv_new[0], .y = iv_new[1], .z = iv_new[2]};
69 }
70
77 [[nodiscard]] AMREX_GPU_HOST_DEVICE Dim3 Inverse(Dim3 i) const noexcept {
78 int iv_new[3]{i.x, i.y, i.z};
79 int iv[3]{};
80 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
81 AMREX_ASSERT(sign[d] == 1 || sign[d] == -1);
82 iv[permutation[d]] = iv_new[d] * sign[d] + offset[d];
83 }
84 return Dim3{.x = iv[0], .y = iv[1], .z = iv[2]};
85 }
86
87 [[nodiscard]] IndexType operator()(IndexType it) const noexcept {
88 return IndexType{IntVect{AMREX_D_DECL(it[permutation[0]], it[permutation[1]], it[permutation[2]])}};
89 }

91 [[nodiscard]] IndexType Inverse(IndexType it) const noexcept {
92 IntVect inverse_permutation;
93 for (int i = 0; i < AMREX_SPACEDIM; ++i) {
94 inverse_permutation[permutation[i]] = i;
95 }
96 return IndexType{IntVect{AMREX_D_DECL(it[inverse_permutation[0]], it[inverse_permutation[1]], it[inverse_permutation[2]])}};

132Box
133Image (DTOS const& dtos, const Box& box)
134{
135 IntVect mapped_smallEnd = Apply(dtos, box.smallEnd());
136 IntVect mapped_bigEnd = Apply(dtos, box.bigEnd());
137 IntVect smallEnd;
138 IntVect bigEnd;
139 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
140 smallEnd[d] = std::min(mapped_smallEnd[d], mapped_bigEnd[d]);
141 bigEnd[d] = std::max(mapped_smallEnd[d], mapped_bigEnd[d]);
142 }
143 return Box{smallEnd, bigEnd, box.ixType()};

1012 if (comm == MPI_COMM_NULL) { comm = ParallelContext::CommunicatorSub(); }
1013 int nprocs_comm;

1030 PrepareAlltoallvCounts(handler.recv, comm);
1031 PrepareAlltoallvCounts(handler.send, comm);

1035 }
1036

1038 PackSendBuffers(data_packing, src, handler.send);

1042 // it, but it was slower than the blocking call on the systems tested.
