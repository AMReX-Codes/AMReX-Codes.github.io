38struct MultiBlockIndexMapping {
40 IntVect permutation{AMREX_D_DECL(0, 1, 2)};
42 IntVect offset{AMREX_D_DECL(0, 0, 0)};
44 IntVect sign{AMREX_D_DECL(1, 1, 1)};
45
52 [[nodiscard]] AMREX_GPU_HOST_DEVICE Dim3 operator()(Dim3 i) const noexcept {
53 int iv[3]{i.x, i.y, i.z};
54 int iv_new[3]{};
55 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
56 iv_new[d] = sign[d] * (iv[permutation[d]] - offset[d]);
57 }
58 return Dim3{.x = iv_new[0], .y = iv_new[1], .z = iv_new[2]};
59 }
60
67 [[nodiscard]] AMREX_GPU_HOST_DEVICE Dim3 Inverse(Dim3 i) const noexcept {
68 int iv_new[3]{i.x, i.y, i.z};
69 int iv[3]{};
70 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
71 AMREX_ASSERT(sign[d] == 1 || sign[d] == -1);
72 iv[permutation[d]] = iv_new[d] * sign[d] + offset[d];
73 }
74 return Dim3{.x = iv[0], .y = iv[1], .z = iv[2]};
75 }
76
77 [[nodiscard]] IndexType operator()(IndexType it) const noexcept {
78 return IndexType{IntVect{AMREX_D_DECL(it[permutation[0]], it[permutation[1]], it[permutation[2]])}};
79 }
80
81 [[nodiscard]] IndexType Inverse(IndexType it) const noexcept {
82 IntVect inverse_permutation;
83 for (int i = 0; i < AMREX_SPACEDIM; ++i) {
84 inverse_permutation[permutation[i]] = i;
85 }
86 return IndexType{IntVect{AMREX_D_DECL(it[inverse_permutation[0]], it[inverse_permutation[1]], it[inverse_permutation[2]])}};

120template <typename DTOS>
121requires (IsCallableR<Dim3, DTOS, Dim3>::value && !IsCallableR<IndexType, DTOS, IndexType>::value)
122Box
123Image (DTOS const& dtos, const Box& box)
124{
125 IntVect mapped_smallEnd = Apply(dtos, box.smallEnd());
126 IntVect mapped_bigEnd = Apply(dtos, box.bigEnd());
127 IntVect smallEnd;
128 IntVect bigEnd;
129 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
130 smallEnd[d] = std::min(mapped_smallEnd[d], mapped_bigEnd[d]);
131 bigEnd[d] = std::max(mapped_smallEnd[d], mapped_bigEnd[d]);
132 }
133 return Box{smallEnd, bigEnd, box.ixType()};
134}

783 // Otherwise sequence numbers will not match across MPI processes.
784 //

954template <BaseFabType FAB, typename DataPacking>
955requires (IsDataPacking<DataPacking, FAB>::value)
956void
957ParallelCopy_alltoallv (FabArray<FAB>& dest, const FabArray<FAB>& src,
958 const FabArrayBase::CommMetaData& cmd, const DataPacking& data_packing,
959 CommHandler* cached = nullptr, MPI_Comm comm = MPI_COMM_NULL)
960{
961 if (dest.empty()) { return; }
962
963#ifdef AMREX_USE_MPI
964 if (comm == MPI_COMM_NULL) { comm = ParallelContext::CommunicatorSub(); }
965 int nprocs_comm;
966 MPI_Comm_size(comm, &nprocs_comm);
967 if (nprocs_comm > 1)
968 {
969 CommHandler local_handler{};
970 CommHandler& handler = cached ? *cached : local_handler;
971
972 static const FabArrayBase::MapOfCopyComTagContainers no_tags{};
973 auto const& rcv_tags = cmd.m_RcvTags ? *cmd.m_RcvTags : no_tags;
974 auto const& snd_tags = cmd.m_SndTags ? *cmd.m_SndTags : no_tags;
975
976 // PrepareCommBuffers sets the id even when there is nothing to communicate.
977 if (handler.send.id == std::numeric_limits<std::uint64_t>::max()) {
978 PrepareRecvBuffers(data_packing, dest, src, handler.recv, rcv_tags);
979 PrepareSendBuffers(data_packing, dest, src, handler.send, snd_tags);
980 TranslateRanks(handler.recv, comm);
981 TranslateRanks(handler.send, comm);
982 PrepareAlltoallvCounts(handler.recv, comm);
983 PrepareAlltoallvCounts(handler.send, comm);
984 } else {
985 AcquireCommBuffer(handler.recv);
986 AcquireCommBuffer(handler.send);
987 }
988
989 if (!snd_tags.empty()) {
990 PackSendBuffers(data_packing, src, handler.send);
991 }
992
993 // The local copy comes first. MPI_Ialltoallv would allow overlapping
994 // it, but it was slower than the blocking call on the systems tested.
995 if (cmd.m_LocTags && !cmd.m_LocTags->empty()) {
996 LocalCopy(data_packing, dest, src, *cmd.m_LocTags);
997 }
998
999 Alltoallv(handler.send, handler.recv, comm);
1000
1001 if (!rcv_tags.empty()) {
1002 UnpackRecvBuffers(data_packing, dest, handler.recv);
1003 }
1004 if (cached) {
1005 ReleaseCommBuffer(handler.send);
1006 ReleaseCommBuffer(handler.recv);
1007 }
1008 return;
1009 }
1010#else
1011 amrex::ignore_unused(cached, comm);
1012#endif
1013
1014 if (cmd.m_LocTags && !cmd.m_LocTags->empty()) {
1015 LocalCopy(data_packing, dest, src, *cmd.m_LocTags);
1016 }
