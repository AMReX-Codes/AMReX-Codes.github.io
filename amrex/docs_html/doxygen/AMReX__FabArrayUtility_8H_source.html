1348template <BaseFabType FAB>
1349void
1350Swap (FabArray<FAB>& dst, FabArray<FAB>& src, int srccomp, int dstcomp, int numcomp, const IntVect& nghost)
1351{
1352 // We can take a shortcut and do a std::swap if we're swapping all of the data
1353 // and they are allocated in the same Arena.
1354
1355 bool explicit_swap = true;
1356
1357 if (srccomp == dstcomp && dstcomp == 0 && src.nComp() == dst.nComp() &&
1358 src.nGrowVect() == nghost && src.nGrowVect() == dst.nGrowVect() &&
1359 src.arena() == dst.arena() && src.hasEBFabFactory() == dst.hasEBFabFactory()) {
1360 explicit_swap = false;
1361 }
1362
1363 if (!explicit_swap) {
1364
1365 std::swap(dst, src);
1366
1367 } else {
1368#ifdef AMREX_USE_GPU
1369 if (Gpu::inLaunchRegion() && dst.isFusingCandidate()) {
1370 auto const& dstma = dst.arrays();
1371 auto const& srcma = src.arrays();
1372 ParallelFor(dst, nghost, numcomp,
1373 [=] AMREX_GPU_DEVICE (int box_no, int i, int j, int k, int n) noexcept
1374 {
1375 const auto tmp = dstma[box_no](i,j,k,n+dstcomp);
1376 dstma[box_no](i,j,k,n+dstcomp) = srcma[box_no](i,j,k,n+srccomp);
1377 srcma[box_no](i,j,k,n+srccomp) = tmp;
1378 });
1379 if (!Gpu::inNoSyncRegion()) {
1380 Gpu::streamSynchronize();
1381 }
1382 } else
1383#endif
1384 {
1385#ifdef AMREX_USE_OMP
1386#pragma omp parallel if (Gpu::notInLaunchRegion())
1387#endif
1388 for (MFIter mfi(dst,TilingIfNotGPU()); mfi.isValid(); ++mfi)
1389 {
1390 const Box& bx = mfi.growntilebox(nghost);
1391 if (bx.ok()) {
1392 auto sfab = src.array(mfi);
1393 auto dfab = dst.array(mfi);
1394 AMREX_HOST_DEVICE_PARALLEL_FOR_4D( bx, numcomp, i, j, k, n,
1395 {
1396 const auto tmp = dfab(i,j,k,n+dstcomp);
1397 dfab(i,j,k,n+dstcomp) = sfab(i,j,k,n+srccomp);
1398 sfab(i,j,k,n+srccomp) = tmp;
1399 });
1400 }
1401 }
1402 }
1403 }

1780template <BaseFabType FAB>
1781IntVect
1782indexFromValue (FabArray<FAB> const& mf, int comp, IntVect const& nghost,
1783 typename FAB::value_type value)
1784{
1785 IntVect loc = IntVect::TheMinVector();
1786
1787#ifdef AMREX_USE_GPU
1788 if (Gpu::inLaunchRegion())
1789 {
1790 constexpr int nofind = std::numeric_limits<int>::lowest();
1791 amrex::Gpu::Buffer<int> aa({0,AMREX_D_DECL(nofind,nofind,nofind)});
1792 int* p = aa.data();
1793 // This is a device ptr to 1+AMREX_SPACEDIM ints. The first is zero
1794 // and used as an atomic bool. The others hold the intvect and start
1795 // at the lowest int so that they stay TheMinVector if there is no
1796 // match.
1797 if (mf.isFusingCandidate()) {
1798 auto const& ma = mf.const_arrays();
1799 ParallelFor(mf, nghost, [=] AMREX_GPU_DEVICE (int box_no, int i, int j, int k) noexcept
1800 {
1801 int* flag = p;
1802 if (*flag == 0) {
1803 if (ma[box_no](i,j,k,comp) == value) {
1804 if (Gpu::Atomic::Exch(flag,1) == 0) {
1805 AMREX_D_TERM(p[1] = i;,
1806 p[2] = j;,
1807 p[3] = k;);
1808 }
1809 }
1810 }
1811 });
1812 } else {
1813 for (MFIter mfi(mf,MFItInfo().SetDeviceSync(false)); mfi.isValid(); ++mfi) {
1814 const Box& bx = amrex::grow(mfi.validbox(), nghost);
1815 auto const& arr = mf.const_array(mfi);
1816 amrex::ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k) noexcept
1817 {
1818 int* flag = p;
1819 if (*flag == 0) {
1820 if (arr(i,j,k,comp) == value) {
1821 if (Gpu::Atomic::Exch(flag,1) == 0) {
1822 AMREX_D_TERM(p[1] = i;,
1823 p[2] = j;,
1824 p[3] = k;);
1825 }
1826 }
1827 }
1828 });
1829 }
1830 }
1831 int const* tmp = aa.copyToHost();
1832 AMREX_D_TERM(loc[0] = tmp[1];,
1833 loc[1] = tmp[2];,
1834 loc[2] = tmp[3];);
1835 }
1836 else
1837#endif
1838 {
1839 bool f = false;
1840#ifdef AMREX_USE_OMP
1841#pragma omp parallel
1842#endif
1843 {
1844 IntVect priv_loc = IntVect::TheMinVector();
1845 for (MFIter mfi(mf,true); mfi.isValid(); ++mfi)
1846 {
1847 const Box& bx = mfi.growntilebox(nghost);
1848 auto const& fab = mf.const_array(mfi);
1849 AMREX_LOOP_3D(bx, i, j, k,
1850 {
1851 if (fab(i,j,k,comp) == value) {
1852 priv_loc = IntVect(AMREX_D_DECL(i,j,k));
1853 }
1854 });
1855 }
1856
1857 if (priv_loc.allGT(IntVect::TheMinVector())) {
1858 bool old;
1859// we should be able to test on _OPENMP < 201107 for capture (version 3.1)
1860// but we must work around a bug in gcc < 4.9
1861// And, with NVHPC 21.9 to <23.1, we saw an ICE with the atomic capture (NV bug: #3390723)
1862#if defined(AMREX_USE_OMP) && defined(_OPENMP) && (_OPENMP < 201307 || (defined(__NVCOMPILER) && __NVCOMPILER_MAJOR__ < 23)) // OpenMP 4.0
1863#pragma omp critical (amrex_indexfromvalue)
1864#elif defined(AMREX_USE_OMP)
1865#pragma omp atomic capture
1866#endif
1867 {
1868 old = f;
1869 f = true;
1870 }
1871
1872 if (old == false) { loc = priv_loc; }
1873 }
1874 }
1875 }
1876
1877 return loc;
1878}
