38// Sorts the n entries of a row by column; `tmp` is scratch for long rows.
39template <typename T, typename I>
40void sort_row_cpu (I* AMREX_RESTRICT col, T* AMREX_RESTRICT val, Long n,
41 Vector<std::pair<I,T>>& tmp)
42{
43 if (n <= 32) {
44 for (Long k = 1; k < n; ++k) {
45 I const c = col[k];
46 T const v = val[k];
47 Long m = k;
48 for (; m > 0 && col[m-1] > c; --m) {
49 col[m] = col[m-1];
50 val[m] = val[m-1];
51 }
52 col[m] = c;
53 val[m] = v;
54 }
55 } else {
56 tmp.resize(n);
57 for (Long k = 0; k < n; ++k) { tmp[k] = {col[k], val[k]}; }
58 std::sort(tmp.begin(), tmp.end(),
59 [] (auto const& x, auto const& y) { return x.first < y.first; });
60 for (Long k = 0; k < n; ++k) { col[k] = tmp[k].first; val[k] = tmp[k].second; }
61 }
62}

74template <typename T, template<typename> class V, typename I, typename W, typename F>
75CSR<T,V,I> csr_from_rows_cpu (Long nrows, W const& work, F const& make_row)
76{
77 CSR<T,V,I> C;
78 C.row_offset.resize(nrows+1);
79 I* AMREX_RESTRICT crow = C.row_offset.data();
80 AMREX_ASSUME(crow != nullptr); // gcc -Wnull-dereference false positive
81 crow[0] = 0;
82
83 int const nblocks = int(std::max(Long(1), std::min(Long(OpenMP::get_max_threads()), nrows)));
84 // Built with push_back: gcc -Wnull-dereference flags writes through data().
85 Vector<Long> rbegin{0};
86 rbegin.reserve(nblocks+1);
87 if (nblocks > 1) {
88 Vector<Long> wsum{0};
89 wsum.reserve(nrows+1);
90 for (Long i = 0; i < nrows; ++i) { wsum.push_back(wsum.back() + work(i)); }
91 for (int t = 1; t < nblocks; ++t) {
92 rbegin.push_back(std::lower_bound(wsum.begin(), wsum.end(), wsum.back()/nblocks*t)
93 - wsum.begin());
94 }
95 }
96 rbegin.push_back(nrows);
97
98 if (nblocks == 1) {
99 // Rows go straight into the result. The capacity is a guess from
100 // `work`; pages that are never written are never touched.
101 Long cap = 0;
102 for (Long i = 0; i < nrows; ++i) { cap += work(i); }
103 C.col_index.resize(cap);
104 C.mat.resize(cap);
105 auto row = make_row(0);
106 Long p = 0;
107 for (Long i = 0; i < nrows; ++i) {
108 I const* rc = nullptr;
109 T const* rv = nullptr;
110 Long const n = row(i, rc, rv);
111 if (p + n > Long(C.col_index.size())) {
112 Long const newcap = std::max(p + n, Long(C.col_index.size())*3/2);
113 C.col_index.resize(p); // only the filled part is copied
114 C.mat.resize(p);
115 C.col_index.resize(newcap);
116 C.mat.resize(newcap);
117 }
118 std::copy(rc, rc+n, C.col_index.data()+p);
119 std::copy(rv, rv+n, C.mat.data()+p);
120 p += n;
121 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(p <= Long(std::numeric_limits<I>::max()),
122 "SpGEMM: too many nonzeros for the index type");
123 crow[i+1] = I(p);
124 }
125 C.col_index.resize(p);
126 C.mat.resize(p);
127 C.nnz = p;
128 return C;
129 }

131 struct Buffer {
132 V<I> col;
133 V<T> val;
134 Long n = 0;
135 };
136 Vector<Vector<Buffer>> buffers(nblocks);
137 constexpr Long buffer_size = Long(1) << 20;
138
139#ifdef AMREX_USE_OMP
140#pragma omp parallel for schedule(static,1)
141#endif
142 for (int t = 0; t < nblocks; ++t) {
143 auto row = make_row(t);
144 auto& bufs = buffers[t];
145 for (Long i = rbegin[t]; i < rbegin[t+1]; ++i) {
146 I const* rc = nullptr;
147 T const* rv = nullptr;
148 Long const n = row(i, rc, rv);
149 if (bufs.empty() || bufs.back().n + n > Long(bufs.back().col.size())) {
150 bufs.emplace_back();
151 bufs.back().col.resize(std::max(buffer_size, n));
152 bufs.back().val.resize(std::max(buffer_size, n));
153 }
154 auto& buf = bufs.back();
155 std::copy(rc, rc+n, buf.col.data()+buf.n);
156 std::copy(rv, rv+n, buf.val.data()+buf.n);
157 buf.n += n;
158 crow[i+1] = I(n);
159 }
160 }
161
162 // Exclusive scan in Long: the result may not fit the index type.
163 Long total = 0;
164 for (Long i = 0; i < nrows; ++i) {
165 Long const cnt = crow[i+1];
166 crow[i] = I(total);
167 total += cnt;
168 }
169 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(total <= Long(std::numeric_limits<I>::max()),
170 "SpGEMM: too many nonzeros for the index type");
171 crow[nrows] = I(total);
172 C.nnz = total;
173 C.col_index.resize(C.nnz);
174 C.mat.resize(C.nnz);
175 I* AMREX_RESTRICT ccol = C.col_index.data();
176 T* AMREX_RESTRICT cmat = C.mat.data();
177
178#ifdef AMREX_USE_OMP
179#pragma omp parallel for schedule(static,1)
180#endif
181 for (int t = 0; t < nblocks; ++t) {
182 Long p = crow[rbegin[t]];
183 for (auto& buf : buffers[t]) {
184 std::copy(buf.col.data(), buf.col.data()+buf.n, ccol+p);
185 std::copy(buf.val.data(), buf.val.data()+buf.n, cmat+p);
186 p += buf.n;
187 buf = Buffer{};
188 }
189 }
190
191 return C;
192}

194// Gustavson's algorithm with a dense marker holding the position of each
195// column in the current row. Output rows are sorted, then passed to
196// `row_post(i, col, val, n)`, which may change them in place and returns
197// the new length.
198template <typename T, template<typename> class V, typename I, typename F = NoRowPost>
199CSR<T,V,I> spgemm_local_cpu (Long nrows, Long ncols,
200 CsrView<T const,I> const& A, CsrView<T const,I> const& B,
201 F const& row_post = {})
202{
203 auto nprod = [&] (Long i) {
204 Long n = 0;
205 for (Long ap = A.row_offset[i]; ap < A.row_offset[i+1]; ++ap) {
206 Long const k = A.col_index[ap];
207 n += B.row_offset[k+1] - B.row_offset[k];
208 }
209 return n;
210 };
211 return csr_from_rows_cpu<T,V,I>(nrows, nprod, [&] (int) {
212 return [&, marker = Vector<I>(ncols, I(-1)), col = Vector<I>(), val = Vector<T>(),
213 tmp = Vector<std::pair<I,T>>()]
214 (Long i, I const*& rc, T const*& rv) mutable -> Long
215 {
216 Long const maxlen = std::min(nprod(i), ncols);
217 if (Long(col.size()) < maxlen) {
218 col.resize(maxlen);
219 val.resize(maxlen);
220 }
221 Long n = 0;
222 for (Long ap = A.row_offset[i]; ap < A.row_offset[i+1]; ++ap) {
223 Long const k = A.col_index[ap];
224 T const a = A.mat[ap];
225 for (Long bp = B.row_offset[k]; bp < B.row_offset[k+1]; ++bp) {
226 I const j = B.col_index[bp];
227 Long const m = marker[j]; // may be left over from an earlier row
228 if (m >= 0 && m < n && col[m] == j) {
229 val[m] += a * B.mat[bp];
230 } else {
231 marker[j] = I(n);
232 col[n] = j;
233 val[n] = a * B.mat[bp];
234 ++n;
235 }
236 }
237 }
238 sort_row_cpu(col.data(), val.data(), n, tmp);
239 n = row_post(i, col.data(), val.data(), n);
240 rc = col.data();
241 rv = val.data();
242 return n;
243 };
244 });
245}

258template <typename T, template<typename> class V, typename I, typename F>
259CSR<T,V,I> spgemm_split_cpu (Long nrows, Long ncols, Long nb,
260 CsrView<T const,I> const& A0, CsrView<T const,I> const& A1,
261 CsrView<T const,I> const& B0, CsrView<T const,I> const& B1,
262 I const* b1map, CsrView<T const,I> const& E, F const& row_post)
263{
264 auto blen = [&] (Long k) -> Long {
265 if (k < nb) {
266 return (B0.row_offset[k+1] - B0.row_offset[k])
267 + ((B1.nnz > 0) ? (B1.row_offset[k+1] - B1.row_offset[k]) : 0);
268 } else {
269 return E.row_offset[k-nb+1] - E.row_offset[k-nb];
270 }
271 };
272 auto nprod = [&] (Long i) {
273 Long n = 0;
274 for (Long ap = A0.row_offset[i]; ap < A0.row_offset[i+1]; ++ap) { n += blen(A0.col_index[ap]); }
275 if (A1.nnz > 0) {
276 for (Long ap = A1.row_offset[i]; ap < A1.row_offset[i+1]; ++ap) { n += blen(nb + A1.col_index[ap]); }
277 }
278 return n;
279 };

281 return [&, marker = Vector<I>(ncols, I(-1)), col = Vector<I>(), val = Vector<T>(),
282 tmp = Vector<std::pair<I,T>>()]
283 (Long i, I const*& rc, T const*& rv) mutable -> Long
284 {
285 Long const maxlen = std::min(nprod(i), ncols);
286 if (Long(col.size()) < maxlen) {
287 col.resize(maxlen);
288 val.resize(maxlen);
289 }
290 Long n = 0;
291 auto add = [&] (I j, T x) {
292 Long const m = marker[j]; // may be left over from an earlier row
293 if (m >= 0 && m < n && col[m] == j) {
294 val[m] += x;
295 } else {
296 marker[j] = I(n);
297 col[n] = j;
298 val[n] = x;
299 ++n;
300 }
301 };
302 auto brow = [&] (Long k, T a) {
303 if (k < nb) {
304 for (Long bp = B0.row_offset[k]; bp < B0.row_offset[k+1]; ++bp) {
305 add(B0.col_index[bp], a * B0.mat[bp]);
306 }
307 if (B1.nnz > 0) {
308 for (Long bp = B1.row_offset[k]; bp < B1.row_offset[k+1]; ++bp) {
309 add(b1map[B1.col_index[bp]], a * B1.mat[bp]);
310 }
311 }
312 } else {
313 for (Long bp = E.row_offset[k-nb]; bp < E.row_offset[k-nb+1]; ++bp) {
314 add(E.col_index[bp], a * E.mat[bp]);
315 }
316 }
317 };
318 for (Long ap = A0.row_offset[i]; ap < A0.row_offset[i+1]; ++ap) {
319 brow(A0.col_index[ap], A0.mat[ap]);
320 }
321 if (A1.nnz > 0) {
322 for (Long ap = A1.row_offset[i]; ap < A1.row_offset[i+1]; ++ap) {
323 brow(nb + A1.col_index[ap], A1.mat[ap]);
324 }
325 }
326 sort_row_cpu(col.data(), val.data(), n, tmp);
327 n = row_post(i, col.data(), val.data(), n);
328 rc = col.data();
329 rv = val.data();

331 };

345template <typename T, typename I>
346struct APRowsCpu
347{
348 CsrView<T const,I> A0, A1, P0, P1, PE;
349 I const* p1map = nullptr;
350
351 [[nodiscard]] Long plen (Long k) const {
352 return (P0.row_offset[k+1] - P0.row_offset[k])
353 + ((P1.nnz > 0) ? (P1.row_offset[k+1] - P1.row_offset[k]) : 0);
354 }
355
357 [[nodiscard]] Long maxlen (Long i) const {
358 Long n = 0;
359 for (Long ap = A0.row_offset[i]; ap < A0.row_offset[i+1]; ++ap) { n += plen(A0.col_index[ap]); }
360 if (A1.nnz > 0) {
361 for (Long ap = A1.row_offset[i]; ap < A1.row_offset[i+1]; ++ap) {
362 Long const k = A1.col_index[ap];
363 n += PE.row_offset[k+1] - PE.row_offset[k];
364 }
365 }
366 return n;
367 }
368
371 Long row (Long i, I* marker, I* col, T* val, Long p) const {
372 Long const rs = p;
373 auto add = [&] (I j, T x) {
374 Long const m = marker[j]; // may be left over from elsewhere
375 if (m >= rs && m < p && col[m] == j) {
376 val[m] += x;
377 } else {
378 marker[j] = I(p);
379 col[p] = j;
380 val[p] = x;
381 ++p;
382 }
383 };
384 for (Long ap = A0.row_offset[i]; ap < A0.row_offset[i+1]; ++ap) {
385 Long const k = A0.col_index[ap];
386 T const a = A0.mat[ap];
387 for (Long pp = P0.row_offset[k]; pp < P0.row_offset[k+1]; ++pp) {
388 add(P0.col_index[pp], a * P0.mat[pp]);
389 }
390 if (P1.nnz > 0) {
391 for (Long pp = P1.row_offset[k]; pp < P1.row_offset[k+1]; ++pp) {
392 add(p1map[P1.col_index[pp]], a * P1.mat[pp]);
393 }
394 }
395 }
396 if (A1.nnz > 0) {
397 for (Long ap = A1.row_offset[i]; ap < A1.row_offset[i+1]; ++ap) {
398 Long const k = A1.col_index[ap];
399 T const a = A1.mat[ap];
400 for (Long pp = PE.row_offset[k]; pp < PE.row_offset[k+1]; ++pp) {
401 add(PE.col_index[pp], a * PE.mat[pp]);
402 }
403 }
404 }

419template <typename T, template<typename> class V, typename I>
420CSR<T,V,I> rap_local_cpu (Long nf, Long nc, Long ncols,
421 CsrView<T const,I> const& R0, CsrView<T const,I> const& R1,
422 CsrView<T const,I> const& APE, APRowsCpu<T,I> const& ap)
423{
424 constexpr Long chunk_rows = 1024;
425 Long const nchunks = (nf + chunk_rows - 1) / chunk_rows;
426
427 // About the length of row c: also the capacity guess of csr_from_rows_cpu.
428 auto work = [&] (Long c) {
429 Long n = 0;
430 for (Long rp = R0.row_offset[c]; rp < R0.row_offset[c+1]; ++rp) {
431 Long const i = R0.col_index[rp];
432 n += ap.A0.row_offset[i+1] - ap.A0.row_offset[i];
433 if (ap.A1.nnz > 0) { n += ap.A1.row_offset[i+1] - ap.A1.row_offset[i]; }
434 }
435 if (R1.nnz > 0) {
436 for (Long rp = R1.row_offset[c]; rp < R1.row_offset[c+1]; ++rp) {
437 Long const j = R1.col_index[rp];
438 n += APE.row_offset[j+1] - APE.row_offset[j];
439 }
440 }
441 return n;
442 };
443
444 // Last row of R that needs a chunk: the largest local column of P in it.
445 Vector<Long> chunk_last(nchunks, -1);
446 for (Long q = 0; q < nchunks; ++q) {
447 for (Long i = q*chunk_rows; i < std::min(nf, (q+1)*chunk_rows); ++i) {
448 if (ap.P0.row_offset[i+1] > ap.P0.row_offset[i]) {
449 chunk_last[q] = std::max(chunk_last[q],
450 Long(ap.P0.col_index[ap.P0.row_offset[i+1]-1]));
451 }
452 }
453 }
454
455 struct Chunk {
456 Vector<Long> off;
457 Vector<I> col;
458 Vector<T> val;
459 };
460
461 return csr_from_rows_cpu<T,V,I>(nc, work, [&] (int) {
462 return [&, cmarker = Vector<I>(ncols, I(-1)), marker = Vector<I>(ncols, I(-1)),
463 slot = Vector<int>(nchunks, -1), chunks = Vector<Chunk>(),
464 live = Vector<Long>(), free_slots = Vector<int>(),
465 next_release = std::numeric_limits<Long>::max(),
466 col = Vector<I>(), val = Vector<T>(), tmp = Vector<std::pair<I,T>>()]
467 (Long c, I const*& rc, T const*& rv) mutable -> Long
468 {
469 auto compute_chunk = [&] (Long q) {
470 int s;
471 if (free_slots.empty()) {
472 s = int(chunks.size());
473 chunks.emplace_back();
474 } else {
475 s = free_slots.back();
476 free_slots.pop_back();
477 }
478 slot[q] = s;
479 live.push_back(q);
480 next_release = std::min(next_release, chunk_last[q]);
481 auto& ch = chunks[s];
482 Long const i0 = q*chunk_rows;
483 Long const i1 = std::min(nf, i0+chunk_rows);
484 ch.off.resize(i1-i0+1);
485 ch.off[0] = 0;
486 Long p = 0;
487 for (Long i = i0; i < i1; ++i) {
488 Long const maxlen = ap.maxlen(i);
489 if (Long(ch.col.size()) < p + maxlen) {
490 ch.col.resize(std::max(p + maxlen, Long(ch.col.size())*2));
491 ch.val.resize(ch.col.size());
492 }
493 p = ap.row(i, cmarker.data(), ch.col.data(), ch.val.data(), p);
494 ch.off[i-i0+1] = p;
495 }
496 };
497
498 // Chunks no later row needs are recycled.
499 if (c > next_release) {
500 next_release = std::numeric_limits<Long>::max();
501 for (Long pos = 0; pos < Long(live.size()); ) {
502 Long const q = live[pos];
503 if (chunk_last[q] < c) {
504 free_slots.push_back(slot[q]);
505 slot[q] = -1;
506 live[pos] = live.back();
507 live.pop_back();
508 } else {
509 next_release = std::min(next_release, chunk_last[q]);
510 ++pos;
511 }
512 }
513 }
514
515 Long maxlen = 0;
516 for (Long rp = R0.row_offset[c]; rp < R0.row_offset[c+1]; ++rp) {
517 Long const i = R0.col_index[rp];
518 Long const q = i / chunk_rows;
519 if (slot[q] < 0) { compute_chunk(q); }
520 auto const& ch = chunks[slot[q]];
521 maxlen += ch.off[i-q*chunk_rows+1] - ch.off[i-q*chunk_rows];
522 }
523 if (R1.nnz > 0) {
524 for (Long rp = R1.row_offset[c]; rp < R1.row_offset[c+1]; ++rp) {
525 Long const j = R1.col_index[rp];
526 maxlen += APE.row_offset[j+1] - APE.row_offset[j];
527 }
528 }
529 maxlen = std::min(maxlen, ncols);
530 if (Long(col.size()) < maxlen) {
531 col.resize(maxlen);
532 val.resize(maxlen);
533 }
534
535 Long n = 0;
536 auto add = [&] (I j, T x) {
537 Long const m = marker[j]; // may be left over from an earlier row
538 if (m >= 0 && m < n && col[m] == j) {
539 val[m] += x;
540 } else {
541 marker[j] = I(n);
542 col[n] = j;
543 val[n] = x;
544 ++n;
545 }
546 };
547 for (Long rp = R0.row_offset[c]; rp < R0.row_offset[c+1]; ++rp) {
548 Long const i = R0.col_index[rp];
549 T const r = R0.mat[rp];
550 Long const q = i / chunk_rows;
551 auto const& ch = chunks[slot[q]];
552 for (Long x = ch.off[i-q*chunk_rows]; x < ch.off[i-q*chunk_rows+1]; ++x) {
553 add(ch.col[x], r * ch.val[x]);
554 }
555 }
556 if (R1.nnz > 0) {
557 for (Long rp = R1.row_offset[c]; rp < R1.row_offset[c+1]; ++rp) {
558 Long const j = R1.col_index[rp];
559 T const r = R1.mat[rp];
560 for (Long x = APE.row_offset[j]; x < APE.row_offset[j+1]; ++x) {
561 add(APE.col_index[x], r * APE.mat[x]);
562 }
563 }
564 }
565 sort_row_cpu(col.data(), val.data(), n, tmp);
566 rc = col.data();
567 rv = val.data();
568 return n;
569 };
570 });
571}

575template <typename T, template<typename> class V, typename I>
576CSR<T,V,I> spgemm_local_cusparse (Long nrows, Long ncols,
577 CsrView<T const,I> const& A, CsrView<T const,I> const& B)
578{
579 static_assert(std::is_same_v<I,int>, "spgemm_local_cusparse: 32-bit indices only");
580

582 AMREX_CUSPARSE_SAFE_CALL(cusparseCreate(&handle));
583 AMREX_CUSPARSE_SAFE_CALL(cusparseSetStream(handle, Gpu::gpuStream()));
584
585 cudaDataType data_type;

601 AMREX_ALWAYS_ASSERT(nrows < Long(std::numeric_limits<int>::max()) &&

603 constexpr cusparseIndexType_t index_type = CUSPARSE_INDEX_32I;
604 void* rowA = (void*)A.row_offset;
605 void* colA = (void*)A.col_index;
606 void* rowB = (void*)B.row_offset;
607 void* colB = (void*)B.col_index;
608 void* rowC = (void*)C.row_offset.data();

912 F0 const& map0, F1 const& map1)
913{
914 AMREX_ALWAYS_ASSERT(X0.nnz + X1.nnz < Long(std::numeric_limits<int>::max()));

931 ccol[p] = map1(X1.col_index[q]);
932 cmat[p] = X1.mat[q];
933 ++p;
934 }
935 }

940// Append rows with global column indices, sorted within each row, to Bh
941// whose first rows are already in the compact column space. Each row is
942// rotated so that the local block [c0,c1) comes first, which keeps it
943// sorted in the compact space [local | remote_union].
944template <typename T, template<typename> class V>
945void append_ext_rows (CSR<T,V,int>& Bh, Long const* ext_row_offset, Long const* ext_col,
946 T const* ext_mat, Long n_ext, Long nnz_ext,
947 Long c0, Long c1, Long const* ru, Long nru)
948{
949 Long const nb = Bh.nrows();
950 Long const nnz_b = Bh.nnz;
951 AMREX_ALWAYS_ASSERT(nnz_b + nnz_ext < Long(std::numeric_limits<int>::max()));
952 Bh.mat.resize(nnz_b + nnz_ext);
953 Bh.col_index.resize(nnz_b + nnz_ext);
954 Bh.row_offset.resize(nb + n_ext + 1);
955 Bh.nnz = nnz_b + nnz_ext;
956 int* AMREX_RESTRICT brow = Bh.row_offset.data();
957 int* AMREX_RESTRICT bcol = Bh.col_index.data();
958 T* AMREX_RESTRICT bmat = Bh.mat.data();
959 Long const nlocal = c1 - c0;
960 ParallelForOMP(n_ext, [=] AMREX_GPU_DEVICE (Long r)
961 {
962 Long const b = ext_row_offset[r];
963 Long const e = ext_row_offset[r+1];
964 brow[nb+r+1] = int(nnz_b + e);
965 Long const p0 = amrex::lower_bound(ext_col+b, ext_col+e, c0) - ext_col;
966 Long const p1 = amrex::lower_bound(ext_col+b, ext_col+e, c1) - ext_col;
967 Long p = nnz_b + b;
968 for (Long q = p0; q < p1; ++q) {
969 bcol[p] = int(ext_col[q] - c0);
970 bmat[p] = ext_mat[q];
971 ++p;
972 }
973 for (Long q = b; q < e; ++q) {
974 if (q < p0 || q >= p1) {
975 bcol[p] = int(nlocal + (amrex::lower_bound(ru, ru+nru, ext_col[q]) - ru));
976 bmat[p] = ext_mat[q];
977 ++p;
978 }
979 }
980 });
981}

1032template <typename T, template <typename> class Allocator, typename F>
1033SpMatrix<T,Allocator>
1034SpGEMM (SpMatrix<T,Allocator> const& A, SpMatrix<T,Allocator> const& B,
1035 AlgPartition const& col_partition, F const& row_post)
1036{
1037 BL_PROFILE("SpGEMM");
1038#ifdef AMREX_USE_GPU
1039 static_assert(std::is_same_v<F,detail::NoRowPost>, "SpGEMM: row_post is CPU only");
1040 amrex::ignore_unused(row_post);
1041#endif
1042
1043 using SpMat = SpMatrix<T,Allocator>;
1044
1045 auto& Am = const_cast<SpMat&>(A);
1046 auto& Bm = const_cast<SpMat&>(B);
1047 Long const nrows = Am.numLocalRows();
1048
1049#ifdef AMREX_USE_MPI
1050 using LongVec = typename SpMat::template container_type<Long>;
1051
1052 Am.setColumnPartition(Bm.partition());
1053 Bm.setColumnPartition(col_partition);
1054
1055 Long const nb = Bm.numLocalRows();
1056 Long const c0 = col_partition.globalRowBegin();
1057 Long const c1 = col_partition.globalRowEnd();
1058 Long const nlocal = c1 - c0;
1059
1060 // Rows of B needed by A's off-diagonal columns, global column indices.
1061 typename SpMat::RemoteRowsMM ext;
1062 if (! detail::spmat_comm_is_local(Am.partition(), Bm.partition())) {
1063 ext = Am.fetch_remote_rows_mm(Bm);
1064 }
1065
1066 if (ext.nrows == 0 && Am.m_remote_cols_v.empty() && Bm.m_remote_cols_v.empty()) {
1067 auto Ch = detail::spgemm_local<T,SpMat::template container_type,int>
1068 (nrows, nlocal, Am.m_csr_local.const_view(), Bm.m_csr_local.const_view(), row_post);
1069 SpMat C;
1070 C.define_split(Am.partition(), col_partition, std::move(Ch), nlocal, nullptr, 0);
1071 return C;
1072 }
1073
1074 // Compact column space: [local columns | sorted remote columns].
1075 // TODO: this serial host loop over ext.nnz limits scaling on GPUs.
1076 Vector<Long> ru_h = Bm.m_remote_cols_v;
1077 for (Long i = 0; i < ext.nnz; ++i) {
1078 auto g = ext.col_index[i];
1079 if (g < c0 || g >= c1) { ru_h.push_back(g); }
1080 }
1081 RemoveDuplicates(ru_h);
1082 Long const nru = Long(ru_h.size());
1083 LongVec ru_d(nru);
1084 Gpu::copyAsync(Gpu::hostToDevice, ru_h.begin(), ru_h.end(), ru_d.begin());
1085 Long const* ru = ru_d.data();
1086 Long const ncols_hat = nlocal + nru;
1087 AMREX_ALWAYS_ASSERT(ncols_hat < Long(std::numeric_limits<int>::max()) &&
1088 nb + Long(Am.m_remote_cols_v.size()) < Long(std::numeric_limits<int>::max()));
1089
1090 using local_csr_type = typename SpMat::local_csr_type;
1091#ifndef AMREX_USE_GPU
1092 {
1093 // The rows of B fetched from other processes, in the compact
1094 // column space; B's own remote columns are mapped once each.
1095 local_csr_type E;
1096 E.resize(0, 0);
1097 E.row_offset[0] = 0;
1098 if (ext.nrows > 0) {
1099 detail::append_ext_rows(E, ext.row_offset.data(), ext.col_index, ext.mat,
1100 ext.nrows, ext.nnz, c0, c1, ru, nru);
1101 }
1102 ext.clear();
