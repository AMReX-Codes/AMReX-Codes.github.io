30// Gustavson's algorithm. Pass 1 counts, pass 2 accumulates into a dense
31// per-thread accumulator. Output rows are sorted.
32template <typename T, template<typename> class V, typename I>
33CSR<T,V,I> spgemm_local_cpu (Long nrows, Long ncols,
34 CsrView<T const,I> const& A, CsrView<T const,I> const& B)
35{
36 CSR<T,V,I> C;
37 C.row_offset.resize(nrows+1);
38 Long* AMREX_RESTRICT crow = C.row_offset.data();
39 AMREX_ASSUME(crow != nullptr); // gcc -Wnull-dereference false positive
40 crow[0] = 0;
41
42#ifdef AMREX_USE_OMP
43#pragma omp parallel
44#endif
45 {
46 Vector<Long> marker(ncols, -1);
47#ifdef AMREX_USE_OMP
48#pragma omp for schedule(dynamic,64)
49#endif
50 for (Long i = 0; i < nrows; ++i) {
51 Long cnt = 0;
52 for (Long ap = A.row_offset[i]; ap < A.row_offset[i+1]; ++ap) {
53 Long const k = A.col_index[ap];
54 for (Long bp = B.row_offset[k]; bp < B.row_offset[k+1]; ++bp) {
55 Long const j = B.col_index[bp];
56 if (marker[j] != i) {
57 marker[j] = i;
58 ++cnt;
59 }
60 }
61 }
62 crow[i+1] = cnt;
63 }
64 }
65
66 for (Long i = 0; i < nrows; ++i) { crow[i+1] += crow[i]; }
67 C.nnz = crow[nrows];
68 C.col_index.resize(C.nnz);
69 C.mat.resize(C.nnz);
70 I* AMREX_RESTRICT ccol = C.col_index.data();
71 T* AMREX_RESTRICT cmat = C.mat.data();
72
73#ifdef AMREX_USE_OMP
74#pragma omp parallel
75#endif
76 {
77 Vector<Long> marker(ncols, -1);
78 Vector<T> acc(ncols);
79 Vector<I> cols;
80#ifdef AMREX_USE_OMP
81#pragma omp for schedule(dynamic,64)
82#endif
83 for (Long i = 0; i < nrows; ++i) {
84 cols.clear();
85 for (Long ap = A.row_offset[i]; ap < A.row_offset[i+1]; ++ap) {
86 Long const k = A.col_index[ap];
87 T const a = A.mat[ap];
88 for (Long bp = B.row_offset[k]; bp < B.row_offset[k+1]; ++bp) {
89 Long const j = B.col_index[bp];
90 if (marker[j] != i) {
91 marker[j] = i;
92 cols.push_back(j);
93 acc[j] = a * B.mat[bp];
94 } else {
95 acc[j] += a * B.mat[bp];
96 }
97 }
98 }
99 std::sort(cols.begin(), cols.end());
100 Long p = crow[i];
101 for (I j : cols) {
102 ccol[p] = j;
103 cmat[p] = acc[j];
104 ++p;
105 }
106 }
107 }
108
109 return C;
110}

114template <typename T, template<typename> class V, typename I>

119
120 cusparseHandle_t handle;
121 AMREX_CUSPARSE_SAFE_CALL(cusparseCreate(&handle));
122 AMREX_CUSPARSE_SAFE_CALL(cusparseSetStream(handle, Gpu::gpuStream()));
123
124 cudaDataType data_type;
125 if constexpr (std::is_same_v<T,float>) {
126 data_type = CUDA_R_32F;
127 } else if constexpr (std::is_same_v<T,double>) {
128 data_type = CUDA_R_64F;
129 } else if constexpr (std::is_same_v<T,GpuComplex<float>>) {
130 data_type = CUDA_C_32F;
131 } else if constexpr (std::is_same_v<T,GpuComplex<double>>) {
132 data_type = CUDA_C_64F;

137 CSR<T,V,I> C;

139
140 // cuSPARSE SpGEMM needs one index type: 32-bit copies of the row offsets.
141 AMREX_ALWAYS_ASSERT(nrows < Long(std::numeric_limits<int>::max()) &&
142 ncols < Long(std::numeric_limits<int>::max()) &&
143 A.nnz < Long(std::numeric_limits<int>::max()) &&
144 B.nnz < Long(std::numeric_limits<int>::max()));
145 constexpr cusparseIndexType_t index_type = CUSPARSE_INDEX_32I;
146 Gpu::DeviceVector<int> rowA32(nrows+1), rowB32(B.nrows+1), rowC32(nrows+1);
147 {
148 auto* pa = rowA32.data(); auto const* sa = A.row_offset;
149 auto* pb = rowB32.data(); auto const* sb = B.row_offset;
150 Long const na = nrows+1, nb = B.nrows+1;
151 ParallelFor(std::max(na,nb), [=] AMREX_GPU_DEVICE (Long i) {
152 if (i < na) { pa[i] = int(sa[i]); }
153 if (i < nb) { pb[i] = int(sb[i]); }
154 });
155 }
156 void* rowA = (void*)rowA32.data();
157 void* colA = (void*)A.col_index;
158 void* rowB = (void*)rowB32.data();
159 void* colB = (void*)B.col_index;
160 void* rowC = (void*)rowC32.data();
161
162 cusparseSpMatDescr_t mat_A, mat_B, mat_C;
163 AMREX_CUSPARSE_SAFE_CALL
164 (cusparseCreateCsr(&mat_A, nrows, B.nrows, A.nnz, rowA, colA, (void*)A.mat,
165 index_type, index_type, CUSPARSE_INDEX_BASE_ZERO, data_type));
166 AMREX_CUSPARSE_SAFE_CALL
167 (cusparseCreateCsr(&mat_B, B.nrows, ncols, B.nnz, rowB, colB, (void*)B.mat,
168 index_type, index_type, CUSPARSE_INDEX_BASE_ZERO, data_type));
169 AMREX_CUSPARSE_SAFE_CALL
170 (cusparseCreateCsr(&mat_C, nrows, ncols, 0, rowC, nullptr, nullptr,
171 index_type, index_type, CUSPARSE_INDEX_BASE_ZERO, data_type));
172
173 cusparseSpGEMMDescr_t spgemm_descr;
174 AMREX_CUSPARSE_SAFE_CALL(cusparseSpGEMM_createDescr(&spgemm_descr));
175
176 T alpha = T(1);
177 T beta = T(0);
178 cusparseOperation_t op = CUSPARSE_OPERATION_NON_TRANSPOSE;
179
180 // TODO: fall back to the chunked algorithm (CUSPARSE_SPGEMM_ALG3) when
181 // the work buffer does not fit in memory.
182 std::size_t buffer_size1 = 0;
183 AMREX_CUSPARSE_SAFE_CALL
184 (cusparseSpGEMM_workEstimation(handle, op, op, &alpha, mat_A, mat_B, &beta, mat_C,
185 data_type, CUSPARSE_SPGEMM_DEFAULT, spgemm_descr,
186 &buffer_size1, nullptr));
187 auto* buffer1 = (void*)The_Arena()->alloc(buffer_size1);
188 AMREX_CUSPARSE_SAFE_CALL
189 (cusparseSpGEMM_workEstimation(handle, op, op, &alpha, mat_A, mat_B, &beta, mat_C,
190 data_type, CUSPARSE_SPGEMM_DEFAULT, spgemm_descr,
191 &buffer_size1, buffer1));
192
193 std::size_t buffer_size2 = 0;
194 AMREX_CUSPARSE_SAFE_CALL
195 (cusparseSpGEMM_compute(handle, op, op, &alpha, mat_A, mat_B, &beta, mat_C,
196 data_type, CUSPARSE_SPGEMM_DEFAULT, spgemm_descr,
197 &buffer_size2, nullptr));
198 auto* buffer2 = (void*)The_Arena()->alloc(buffer_size2);
199 AMREX_CUSPARSE_SAFE_CALL
200 (cusparseSpGEMM_compute(handle, op, op, &alpha, mat_A, mat_B, &beta, mat_C,
201 data_type, CUSPARSE_SPGEMM_DEFAULT, spgemm_descr,
202 &buffer_size2, buffer2));
203
204 std::int64_t c_nrows, c_ncols, c_nnz;
205 AMREX_CUSPARSE_SAFE_CALL(cusparseSpMatGetSize(mat_C, &c_nrows, &c_ncols, &c_nnz));
206 AMREX_ALWAYS_ASSERT(c_nrows == nrows && c_ncols == ncols);
207 AMREX_ALWAYS_ASSERT(c_nnz < std::int64_t(std::numeric_limits<int>::max()));
208
209 C.mat.resize(c_nnz);
210 C.col_index.resize(c_nnz);
211 C.nnz = c_nnz;
212 void* colC = (void*)C.col_index.data();
213 AMREX_CUSPARSE_SAFE_CALL(cusparseCsrSetPointers(mat_C, rowC, colC, (void*)C.mat.data()));
214
215 // cuSPARSE guarantees sorted column indices in C.
216 AMREX_CUSPARSE_SAFE_CALL
217 (cusparseSpGEMM_copy(handle, op, op, &alpha, mat_A, mat_B, &beta, mat_C,
218 data_type, CUSPARSE_SPGEMM_DEFAULT, spgemm_descr));
219
220 {
221 auto* pc = C.row_offset.data(); auto const* sc = rowC32.data();
222 ParallelFor(nrows+1, [=] AMREX_GPU_DEVICE (Long i) { pc[i] = Long(sc[i]); });
223 }
224
225 Gpu::streamSynchronize();
226 AMREX_CUSPARSE_SAFE_CALL(cusparseSpGEMM_destroyDescr(spgemm_descr));
227 AMREX_CUSPARSE_SAFE_CALL(cusparseDestroySpMat(mat_A));
228 AMREX_CUSPARSE_SAFE_CALL(cusparseDestroySpMat(mat_B));
229 AMREX_CUSPARSE_SAFE_CALL(cusparseDestroySpMat(mat_C));
230 AMREX_CUSPARSE_SAFE_CALL(cusparseDestroy(handle));
231 The_Arena()->free(buffer1);
232 The_Arena()->free(buffer2);
233
234 return C;
235}

478template <typename T, template<typename> class V, typename I>
479CSR<T,V,I> spgemm_local (Long nrows, Long ncols,
480 CsrView<T const,I> const& A, CsrView<T const,I> const& B)
481{
482 AMREX_ASSERT(A.nrows == nrows);
483
484 if (nrows <= 0 || ncols <= 0 || A.nnz <= 0 || B.nnz <= 0 || B.nrows <= 0) {
485 return spgemm_empty<T,V,I>(nrows);
486 }
487

501// Row i of the result is row i of X0 followed by row i of X1, with column
502// indices transformed by map0 and map1. Both views must have full row offsets.
503template <typename T, template<typename> class V, typename I, typename F0, typename F1>
504CSR<T,V,int> concat_csr_cols (Long nrows, CsrView<T const,I> const& X0, CsrView<T const,I> const& X1,
505 F0 const& map0, F1 const& map1)
506{
507 CSR<T,V,int> C;
508 C.resize(nrows, X0.nnz + X1.nnz);
509 Long* AMREX_RESTRICT crow = C.row_offset.data();
510 int* AMREX_RESTRICT ccol = C.col_index.data();
511 T* AMREX_RESTRICT cmat = C.mat.data();
512 ParallelForOMP(nrows+1, [=] AMREX_GPU_DEVICE (Long i)
513 {
514 crow[i] = X0.row_offset[i] + X1.row_offset[i];
515 if (i < nrows) {
516 Long p = crow[i];
517 for (Long q = X0.row_offset[i]; q < X0.row_offset[i+1]; ++q) {
518 ccol[p] = map0(X0.col_index[q]);
519 cmat[p] = X0.mat[q];
520 ++p;
521 }
522 for (Long q = X1.row_offset[i]; q < X1.row_offset[i+1]; ++q) {
523 ccol[p] = map1(X1.col_index[q]);
524 cmat[p] = X1.mat[q];
525 ++p;
526 }
527 }
528 });
529 return C;
530}

532// Append rows with global column indices, sorted within each row, to Bh
533// whose first rows are already in the compact column space. Each row is
534// rotated so that the local block [c0,c1) comes first, which keeps it
535// sorted in the compact space [local | remote_union].
536template <typename T, template<typename> class V>
537void append_ext_rows (CSR<T,V,int>& Bh, Long const* ext_row_offset, Long const* ext_col,
538 T const* ext_mat, Long n_ext, Long nnz_ext,
539 Long c0, Long c1, Long const* ru, Long nru)
540{
541 Long const nb = Bh.nrows();
542 Long const nnz_b = Bh.nnz;
543 Bh.mat.resize(nnz_b + nnz_ext);
544 Bh.col_index.resize(nnz_b + nnz_ext);
545 Bh.row_offset.resize(nb + n_ext + 1);
546 Bh.nnz = nnz_b + nnz_ext;
547 Long* AMREX_RESTRICT brow = Bh.row_offset.data();
548 int* AMREX_RESTRICT bcol = Bh.col_index.data();
549 T* AMREX_RESTRICT bmat = Bh.mat.data();
550 Long const nlocal = c1 - c0;
551 ParallelForOMP(n_ext, [=] AMREX_GPU_DEVICE (Long r)
552 {
553 Long const b = ext_row_offset[r];
554 Long const e = ext_row_offset[r+1];
555 brow[nb+r+1] = nnz_b + e;
556 Long const p0 = amrex::lower_bound(ext_col+b, ext_col+e, c0) - ext_col;
557 Long const p1 = amrex::lower_bound(ext_col+b, ext_col+e, c1) - ext_col;
558 Long p = nnz_b + b;
559 for (Long q = p0; q < p1; ++q) {
560 bcol[p] = int(ext_col[q] - c0);
561 bmat[p] = ext_mat[q];
562 ++p;
563 }
564 for (Long q = b; q < e; ++q) {
565 if (q < p0 || q >= p1) {
566 bcol[p] = int(nlocal + (amrex::lower_bound(ru, ru+nru, ext_col[q]) - ru));
567 bmat[p] = ext_mat[q];
568 ++p;
569 }
570 }
571 });
572}

595template <typename T, template <typename> class Allocator>
596SpMatrix<T,Allocator>
597SpGEMM (SpMatrix<T,Allocator> const& A, SpMatrix<T,Allocator> const& B,
598 AlgPartition const& col_partition)
599{
600 BL_PROFILE("SpGEMM");
601
602 using SpMat = SpMatrix<T,Allocator>;
603
604 auto& Am = const_cast<SpMat&>(A);
605 auto& Bm = const_cast<SpMat&>(B);
606 Long const nrows = Am.numLocalRows();
607
608#ifdef AMREX_USE_MPI
609 using LongVec = typename SpMat::template container_type<Long>;
610
611 Am.setColumnPartition(Bm.partition());
612 Bm.setColumnPartition(col_partition);
613
614 Long const nb = Bm.numLocalRows();
615 Long const c0 = col_partition.globalRowBegin();
616 Long const c1 = col_partition.globalRowEnd();
617 Long const nlocal = c1 - c0;
618
619 // Rows of B needed by A's off-diagonal columns, global column indices.
620 typename SpMat::RemoteRowsMM ext;
621 if (! detail::spmat_comm_is_local(Am.partition(), Bm.partition())) {
622 ext = Am.fetch_remote_rows_mm(Bm);
623 }
624
625 // Compact column space: [local columns | sorted remote columns].
626 // TODO: this serial host loop over ext.nnz limits scaling on GPUs.
627 Vector<Long> ru_h = Bm.m_remote_cols_v;
628 for (Long i = 0; i < ext.nnz; ++i) {
629 auto g = ext.col_index[i];
630 if (g < c0 || g >= c1) { ru_h.push_back(g); }
631 }
632 RemoveDuplicates(ru_h);
633 Long const nru = Long(ru_h.size());
634 LongVec ru_d(nru);
635 Gpu::copyAsync(Gpu::hostToDevice, ru_h.begin(), ru_h.end(), ru_d.begin());
636 Long const* ru = ru_d.data();
637 Long const ncols_hat = nlocal + nru;
638 AMREX_ALWAYS_ASSERT(ncols_hat < Long(std::numeric_limits<int>::max()) &&
639 nb + Long(Am.m_remote_cols_v.size()) < Long(std::numeric_limits<int>::max()));
640
641 using local_csr_type = typename SpMat::local_csr_type;
642 local_csr_type Ah = detail::concat_csr_cols<T,SpMat::template container_type>
643 (nrows, Am.m_csr_local.const_view(), Am.remote_full_const_view(),
644 [=] AMREX_GPU_DEVICE (int c) { return c; },
645 [=] AMREX_GPU_DEVICE (int j) { return int(nb + j); });
646
647#ifdef AMREX_USE_GPU
648 Long const* b_rcols = Bm.m_remote_cols_dv.data();
649#else
650 Long const* b_rcols = Bm.m_remote_cols_v.data();
651#endif
652 local_csr_type Bh = detail::concat_csr_cols<T,SpMat::template container_type>
653 (nb, Bm.m_csr_local.const_view(), Bm.remote_full_const_view(),
654 [=] AMREX_GPU_DEVICE (int c) { return c; },
655 [=] AMREX_GPU_DEVICE (int j) {
656 return int(nlocal + (amrex::lower_bound(ru, ru+nru, b_rcols[j]) - ru)); });
657
658 if (ext.nrows > 0) {
659 LongVec ext_col_d(ext.nnz);
660 Gpu::copyAsync(Gpu::hostToDevice, ext.col_index, ext.col_index+ext.nnz,
661 ext_col_d.begin());
662 detail::append_ext_rows(Bh, ext.row_offset.data(), ext_col_d.data(), ext.mat,
663 ext.nrows, ext.nnz, c0, c1, ru, nru);
664 Gpu::streamSynchronize();
665 }
666 ext.clear();
667
668 local_csr_type Ch = detail::spgemm_local<T,SpMat::template container_type,int>
669 (nrows, ncols_hat, Ah.const_view(), Bh.const_view());
670 Ah = local_csr_type{};
671 Bh = local_csr_type{};
672
673 // The compact column space is [local | sorted remote], so the result
674 // can be defined directly in split form.
675 SpMat C;
676 C.define_split(Am.partition(), col_partition, std::move(Ch), nlocal, ru, nru);
677 return C;
678
679#else
680
681 Am.setColumnPartition(Bm.partition());
682 Bm.setColumnPartition(col_partition);
683
684 // One rank: local column indices are the global ones.
685 Long const ncols = col_partition.numGlobalRows();
686 AMREX_ALWAYS_ASSERT(ncols < Long(std::numeric_limits<int>::max()));
687 auto Ch = detail::spgemm_local<T,SpMat::template container_type,int>
688 (nrows, ncols, Am.m_csr_local.const_view(), Bm.m_csr_local.const_view());
689 SpMat C;
690 C.define_split(Am.partition(), col_partition, std::move(Ch), ncols, nullptr, 0);
691 return C;
692
693#endif
694}
