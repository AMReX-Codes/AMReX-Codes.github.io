60 void define (Long a_row_begin, Long a_nrows, Vector<Triplet> trips)
61 {
62 row_begin = a_row_begin;
63 nrows = a_nrows;
64 std::sort(trips.begin(), trips.end(), [] (Triplet const& a, Triplet const& b) {
65 return (a.row < b.row) || (a.row == b.row && a.col < b.col);
66 });
67 row_offset.assign(nrows+1, 0);
68 col_index.clear();
69 values.clear();
70 col_index.reserve(trips.size());
71 values.reserve(trips.size());
72 Long prev_row = -1, prev_col = -1;
73 for (auto const& t : trips) {
74 AMREX_ALWAYS_ASSERT(t.row >= row_begin && t.row < row_begin+nrows);
75 if (t.row == prev_row && t.col == prev_col) {
76 values.back() += t.val;
77 } else {
78 col_index.push_back(t.col);
79 values.push_back(t.val);
80 ++row_offset[t.row-row_begin+1];
81 prev_row = t.row;
82 prev_col = t.col;
83 }
84 }
85 for (Long r = 0; r < nrows; ++r) { row_offset[r+1] += row_offset[r]; }
86 }

109inline Vector<Triplet> exchange_triplets (Vector<Triplet>&& trips, Vector<Long> const& rank_offset,
110 MPI_Comm comm)
111{
112 const int nprocs = ParallelDescriptor::NProcs(comm);
113 if (nprocs == 1) { return std::move(trips); }
114#ifdef AMREX_USE_MPI
115 const int myproc = ParallelDescriptor::MyProc(comm);
116 Vector<Vector<Triplet>> sendbuf(nprocs);
117 Vector<Triplet> local;
118 for (auto const& t : trips) {
119 auto it = std::upper_bound(rank_offset.begin(), rank_offset.end(), t.row);
120 const int p = static_cast<int>(it - rank_offset.begin()) - 1;
121 AMREX_ALWAYS_ASSERT(p >= 0 && p < nprocs); // row in [0, total)
122 if (p == myproc) { local.push_back(t); } else { sendbuf[p].push_back(t); }
123 }
124 trips.clear();
125 // Counts in triplets (one MPI element per triplet), checked against int
126 constexpr Long int_max = std::numeric_limits<int>::max();
127 Vector<int> sendcnt(nprocs), recvcnt(nprocs), sdispl(nprocs), rdispl(nprocs);
128 for (int p = 0; p < nprocs; ++p) {
129 AMREX_ALWAYS_ASSERT(sendbuf[p].size() <= int_max);
130 sendcnt[p] = static_cast<int>(sendbuf[p].size());
131 }
132 MPI_Alltoall(sendcnt.data(), 1, MPI_INT, recvcnt.data(), 1, MPI_INT, comm);
133 Long stot = 0, rtot = 0;
134 for (int p = 0; p < nprocs; ++p) {
135 sdispl[p] = static_cast<int>(stot); stot += sendcnt[p];
136 rdispl[p] = static_cast<int>(rtot); rtot += recvcnt[p];
137 }
138 AMREX_ALWAYS_ASSERT(stot <= int_max && rtot <= int_max);
139 Vector<Triplet> sbuf;
140 sbuf.reserve(stot);
141 for (auto const& b : sendbuf) { sbuf.insert(sbuf.end(), b.begin(), b.end()); }
142 sendbuf.clear();
143 const Long nlocal = local.size();
144 local.resize(nlocal + rtot);
145 MPI_Datatype ttype;
146 MPI_Type_contiguous(int(sizeof(Triplet)), MPI_BYTE, &ttype);
147 MPI_Type_commit(&ttype);
148 MPI_Alltoallv(sbuf.data(), sendcnt.data(), sdispl.data(), ttype,
149 local.data()+nlocal, recvcnt.data(), rdispl.data(), ttype, comm);
150 MPI_Type_free(&ttype);
151 return local;
152#else
153 amrex::ignore_unused(rank_offset);
154 return std::move(trips);
155#endif
156}

170inline void triplets_to_spmatrix (Gpu::DeviceVector<Long>& rows, Gpu::DeviceVector<Long>& cols,
171 Gpu::DeviceVector<Real>& vals, Long row_begin, Long nrows,
172 Vector<Long> const& rank_offset, MPI_Comm comm,
173 SpMatrix<Real>& mat)
174{
175 BL_PROFILE("composite::triplets_to_spmatrix");
176 AMREX_ALWAYS_ASSERT(rows.size() == cols.size() && rows.size() == vals.size());
177 const Long row_end = row_begin + nrows;
178
179 if (ParallelDescriptor::NProcs(comm) > 1) {
180 // Gather the triplets of other ranks' rows, exchange them on the
181 // host and append the ones received
182 const auto n = Long(rows.size());
183 auto const* pr = rows.data();
184 auto const* pc = cols.data();
185 auto const* pv = vals.data();
186 Gpu::DeviceVector<Long> pos(n);
187 auto* ppos = pos.data();
188 const Long nsend = Scan::PrefixSum<Long>(n,
189 [=] AMREX_GPU_DEVICE (Long i) -> Long { return (pr[i] < row_begin || pr[i] >= row_end) ? 1 : 0; },
190 [=] AMREX_GPU_DEVICE (Long i, Long const& x) { ppos[i] = x; },
191 Scan::Type::exclusive, Scan::retSum);
192 Gpu::DeviceVector<Triplet> dsend(nsend);
193 auto* ps = dsend.data();
194 amrex::ParallelFor(n, [=] AMREX_GPU_DEVICE (Long i) noexcept
195 {
196 if (pr[i] < row_begin || pr[i] >= row_end) {
197 ps[ppos[i]] = Triplet{.row = pr[i], .col = pc[i], .val = pv[i]};
198 }
199 });
200 Vector<Triplet> hsend(nsend);
201 Gpu::copyAsync(Gpu::deviceToHost, dsend.begin(), dsend.end(), hsend.begin());
202 Gpu::streamSynchronize();
203 Vector<Triplet> const hrecv = exchange_triplets(std::move(hsend), rank_offset, comm);
204 const auto nrecv = Long(hrecv.size());
205 if (nrecv > 0) {
206 Gpu::DeviceVector<Triplet> drecv(nrecv);
207 Gpu::copyAsync(Gpu::hostToDevice, hrecv.begin(), hrecv.end(), drecv.begin());
208 rows.resize(n+nrecv);
209 cols.resize(n+nrecv);
210 vals.resize(n+nrecv);
211 auto const* pt = drecv.data();
212 auto* r = rows.data();
213 auto* c = cols.data();
214 auto* v = vals.data();
215 amrex::ParallelFor(nrecv, [=] AMREX_GPU_DEVICE (Long i) noexcept
216 {
217 r[n+i] = pt[i].row;
218 c[n+i] = pt[i].col;
219 v[n+i] = pt[i].val;
220 });
221 Gpu::streamSynchronize();
222 }
223 }
224
225 // Bucket the triplet indices by row
226 const auto ntrip = Long(rows.size());
227 auto const* pr = rows.data();
228 auto const* pc = cols.data();
229 auto const* pv = vals.data();
230 Gpu::DeviceVector<int> cnt(nrows, 0);
231 Gpu::DeviceVector<Long> start(nrows+1);
232 auto* pcnt = cnt.data();
233 auto* pstart = start.data();
234 amrex::ParallelFor(ntrip, [=] AMREX_GPU_DEVICE (Long i) noexcept
235 {
236 if (pr[i] >= row_begin && pr[i] < row_end) {
237 Gpu::Atomic::AddNoRet(pcnt + (pr[i]-row_begin), 1);
238 }
239 });
240 const Long n_owned = Scan::PrefixSum<Long>(nrows,
241 [=] AMREX_GPU_DEVICE (Long r) -> Long { return pcnt[r]; },
242 [=] AMREX_GPU_DEVICE (Long r, Long const& x) { pstart[r] = x; },
243 Scan::Type::exclusive, Scan::retSum);
244 amrex::ParallelFor(nrows+1, [=] AMREX_GPU_DEVICE (Long r) noexcept
245 {
246 if (r < nrows) { pcnt[r] = 0; } else { pstart[r] = n_owned; }
247 });
248 Gpu::DeviceVector<Long> perm(n_owned);
249 auto* pp = perm.data();
250 amrex::ParallelFor(ntrip, [=] AMREX_GPU_DEVICE (Long i) noexcept
251 {
252 if (pr[i] >= row_begin && pr[i] < row_end) {
253 const Long r = pr[i]-row_begin;
254 pp[pstart[r] + Gpu::Atomic::Add(pcnt+r, 1)] = i;
255 }
256 });
257
258 // Sort each row by (column, triplet index) and count its distinct columns
259 Gpu::DeviceVector<Long> nuniq(nrows);
260 auto* pnu = nuniq.data();
261 amrex::ParallelFor(nrows, [=] AMREX_GPU_DEVICE (Long r) noexcept
262 {
263 const Long b = pstart[r], e = pstart[r+1];
264 for (Long k = b+1; k < e; ++k) {
265 const Long t = pp[k];
266 const Long ct = pc[t];
267 Long m = k;
268 while (m > b && (pc[pp[m-1]] > ct || (pc[pp[m-1]] == ct && pp[m-1] > t))) {
269 pp[m] = pp[m-1];
270 --m;
271 }
272 pp[m] = t;
273 }
274 Long u = 0;
275 for (Long k = b; k < e; ++k) {
276 if (k == b || pc[pp[k]] != pc[pp[k-1]]) { ++u; }
277 }
278 pnu[r] = u;
279 });
280
281 // Sum the duplicates
282 SpMatrix<Real>::csr_type csr;
283 csr.row_offset.resize(nrows+1);
284 auto* pro = csr.row_offset.data();
285 const Long nnz = Scan::PrefixSum<Long>(nrows,
286 [=] AMREX_GPU_DEVICE (Long r) -> Long { return pnu[r]; },
287 [=] AMREX_GPU_DEVICE (Long r, Long const& x) { pro[r] = x; },
288 Scan::Type::exclusive, Scan::retSum);
289 csr.nnz = nnz;
290 csr.col_index.resize(nnz);
291 csr.mat.resize(nnz);
292 auto* pci = csr.col_index.data();
293 auto* pm = csr.mat.data();
294 amrex::ParallelFor(nrows+1, [=] AMREX_GPU_DEVICE (Long r) noexcept
295 {
296 if (r == nrows) { pro[r] = nnz; return; }
297 Long u = pro[r]-1;
298 for (Long k = pstart[r]; k < pstart[r+1]; ++k) {
299 const Long t = pp[k];
300 if (k == pstart[r] || pc[t] != pc[pp[k-1]]) {
301 ++u;
302 pci[u] = pc[t];
303 pm[u] = pv[t];
304 } else {
305 pm[u] += pv[t];
306 }
307 }
308 });
309 Gpu::streamSynchronize();
310 rows.clear();
311 cols.clear();
312 vals.clear();
313 mat.define(AlgPartition(rank_offset), std::move(csr), CsrSorted{true});
314}

340template <typename T>
341Vector<T> gather_on_root (Vector<T> const& v, MPI_Comm comm)
342{
343 const int nprocs = ParallelDescriptor::NProcs(comm);
344 if (nprocs == 1) { return v; }
345#ifdef AMREX_USE_MPI
346 // MPI counts are int (callers check the total with dump_single_file)
347 AMREX_ALWAYS_ASSERT(Long(v.size()) <= Long(std::numeric_limits<int>::max()));
348 int nloc = static_cast<int>(v.size());
349 Vector<int> counts(nprocs), displs(nprocs, 0);
350 MPI_Gather(&nloc, 1, MPI_INT, counts.data(), 1, MPI_INT, 0, comm);
351 Vector<T> g;
352 if (ParallelDescriptor::MyProc(comm) == 0) {
353 Long ntot = 0;
354 for (int p = 0; p < nprocs; ++p) { displs[p] = int(ntot); ntot += counts[p]; }
355 AMREX_ALWAYS_ASSERT(ntot <= Long(std::numeric_limits<int>::max()));
356 g.resize(ntot);
357 }
358 auto const mt = ParallelDescriptor::Mpi_typemap<T>::type();
359 MPI_Gatherv(v.data(), nloc, mt, g.data(), counts.data(), displs.data(), mt, 0, comm);
360 return g;
361#else
362 return v;
363#endif
364}
