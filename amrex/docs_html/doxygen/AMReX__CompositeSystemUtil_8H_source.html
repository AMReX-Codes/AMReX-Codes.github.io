50 void define (Long a_row_begin, Long a_nrows, Vector<Triplet> trips)
51 {
52 row_begin = a_row_begin;
53 nrows = a_nrows;
54 std::sort(trips.begin(), trips.end(), [] (Triplet const& a, Triplet const& b) {
55 return (a.row < b.row) || (a.row == b.row && a.col < b.col);
56 });
57 row_offset.assign(nrows+1, 0);
58 col_index.clear();
59 values.clear();
60 col_index.reserve(trips.size());
61 values.reserve(trips.size());
62 Long prev_row = -1, prev_col = -1;
63 for (auto const& t : trips) {
64 AMREX_ALWAYS_ASSERT(t.row >= row_begin && t.row < row_begin+nrows);
65 if (t.row == prev_row && t.col == prev_col) {
66 values.back() += t.val;
67 } else {
68 col_index.push_back(t.col);
69 values.push_back(t.val);
70 ++row_offset[t.row-row_begin+1];
71 prev_row = t.row;
72 prev_col = t.col;
73 }
74 }
75 for (Long r = 0; r < nrows; ++r) { row_offset[r+1] += row_offset[r]; }
76 }

79 void toSpMatrix (SpMatrix<Real>& mat, Vector<Long> const& rank_offset) const
80 {
81 SpMatrix<Real>::csr_type csr;
82 csr.mat.resize(values.size());
83 csr.col_index.resize(col_index.size());
84 csr.row_offset.resize(row_offset.size());
85 csr.nnz = nnz();
86 Gpu::copyAsync(Gpu::hostToDevice, values.begin(), values.end(), csr.mat.begin());

89 Gpu::streamSynchronize();

118template <typename T>
119Vector<T> gather_on_root (Vector<T> const& v, MPI_Comm comm)
120{
121 const int nprocs = ParallelDescriptor::NProcs(comm);
122 if (nprocs == 1) { return v; }
123#ifdef AMREX_USE_MPI
124 // MPI counts are int (callers check the total with dump_single_file)
125 AMREX_ALWAYS_ASSERT(Long(v.size()) <= Long(std::numeric_limits<int>::max()));
126 int nloc = static_cast<int>(v.size());
127 Vector<int> counts(nprocs), displs(nprocs, 0);
128 MPI_Gather(&nloc, 1, MPI_INT, counts.data(), 1, MPI_INT, 0, comm);
129 Vector<T> g;
130 if (ParallelDescriptor::MyProc(comm) == 0) {
131 Long ntot = 0;
132 for (int p = 0; p < nprocs; ++p) { displs[p] = int(ntot); ntot += counts[p]; }
133 AMREX_ALWAYS_ASSERT(ntot <= Long(std::numeric_limits<int>::max()));
134 g.resize(ntot);
135 }
136 auto const mt = ParallelDescriptor::Mpi_typemap<T>::type();
137 MPI_Gatherv(v.data(), nloc, mt, g.data(), counts.data(), displs.data(), mt, 0, comm);
138 return g;
139#else
140 return v;
141#endif
142}

177inline void write_row_map (std::string const& file, std::string const& title,
178 Vector<std::array<Long,4>> const& info, Vector<Real> const& scaling,
179 Long row_begin, MPI_Comm comm, Long gather_max)
180{
181 const int nprocs = ParallelDescriptor::NProcs(comm);
182 const int myproc = ParallelDescriptor::MyProc(comm);
183 Long nglobal = info.size();
184 ParallelAllReduce::Sum(nglobal, comm);
185
186 Vector<Long> flat(info.size()*4);
187 for (Long r = 0; r < info.size(); ++r) {
188 for (int c = 0; c < 4; ++c) { flat[r*4+c] = info[r][c]; }
189 }
190 auto write = [&] (std::string const& name, std::string const& note, Long first_row,
191 Vector<Long> const& fl, Vector<Real> const& sc)
192 {
193 auto ofs = open_dump(name);
194 ofs << "# " << title << "\n" << note << "# row level i j k scaling\n";
195 for (Long r = 0; r < sc.size(); ++r) {
196 ofs << first_row+r+1 << ' ' << fl[r*4] << ' ' << fl[r*4+1] << ' '
197 << fl[r*4+2] << ' ' << fl[r*4+3] << ' ' << sc[r] << '\n';
198 }
199 check_dump(ofs, name);
200 };
201 if (dump_single_file(4*nglobal, gather_max, comm)) {
202 auto gflat = gather_on_root(flat, comm);
203 auto gscl = gather_on_root(scaling, comm);
204 if (myproc == 0) { write(file, "", 0, gflat, gscl); }
205 } else {
206 write(file + "." + std::to_string(myproc),
207 "# part " + std::to_string(myproc) + " of " + std::to_string(nprocs) +
208 ": rows owned by this rank\n", row_begin, flat, scaling);
209 }

253inline void write_matrix_market (std::string const& file, OwnedCSR const& csr,
254 Long nglobal, MPI_Comm comm, Long gather_max)
255{
256 const int nprocs = ParallelDescriptor::NProcs(comm);
257 const int myproc = ParallelDescriptor::MyProc(comm);
258 Long nnz_global = csr.nnz();
259 ParallelAllReduce::Sum(nnz_global, comm);
260
261 Vector<Long> rows(csr.nnz());
262 for (Long r = 0; r < csr.nrows; ++r) {
263 for (Long n = csr.row_offset[r]; n < csr.row_offset[r+1]; ++n) { rows[n] = csr.row_begin + r; }
264 }
265 auto write = [&] (std::string const& name, std::string const& note, Vector<Long> const& rr,
266 Vector<Long> const& cc, Vector<Real> const& vv)
267 {
268 auto ofs = open_dump(name);
269 ofs << "%%MatrixMarket matrix coordinate real general\n" << note;
270 ofs << nglobal << ' ' << nglobal << ' ' << vv.size() << '\n';
271 for (Long n = 0; n < vv.size(); ++n) {
272 ofs << rr[n]+1 << ' ' << cc[n]+1 << ' ' << vv[n] << '\n';
273 }
274 check_dump(ofs, name);
275 };
276 if (dump_single_file(nnz_global, gather_max, comm)) {
277 auto grows = gather_on_root(rows, comm);
278 auto gcols = gather_on_root(csr.col_index, comm);
279 auto gvals = gather_on_root(csr.values, comm);
280 if (myproc == 0) { write(file, "", grows, gcols, gvals); }
281 } else {
282 write(file + "." + std::to_string(myproc),
283 "% part " + std::to_string(myproc) + " of " + std::to_string(nprocs) +
284 ": rows owned by this rank\n", rows, csr.col_index, csr.values);
285 }
