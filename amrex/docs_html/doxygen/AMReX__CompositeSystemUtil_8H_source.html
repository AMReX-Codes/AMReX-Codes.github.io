61inline void define_gather (EntryGather& g, Vector<Long> const& entry, Long nentries)
62{
63 const auto ntrip = Long(entry.size());
64 Vector<Long> offset(nentries+1, 0);
65 for (Long n = 0; n < ntrip; ++n) { ++offset[entry[n]+1]; }
66 for (Long e = 0; e < nentries; ++e) { offset[e+1] += offset[e]; }
67 Vector<Long> fill(offset.begin(), offset.end()-1);
68 Vector<Long> trip(ntrip);
69 for (Long n = 0; n < ntrip; ++n) { trip[fill[entry[n]]++] = n; }
70 g.offset.resize(nentries+1);
71 g.trip.resize(ntrip);
72 Gpu::copyAsync(Gpu::hostToDevice, offset.begin(), offset.end(), g.offset.begin());
73 Gpu::copyAsync(Gpu::hostToDevice, trip.begin(), trip.end(), g.trip.begin());
74 Gpu::streamSynchronize();
75}

186 template <class P>
187 bool rescale (Gpu::DeviceVector<Real>& vals, P const& scaled, bool new_setup,
188 MPI_Comm comm, int verbose, std::string const& prefix)
189 {
190 BL_PROFILE("composite::SetupReuse::rescale");
191 const auto n = Long(vals.size());
192 auto* v = vals.data();
193 Real norm2 = (n > 0) ? Reduce::Sum<Real>(n, [=] AMREX_GPU_DEVICE (Long i) noexcept
194 { return scaled(i) ? v[i]*v[i] : Real(0.0); })
195 : Real(0.0);
196 ParallelAllReduce::Sum(norm2, comm);
197 // No setup matrix to match: this is it
198 new_setup = new_setup || !(setup_norm2 > Real(0.0)) || !(norm2 > Real(0.0));
199 bool changed = false;
200 // have_vals and capture are the same on all ranks, so that the
201 // reduction below is collective
202 if (!new_setup && max_change > Real(0.0) && have_vals && !capture) {
203 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(std::cmp_equal(setup_vals.size(), n),
204 "composite::SetupReuse: the matrix pattern changed since the kept setup");
205 // The change of the scaled matrix since the setup, relative to
206 // the setup matrix
207 const Real s = std::sqrt(setup_norm2 / norm2);
208 auto const* v0 = setup_vals.data();
209 Real d2 = Reduce::Sum<Real>(n, [=] AMREX_GPU_DEVICE (Long i) noexcept
210 { Real d = scaled(i) ? s*v[i] - v0[i] : Real(0.0); return d*d; });
211 ParallelAllReduce::Sum(d2, comm);
212 const Real change = std::sqrt(d2 / setup_norm2);
213 if (change > max_change) {
214 if (verbose) {
215 amrex::Print() << prefix << ": the matrix changed by " << change
216 << " since the kept setup; redoing the setup\n";
217 }
218 changed = true;
219 }
220 }
221 if (new_setup || changed) {
222 setup_norm2 = norm2;
223 scale = Real(1.0);
224 } else {
225 scale = std::sqrt(setup_norm2 / norm2);
226 const Real s = scale;
227 amrex::ParallelFor(n, [=] AMREX_GPU_DEVICE (Long i) noexcept
228 {
229 if (scaled(i)) { v[i] *= s; }
230 });
231 Gpu::streamSynchronize();
232 }
233 current = (new_setup || changed);
234 if (max_change > Real(0.0) && (new_setup || changed || capture)) {
235 // The reference: the (scaled) values the setup is, or was, done for
236 setup_vals.resize(n);
237 Gpu::copyAsync(Gpu::deviceToDevice, vals.begin(), vals.end(), setup_vals.begin());
238 Gpu::streamSynchronize();
239 have_vals = true;
240 }
241 capture = false;
242 return changed;
243 }

301struct OwnedCSR
302{
303 Long row_begin = 0;
304 Long nrows = 0;
305 Vector<Long> row_offset;
306 Vector<Long> col_index;
307 Vector<Real> values;
308
309 [[nodiscard]] Long nnz () const { return values.size(); }
310
313 void define (Long a_row_begin, Long a_nrows, Vector<Triplet> trips)
314 {
315 row_begin = a_row_begin;
316 nrows = a_nrows;
317 std::sort(trips.begin(), trips.end(), [] (Triplet const& a, Triplet const& b) {
318 return (a.row < b.row) || (a.row == b.row && a.col < b.col);
319 });
320 row_offset.assign(nrows+1, 0);
321 col_index.clear();
322 values.clear();
323 col_index.reserve(trips.size());
324 values.reserve(trips.size());
325 Long prev_row = -1, prev_col = -1;
326 for (auto const& t : trips) {
327 AMREX_ALWAYS_ASSERT(t.row >= row_begin && t.row < row_begin+nrows);
328 if (t.row == prev_row && t.col == prev_col) {
329 values.back() += t.val;
330 } else {
331 col_index.push_back(t.col);
332 values.push_back(t.val);
333 ++row_offset[t.row-row_begin+1];
334 prev_row = t.row;
335 prev_col = t.col;
336 }
337 }
338 for (Long r = 0; r < nrows; ++r) { row_offset[r+1] += row_offset[r]; }
339 }

362inline Vector<Triplet> exchange_triplets (Vector<Triplet>&& trips, Vector<Long> const& rank_offset,
363 MPI_Comm comm)
364{
365 const int nprocs = ParallelDescriptor::NProcs(comm);
366 if (nprocs == 1) { return std::move(trips); }
367#ifdef AMREX_USE_MPI
368 const int myproc = ParallelDescriptor::MyProc(comm);
369 Vector<Vector<Triplet>> sendbuf(nprocs);
370 Vector<Triplet> local;
371 for (auto const& t : trips) {
372 auto it = std::upper_bound(rank_offset.begin(), rank_offset.end(), t.row);
373 const int p = static_cast<int>(it - rank_offset.begin()) - 1;
374 AMREX_ALWAYS_ASSERT(p >= 0 && p < nprocs); // row in [0, total)
375 if (p == myproc) { local.push_back(t); } else { sendbuf[p].push_back(t); }
376 }
377 trips.clear();
378 // Counts in triplets (one MPI element per triplet), checked against int
379 constexpr Long int_max = std::numeric_limits<int>::max();
380 Vector<int> sendcnt(nprocs), recvcnt(nprocs), sdispl(nprocs), rdispl(nprocs);
381 for (int p = 0; p < nprocs; ++p) {
382 AMREX_ALWAYS_ASSERT(sendbuf[p].size() <= int_max);
383 sendcnt[p] = static_cast<int>(sendbuf[p].size());
384 }
385 MPI_Alltoall(sendcnt.data(), 1, MPI_INT, recvcnt.data(), 1, MPI_INT, comm);
386 Long stot = 0, rtot = 0;
387 for (int p = 0; p < nprocs; ++p) {
388 sdispl[p] = static_cast<int>(stot); stot += sendcnt[p];
389 rdispl[p] = static_cast<int>(rtot); rtot += recvcnt[p];
390 }
391 AMREX_ALWAYS_ASSERT(stot <= int_max && rtot <= int_max);
392 Vector<Triplet> sbuf;
393 sbuf.reserve(stot);
394 for (auto const& b : sendbuf) { sbuf.insert(sbuf.end(), b.begin(), b.end()); }
395 sendbuf.clear();
396 const Long nlocal = local.size();
397 local.resize(nlocal + rtot);
398 MPI_Datatype ttype;
399 MPI_Type_contiguous(int(sizeof(Triplet)), MPI_BYTE, &ttype);
400 MPI_Type_commit(&ttype);
401 MPI_Alltoallv(sbuf.data(), sendcnt.data(), sdispl.data(), ttype,
402 local.data()+nlocal, recvcnt.data(), rdispl.data(), ttype, comm);
403 MPI_Type_free(&ttype);
404 return local;
405#else
406 amrex::ignore_unused(rank_offset);
407 return std::move(trips);
408#endif
409}

418template <typename I>
419void localize_triplets (Gpu::DeviceVector<I>& rows, Gpu::DeviceVector<I>& cols,
420 Gpu::DeviceVector<Real>& vals, Long row_begin, Long nrows,
421 Vector<Long> const& rank_offset, MPI_Comm comm)
422{
423 if (ParallelDescriptor::NProcs(comm) == 1) { return; }
424 BL_PROFILE("composite::localize_triplets");
425 const Long row_end = row_begin + nrows;
426 const auto n = Long(rows.size());
427 auto* pr = rows.data();
428 auto* pc = cols.data();
429 auto* pv = vals.data();
430 Gpu::DeviceVector<Long> pos(n);
431 auto* ppos = pos.data();
432 const Long nsend = Scan::PrefixSum<Long>(n,
433 [=] AMREX_GPU_DEVICE (Long i) -> Long { return (pr[i] < row_begin || pr[i] >= row_end) ? 1 : 0; },
434 [=] AMREX_GPU_DEVICE (Long i, Long const& x) { ppos[i] = x; },
435 Scan::Type::exclusive, Scan::retSum);
436 Gpu::DeviceVector<Triplet> dsend(nsend);
437 Gpu::DeviceVector<I> lrows(n-nsend), lcols(n-nsend);
438 Gpu::DeviceVector<Real> lvals(n-nsend);
439 auto* ps = dsend.data();
440 auto* lr = lrows.data();
441 auto* lc = lcols.data();
442 auto* lv = lvals.data();
443 amrex::ParallelFor(n, [=] AMREX_GPU_DEVICE (Long i) noexcept
444 {
445 if (pr[i] < row_begin || pr[i] >= row_end) {
446 ps[ppos[i]] = Triplet{.row = Long(pr[i]), .col = Long(pc[i]), .val = pv[i]};
447 } else {
448 const Long il = i - ppos[i];
449 lr[il] = pr[i];
450 lc[il] = pc[i];
451 lv[il] = pv[i];
452 }
453 });
454 Vector<Triplet> hsend(nsend);
455 Gpu::copyAsync(Gpu::deviceToHost, dsend.begin(), dsend.end(), hsend.begin());
456 Gpu::streamSynchronize();
457 Vector<Triplet> const hrecv = exchange_triplets(std::move(hsend), rank_offset, comm);
458 const auto nrecv = Long(hrecv.size());
459 const Long nloc = n - nsend;
460 rows = std::move(lrows);
461 cols = std::move(lcols);
462 vals = std::move(lvals);
463 if (nrecv > 0) {
464 Gpu::DeviceVector<Triplet> drecv(nrecv);
465 Gpu::copyAsync(Gpu::hostToDevice, hrecv.begin(), hrecv.end(), drecv.begin());
466 rows.resize(nloc+nrecv);
467 cols.resize(nloc+nrecv);
468 vals.resize(nloc+nrecv);
469 auto const* pt = drecv.data();
470 auto* r = rows.data();
471 auto* c = cols.data();
472 auto* v = vals.data();
473 amrex::ParallelFor(nrecv, [=] AMREX_GPU_DEVICE (Long i) noexcept
474 {
475 r[nloc+i] = I(pt[i].row);
476 c[nloc+i] = I(pt[i].col);
477 v[nloc+i] = pt[i].val;
478 });
479 Gpu::streamSynchronize(); // before drecv is freed
480 }
481}

495inline void triplets_to_spmatrix (Gpu::DeviceVector<Long>& rows, Gpu::DeviceVector<Long>& cols,
496 Gpu::DeviceVector<Real>& vals, Long row_begin, Long nrows,
497 Vector<Long> const& rank_offset, MPI_Comm comm,
498 SpMatrix<Real>& mat, bool localized = false)
499{
500 BL_PROFILE("composite::triplets_to_spmatrix");
501 AMREX_ALWAYS_ASSERT(rows.size() == cols.size() && rows.size() == vals.size());
502 if (!localized) {
503 localize_triplets(rows, cols, vals, row_begin, nrows, rank_offset, comm);
504 }
505
506 // Bucket the triplet indices by row
507 const auto ntrip = Long(rows.size());
508 auto const* pr = rows.data();
509 auto const* pc = cols.data();
510 auto const* pv = vals.data();
511 Gpu::DeviceVector<int> cnt(nrows, 0);
512 Gpu::DeviceVector<Long> start(nrows+1);
513 auto* pcnt = cnt.data();
514 auto* pstart = start.data();
515 amrex::ParallelFor(ntrip, [=] AMREX_GPU_DEVICE (Long i) noexcept
516 {
517 Gpu::Atomic::AddNoRet(pcnt + (pr[i]-row_begin), 1);
518 });
519 const Long n_owned = Scan::PrefixSum<Long>(nrows,
520 [=] AMREX_GPU_DEVICE (Long r) -> Long { return pcnt[r]; },
521 [=] AMREX_GPU_DEVICE (Long r, Long const& x) { pstart[r] = x; },
522 Scan::Type::exclusive, Scan::retSum);
523 amrex::ParallelFor(nrows+1, [=] AMREX_GPU_DEVICE (Long r) noexcept
524 {
525 if (r < nrows) { pcnt[r] = 0; } else { pstart[r] = n_owned; }
526 });
527 Gpu::DeviceVector<Long> perm(n_owned);
528 auto* pp = perm.data();
529 amrex::ParallelFor(ntrip, [=] AMREX_GPU_DEVICE (Long i) noexcept
530 {
531 const Long r = pr[i]-row_begin;
532 pp[pstart[r] + Gpu::Atomic::Add(pcnt+r, 1)] = i;
533 });
534
535 // Sort each row by (column, triplet index) and count its distinct columns
536 Gpu::DeviceVector<Long> nuniq(nrows);
537 auto* pnu = nuniq.data();
538 amrex::ParallelFor(nrows, [=] AMREX_GPU_DEVICE (Long r) noexcept
539 {
540 const Long b = pstart[r], e = pstart[r+1];
541 for (Long k = b+1; k < e; ++k) {
542 const Long t = pp[k];
543 const Long ct = pc[t];
544 Long m = k;
545 while (m > b && (pc[pp[m-1]] > ct || (pc[pp[m-1]] == ct && pp[m-1] > t))) {
546 pp[m] = pp[m-1];
547 --m;
548 }
549 pp[m] = t;
550 }
551 Long u = 0;
552 for (Long k = b; k < e; ++k) {
553 if (k == b || pc[pp[k]] != pc[pp[k-1]]) { ++u; }
554 }
555 pnu[r] = u;
556 });
557
558 // Sum the duplicates
559 SpMatrix<Real>::csr_type csr;
560 csr.row_offset.resize(nrows+1);
561 auto* pro = csr.row_offset.data();
562 const Long nnz = Scan::PrefixSum<Long>(nrows,
563 [=] AMREX_GPU_DEVICE (Long r) -> Long { return pnu[r]; },
564 [=] AMREX_GPU_DEVICE (Long r, Long const& x) { pro[r] = x; },
565 Scan::Type::exclusive, Scan::retSum);
566 csr.nnz = nnz;
567 csr.col_index.resize(nnz);
568 csr.mat.resize(nnz);
569 auto* pci = csr.col_index.data();
570 auto* pm = csr.mat.data();
571 amrex::ParallelFor(nrows+1, [=] AMREX_GPU_DEVICE (Long r) noexcept
572 {
573 if (r == nrows) { pro[r] = nnz; return; }
574 Long u = pro[r]-1;
575 for (Long k = pstart[r]; k < pstart[r+1]; ++k) {
576 const Long t = pp[k];
577 if (k == pstart[r] || pc[t] != pc[pp[k-1]]) {
578 ++u;
579 pci[u] = pc[t];
580 pm[u] = pv[t];
581 } else {
582 pm[u] += pv[t];
583 }
584 }
585 });
586 Gpu::streamSynchronize();
587 rows.clear();
588 cols.clear();
589 vals.clear();
590 mat.define(AlgPartition(rank_offset), std::move(csr), CsrSorted{true});
591}

617template <typename T>
618Vector<T> gather_on_root (Vector<T> const& v, MPI_Comm comm)
619{
620 const int nprocs = ParallelDescriptor::NProcs(comm);
621 if (nprocs == 1) { return v; }
622#ifdef AMREX_USE_MPI
623 // MPI counts are int (callers check the total with dump_single_file)
624 AMREX_ALWAYS_ASSERT(Long(v.size()) <= Long(std::numeric_limits<int>::max()));
625 int nloc = static_cast<int>(v.size());
626 Vector<int> counts(nprocs), displs(nprocs, 0);
627 MPI_Gather(&nloc, 1, MPI_INT, counts.data(), 1, MPI_INT, 0, comm);
628 Vector<T> g;
629 if (ParallelDescriptor::MyProc(comm) == 0) {
630 Long ntot = 0;
631 for (int p = 0; p < nprocs; ++p) { displs[p] = int(ntot); ntot += counts[p]; }
632 AMREX_ALWAYS_ASSERT(ntot <= Long(std::numeric_limits<int>::max()));
633 g.resize(ntot);
634 }
635 auto const mt = ParallelDescriptor::Mpi_typemap<T>::type();
636 MPI_Gatherv(v.data(), nloc, mt, g.data(), counts.data(), displs.data(), mt, 0, comm);
637 return g;
638#else
639 return v;
640#endif
641}

676inline void write_row_map (std::string const& file, std::string const& title,
677 Vector<std::array<Long,4>> const& info, Vector<Real> const& scaling,
678 Long row_begin, MPI_Comm comm, Long gather_max)
679{
680 const int nprocs = ParallelDescriptor::NProcs(comm);
681 const int myproc = ParallelDescriptor::MyProc(comm);
682 Long nglobal = info.size();
683 ParallelAllReduce::Sum(nglobal, comm);
684

687 for (int c = 0; c < 4; ++c) { flat[r*4+c] = info[r][c]; }
688 }

690 Vector<Long> const& fl, Vector<Real> const& sc)

694 for (Long r = 0; r < sc.size(); ++r) {
695 ofs << first_row+r+1 << ' ' << fl[r*4] << ' ' << fl[r*4+1] << ' '
696 << fl[r*4+2] << ' ' << fl[r*4+3] << ' ' << sc[r] << '\n';
697 }
698 check_dump(ofs, name);
699 };
700 if (dump_single_file(4*nglobal, gather_max, comm)) {
701 auto gflat = gather_on_root(flat, comm);
702 auto gscl = gather_on_root(scaling, comm);
703 if (myproc == 0) { write(file, "", 0, gflat, gscl); }
704 } else {
705 write(file + "." + std::to_string(myproc),
706 "# part " + std::to_string(myproc) + " of " + std::to_string(nprocs) +
707 ": rows owned by this rank\n", row_begin, flat, scaling);
708 }

719inline void algmg_parse_options (AlgMG<Real>& mg, std::string const& prefix,

752inline void write_matrix_market (std::string const& file, OwnedCSR const& csr,
753 Long nglobal, MPI_Comm comm, Long gather_max)
754{
755 const int nprocs = ParallelDescriptor::NProcs(comm);
756 const int myproc = ParallelDescriptor::MyProc(comm);
757 Long nnz_global = csr.nnz();
758 ParallelAllReduce::Sum(nnz_global, comm);
759
760 Vector<Long> rows(csr.nnz());
761 for (Long r = 0; r < csr.nrows; ++r) {
762 for (Long n = csr.row_offset[r]; n < csr.row_offset[r+1]; ++n) { rows[n] = csr.row_begin + r; }
763 }
764 auto write = [&] (std::string const& name, std::string const& note, Vector<Long> const& rr,
765 Vector<Long> const& cc, Vector<Real> const& vv)
766 {
767 auto ofs = open_dump(name);
768 ofs << "%%MatrixMarket matrix coordinate real general\n" << note;
769 ofs << nglobal << ' ' << nglobal << ' ' << vv.size() << '\n';
770 for (Long n = 0; n < vv.size(); ++n) {
771 ofs << rr[n]+1 << ' ' << cc[n]+1 << ' ' << vv[n] << '\n';
772 }
773 check_dump(ofs, name);
774 };
775 if (dump_single_file(nnz_global, gather_max, comm)) {
776 auto grows = gather_on_root(rows, comm);
777 auto gcols = gather_on_root(csr.col_index, comm);
778 auto gvals = gather_on_root(csr.values, comm);
779 if (myproc == 0) { write(file, "", grows, gcols, gvals); }
780 } else {
781 write(file + "." + std::to_string(myproc),
782 "% part " + std::to_string(myproc) + " of " + std::to_string(nprocs) +
783 ": rows owned by this rank\n", rows, csr.col_index, csr.values);
784 }
785}
