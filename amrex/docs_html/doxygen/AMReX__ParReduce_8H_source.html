296// Reduce f(box_no,i,j,k,n) separately for each component n (PerBox false) or
297// for each local box and component (PerBox true, result index box_no*ncomp+n).
298// A class template, so that nvcc can name the function of the device lambdas.
299template <bool PerBox, typename... Ops>
300struct ParReduceComp
301{
302 template <typename RT, BaseFabType FAB, typename F>
303 static Vector<RT>
304 eval (FabArray<FAB> const& fa, IntVect const& nghost, int ncomp, F const& f)
305 {
306 const int nlocal = fa.local_size();
307 const int nkeys = PerBox ? nlocal*ncomp : ncomp;
308 Vector<RT> r(std::max(nkeys,0));
309 for (auto& x : r) { for_each_init<0, RT, Ops...>(x); }
310 if (nlocal == 0 || ncomp <= 0) { return r; }
311
312#ifdef AMREX_USE_GPU
313 // Block b works on component n = b/nb1 of local box L and stores its
314 // partial result in slot b, so each key's slots are contiguous. Box L
315 // owns blocks [hoff[L],hoff[L+1]) of every component.
316 constexpr int MT = AMREX_GPU_MAX_THREADS;
317 const int maxblocks = int(Gpu::Device::maxBlocksPerLaunch());
318 Long ntot = 0;
319 for (int L = 0; L < nlocal; ++L) {
320 Box b = fa.box(fa.IndexArray()[L]);
321 ntot += b.ok() ? amrex::grow(b,nghost).numPts() : Long(0);
322 }
323 // About one launch worth of blocks, with at least 4 cells per thread.
324 const Long cells_per_block = std::max(Long(MT)*4, (ntot*ncomp + maxblocks - 1) / maxblocks);
325 Gpu::PinnedVector<int> hoff(nlocal+1, 0);
326 for (int L = 0; L < nlocal; ++L) {
327 Box b = fa.box(fa.IndexArray()[L]);
328 Long const npts = b.ok() ? amrex::grow(b,nghost).numPts() : Long(0);
329 Long const nb = (npts + cells_per_block - 1) / cells_per_block;
330 AMREX_ALWAYS_ASSERT(Long(hoff[L]) + nb < Long(std::numeric_limits<int>::max()));
331 hoff[L+1] = hoff[L] + int(nb);
332 }
333 const int nb1 = hoff[nlocal];
334 AMREX_ALWAYS_ASSERT(Long(nb1)*ncomp < Long(std::numeric_limits<int>::max()));
335 const int nblocks = nb1*ncomp;
336 if (nblocks == 0) { return r; }
337
338 auto const& stream = Gpu::gpuStream();
339 Gpu::DeviceVector<int> doff(nlocal+1);
340 Gpu::copyAsync(Gpu::hostToDevice, hoff.begin(), hoff.end(), doff.begin());
341 Gpu::DeviceVector<RT> dpart(nblocks);
342 Gpu::DeviceVector<RT> dr(nkeys);
343 int const* poff = doff.data();
344 RT* ppart = dpart.data();
345 RT* pr = dr.data();
346 BoxIndexer const* pbx = fa.getParForInfo(nghost).getBoxes();
347
348#ifdef AMREX_USE_SYCL
349 constexpr std::size_t shared_mem_bytes = sizeof(RT)
350 * std::max(Gpu::Device::warp_size, MT/Gpu::Device::warp_size);
351 amrex::launch<MT>(std::min(nblocks,maxblocks), shared_mem_bytes, stream,
352 [=] AMREX_GPU_DEVICE (Gpu::Handler const& gh) noexcept
353 {
354 Dim1 blockIdx {gh.blockIdx()};
355 Dim1 threadIdx{gh.threadIdx()};
356 Dim1 gridDim {gh.gridDim()};
357#else
358 amrex::launch<MT>(std::min(nblocks,maxblocks), 0, stream,
359 [=] AMREX_GPU_DEVICE () noexcept
360 {
361#endif
362 for (int b = int(blockIdx.x); b < nblocks; b += int(gridDim.x)) {
363 const int n = b / nb1;
364 const int bb = b - n*nb1;
365 const int L = amrex::bisect(poff, 0, nlocal, bb);
366 const int nb = poff[L+1] - poff[L];
367 BoxIndexer const& indexer = pbx[L];
368 RT t;
369 for_each_init<0, RT, Ops...>(t);
370 for (auto icell = std::uint64_t(bb-poff[L])*MT + threadIdx.x;
371 icell < indexer.numPts(); icell += std::uint64_t(nb)*MT)
372 {
373 auto [i, j, k] = indexer(icell);
374 for_each_local<0, RT, Ops...>(t, f(L,i,j,k,n));
375 }
376 RT& dst = ppart[b];
377 if (threadIdx.x == 0) { for_each_init<0, RT, Ops...>(dst); }
378#ifdef AMREX_USE_SYCL
379 for_each_parallel<0, RT, Ops...>(dst, t, gh);
380#else
381 for_each_parallel<0, RT, Ops...>(dst, t);
382#endif
383 }
384 });
385
386 // Combine each key's partial results in a fixed order.
387#ifdef AMREX_USE_SYCL
388 amrex::launch<MT>(std::min(nkeys,maxblocks), shared_mem_bytes, stream,
389 [=] AMREX_GPU_DEVICE (Gpu::Handler const& gh) noexcept
390 {
391 Dim1 blockIdx {gh.blockIdx()};
392 Dim1 threadIdx{gh.threadIdx()};
393 Dim1 gridDim {gh.gridDim()};
394#else
395 amrex::launch<MT>(std::min(nkeys,maxblocks), 0, stream,
396 [=] AMREX_GPU_DEVICE () noexcept
397 {
398#endif
399 for (int key = int(blockIdx.x); key < nkeys; key += int(gridDim.x)) {
400 int begin = 0;
401 int end = 0;
402 if (PerBox) { // not constexpr: nvcc cannot first-capture ncomp there
403 const int L = key / ncomp;
404 const int n = key - L*ncomp;
405 begin = n*nb1 + poff[L];
406 end = n*nb1 + poff[L+1];
407 } else {
408 begin = key*nb1;
409 end = begin + nb1;
410 }
411 RT t;
412 for_each_init<0, RT, Ops...>(t);
413 for (int ib = begin + int(threadIdx.x); ib < end; ib += MT) {
414 for_each_local<0, RT, Ops...>(t, ppart[ib]);
415 }
416 RT& dst = pr[key];
417 if (threadIdx.x == 0) { for_each_init<0, RT, Ops...>(dst); }
418#ifdef AMREX_USE_SYCL
419 for_each_parallel<0, RT, Ops...>(dst, t, gh);
420#else
421 for_each_parallel<0, RT, Ops...>(dst, t);
422#endif
423 }
424 });
425
426 Gpu::dtoh_memcpy_async(r.data(), pr, sizeof(RT)*nkeys);
427 Gpu::streamSynchronize();
428#else
429 if constexpr (PerBox) {
430 // With too few (box, component) pairs for the threads, split each
431 // box along its last dimension into contiguous chunks of about 16K
432 // cells. Items are ordered by box, component and chunk, which is
433 // the memory order, and their partial results are combined in order.
434 constexpr int d = AMREX_SPACEDIM-1;
435 const int nthreads = OpenMP::in_parallel() ? 1 : OpenMP::get_max_threads();
436 const bool split = nthreads > 1 && nkeys < 4*nthreads;
437 Vector<Box> bx(nlocal);
438 Vector<int> thick(nlocal), nchunk(nlocal), ioff(nlocal+1, 0);
439 for (int L = 0; L < nlocal; ++L) {
440 bx[L] = fa.box(fa.IndexArray()[L]);
441 if (bx[L].ok()) { bx[L].grow(nghost); }
442 // An empty box is one empty chunk.
443 const int len = bx[L].ok() ? bx[L].length(d) : 1;
444 thick[L] = (split && bx[L].ok()) ? int(std::max(Long(1), Long(16384)/(bx[L].numPts()/len))) : len;
445 nchunk[L] = (len + thick[L] - 1) / thick[L];
446 ioff[L+1] = ioff[L] + nchunk[L]*ncomp;
447 }
448 // Without splitting, item is the result index.
449 const int nitems = ioff[nlocal];
450 Vector<RT> part(split ? nitems : 0);
451 RT* pr = split ? part.data() : r.data();
452#ifdef AMREX_USE_OMP
453#pragma omp parallel for schedule(dynamic) if (!system::regtest_reduction)
454#endif
455 for (int item = 0; item < nitems; ++item) {
456 const int L = split ? amrex::bisect(ioff.data(), 0, nlocal, item) : item/ncomp;
457 const int m = item - ioff[L];
458 const int n = m / nchunk[L];
459 const int c = m - n*nchunk[L];
460 Box b = bx[L];
461 b.setSmall(d, bx[L].smallEnd(d) + c*thick[L]);
462 b.setBig(d, std::min(b.smallEnd(d)+thick[L]-1, bx[L].bigEnd(d)));
463 const auto lo = amrex::lbound(b);
464 const auto hi = amrex::ubound(b);
465 RT rem;
466 for_each_init<0, RT, Ops...>(rem);
467 Lanes<RT> acc;
468 lanes_set<0>(acc, rem);
469 for (int k = lo.z; k <= hi.z; ++k) {
470 for (int j = lo.y; j <= hi.y; ++j) {
471 reduce_row<Ops...>(lo.x, hi.x, acc, rem,
472 [&] (int i) { return f(L,i,j,k,n); });
473 }}
474 RT& x = pr[item];
475 for_each_init<0, RT, Ops...>(x);
476 lanes_fold<0, RT, Ops...>(acc, x);
477 for_each_local<0, RT, Ops...>(x, rem);
478 }
479 if (split) {
480 for (int L = 0; L < nlocal; ++L) {
481 for (int m = 0; m < nchunk[L]*ncomp; ++m) {
482 for_each_local<0, RT, Ops...>(r[L*ncomp+m/nchunk[L]], part[ioff[L]+m]);
483 }
484 }
485 }
486 } else {
487 const int nthreads = OpenMP::in_parallel() ? 1 : OpenMP::get_max_threads();
488 Vector<RT> rt(std::size_t(nthreads)*ncomp);
489 for (auto& x : rt) { for_each_init<0, RT, Ops...>(x); }
490#ifdef AMREX_USE_OMP
491#pragma omp parallel if (nthreads > 1 && !system::regtest_reduction)
492#endif
493 {
494 RT* rr = rt.data() + std::size_t(nthreads > 1 ? OpenMP::get_thread_num() : 0)*ncomp;
495 // Component loop outermost, so that each pass streams one component.
496 for (int n = 0; n < ncomp; ++n) {
497 RT rem;
498 for_each_init<0, RT, Ops...>(rem);
499 Lanes<RT> acc;
500 lanes_set<0>(acc, rem);
501 for (MFIter mfi(fa,true); mfi.isValid(); ++mfi) {
502 Box const& b = mfi.growntilebox(nghost);
503 const int L = mfi.LocalIndex();
504 const auto lo = amrex::lbound(b);
505 const auto hi = amrex::ubound(b);
506 for (int k = lo.z; k <= hi.z; ++k) {
507 for (int j = lo.y; j <= hi.y; ++j) {
508 reduce_row<Ops...>(lo.x, hi.x, acc, rem,
509 [&] (int i) { return f(L,i,j,k,n); });
510 }}
511 }
512 lanes_fold<0, RT, Ops...>(acc, rr[n]);
513 for_each_local<0, RT, Ops...>(rr[n], rem);
514 }
515 }
516 for (int it = 0; it < nthreads; ++it) {
517 for (int n = 0; n < ncomp; ++n) {
518 for_each_local<0, RT, Ops...>(r[n], rt[std::size_t(it)*ncomp+n]);
519 }
520 }
521 }
522#endif
523 return r;
524 }
525};
