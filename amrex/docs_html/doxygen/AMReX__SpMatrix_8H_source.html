547// Communication may be skipped only when all rows and all columns live on
548// one common rank. Two partitions with a single active rank each may still
549// name different ranks.
550inline bool spmat_comm_is_local (AlgPartition const& row_partition,
551 AlgPartition const& col_partition)
552{
553 int const rp = row_partition.singleActiveProc();
554 int const cp = col_partition.singleActiveProc();
555 return row_partition.numActiveProcs() <= 1
556 && col_partition.numActiveProcs() <= 1
557 && (rp < 0 || cp < 0 || rp == cp);

560// Diagonal of a square matrix: entry with column index == row index + offset.
561template <typename T, typename I>
562void extract_diagonal (T* p, CsrView<T const,I> const& csr, Long offset)
563{
564 auto const* AMREX_RESTRICT mat = csr.mat;
565 auto const* AMREX_RESTRICT col = csr.col_index;
566 auto const* AMREX_RESTRICT row = csr.row_offset;
567 ParallelForOMP(csr.nrows, [=] AMREX_GPU_DEVICE (Long i)
568 {
569 T d = 0;
570 for (Long j = row[i]; j < row[i+1]; ++j) {
571 if (i == Long(col[j]) - offset) {
572 d = mat[j];
573 break;
574 }
575 }
576 p[i] = d;
577 });
578}

791 if (i < nrows+1) { pr[i] = Long(sr[i]); }
792 if (i < nnz) { pc[i] = Long(sc[i]); }
793 });
794 prow_in = lrow_in.data();
795 pcol_in = lcol_in.data();
796 }
797 if constexpr (std::is_same_v<IO,Long>) {
798 prow_out = csrt.row_offset;
799 pcol_out = csrt.col_index;
800 } else {
801 lrow_out.resize(ncols+1);
802 lcol_out.resize(nnz);
803 prow_out = lrow_out.data();
804 pcol_out = lcol_out.data();
805 }
806#if defined(INTEL_MKL_VERSION) && (INTEL_MKL_VERSION < 20250300)
807 mkl::sparse::set_csr_data(Gpu::Device::streamQueue(), handle_in, nrows, ncols,
808 mkl::index_base::zero, (Long*)prow_in, (Long*)pcol_in,
809 (T*)csr.mat);
810 mkl::sparse::set_csr_data(Gpu::Device::streamQueue(), handle_out, ncols, nrows,
811 mkl::index_base::zero, prow_out, pcol_out, (T*)csrt.mat);
812#else
813 mkl::sparse::set_csr_data(Gpu::Device::streamQueue(), handle_in, nrows, ncols, nnz,
814 mkl::index_base::zero, (Long*)prow_in, (Long*)pcol_in,
815 (T*)csr.mat);
816 mkl::sparse::set_csr_data(Gpu::Device::streamQueue(), handle_out, ncols, nrows, nnz,
817 mkl::index_base::zero, prow_out, pcol_out, (T*)csrt.mat);
818#endif
819
820 mkl::sparse::omatcopy(Gpu::Device::streamQueue(), mkl::transpose::trans,
821 handle_in, handle_out);
822
823 mkl::sparse::release_matrix_handle(Gpu::Device::streamQueue(), &handle_in);
824 auto ev = mkl::sparse::release_matrix_handle(Gpu::Device::streamQueue(), &handle_out);
825 ev.wait();
826 if constexpr (!std::is_same_v<IO,Long>) {
827 auto* po = csrt.col_index; auto const* so = lcol_out.data();
828 auto* pr = csrt.row_offset; auto const* sr = lrow_out.data();
829 ParallelFor(std::max(nnz,ncols+1), [=] AMREX_GPU_DEVICE (Long i) {
830 if (i < nnz) { po[i] = IO(so[i]); }
831 if (i < ncols+1) { pr[i] = IO(sr[i]); }
832 });
833 Gpu::streamSynchronize();
834 }
835
836#endif
837
838 AMREX_GPU_ERROR_CHECK();
839
840#else
841
842 auto* p = csrt.row_offset;
843
844 ParallelForOMP(ncols+1, [=] AMREX_GPU_DEVICE (Long i) { p[i] = 0; });
845
846 // nonzeros per column
847#ifdef AMREX_USE_OMP
848#pragma omp parallel for
849#endif
850 for (Long i = 0; i < nnz; ++i) {
851 auto col = csr.col_index[i];
852#ifdef AMREX_USE_OMP
853#pragma omp atomic update
854#endif
855 ++p[col+1];
856 }
857
858 // build row_offset for transposed matrix. Also save a copy.
859 Vector<Long> current_pos(ncols+1);
860 current_pos[0] = 0;
861 for (Long i = 0; i < ncols; ++i) {
862 p[i+1] += p[i];
863 current_pos[i+1] = p[i+1];
864 }
865
866 // The following code is not OMP safe. It's difficult to use OMP and
867 // still keep CSR sorted.
868 for (Long i = 0; i < nrows; ++i) {
869 for (Long idx = csr.row_offset[i]; idx < csr.row_offset[i+1]; ++idx) {
870 auto col = csr.col_index[idx];
871 Long dest = current_pos[col]++;
872 csrt.mat[dest] = csr.mat[idx];
873 csrt.col_index[dest] = IO(i);
874 }
875 }
876
877#endif
878}

1007template <typename T, template<typename> class Allocator>
1008template <typename I>
1009void
1010SpMatrix<T,Allocator>::define_and_filter_doit (T const* mat, Long const* col_index,
1011 Long nentries, Long const* row_offset)
1012{
1013 Gpu::DeviceVector<I> psum(nentries);
1014 auto* ps = psum.data();
1015 m_nnz = Scan::PrefixSum<I>(I(nentries),
1016 [=] AMREX_GPU_DEVICE (I i) -> I {
1017 return col_index[i] >= 0 && mat[i] != 0; },
1018 [=] AMREX_GPU_DEVICE (I i, I x) {
1019 ps[i] = x; },
1020 Scan::Type::exclusive, Scan::retSum);
1021 Long nlocalrows = this->numLocalRows();
1022 m_csr.mat.resize(m_nnz);
1023 m_csr.col_index.resize(m_nnz);
1024 m_csr.row_offset.resize(nlocalrows+1);
1025 m_csr.nnz = m_nnz;
1026 auto* pmat = m_csr.mat.data();
1027 auto* pcol = m_csr.col_index.data();
1028 auto* prow = m_csr.row_offset.data();
1029 auto actual_nnz = m_nnz;
1030 ParallelFor(std::max(nentries,nlocalrows+1), [=] AMREX_GPU_DEVICE (Long i)
1031 {
1032 if (i < nentries) {
1033 if (col_index[i] >= 0 && mat[i] != 0) {
1034 pmat[ps[i]] = mat[i];
1035 pcol[ps[i]] = col_index[i];
1036 }
1037 }
1038 if (i <= nlocalrows) {
1039 prow[i] = (i < nlocalrows && row_offset[i] < nentries)
1040 ? Long(ps[row_offset[i]]) : actual_nnz;
1041 }
1042 });
1043 Gpu::streamSynchronize();
1044}

1247 if (detail::spmat_comm_is_local(this->partition(), x.partition())) { return; }

1249 this->prepare_comm_mv(x.partition());
1250
1251 auto const mpi_tag = ParallelDescriptor::SeqNum();

1262 m_comm_mv.recv_counts[irecv], mpi_t_type,

1271 if (nsends > 0) {

1344 if (nrecvs > 0) {
1345 recv_buffer = (U*)The_Comms_Arena()->alloc(sizeof(U)*m_comm_mv.total_counts_recv);
1346 auto* p_recv = recv_buffer;
1347 for (int irecv = 0; irecv < nrecvs; ++irecv) {
1348 BL_MPI_REQUIRE(MPI_Irecv(p_recv, m_comm_mv.recv_counts[irecv], mpi_type,
1349 m_comm_mv.recv_from[irecv], mpi_tag, mpi_comm,
1350 &recv_reqs[irecv]));
1351 p_recv += m_comm_mv.recv_counts[irecv];
1352 }
1353 }
1354
1355 auto const nsends = int(m_comm_mv.send_to.size());
1356 U* send_buffer = nullptr;
1357 Vector<MPI_Request> send_reqs(nsends, MPI_REQUEST_NULL);
1358 if (nsends > 0) {
1359 send_buffer = (U*)The_Comms_Arena()->alloc(sizeof(U)*m_comm_mv.total_counts_send);
1360 auto* AMREX_RESTRICT pdst = send_buffer;
1361 auto const* AMREX_RESTRICT pidx = m_comm_mv.send_indices.data();
1362 auto const col_begin = m_col_begin;
1363 ParallelForOMP(m_comm_mv.total_counts_send, [=] AMREX_GPU_DEVICE (Long i)
1364 {
1365 pdst[i] = x[pidx[i]-col_begin];
1366 });
1367 Gpu::streamSynchronize();
1368
1369 auto* p_send = send_buffer;
1370 for (int isend = 0; isend < nsends; ++isend) {
1371 auto count = m_comm_mv.send_counts[isend];
1372 BL_MPI_REQUIRE(MPI_Isend(p_send, count, mpi_type, m_comm_mv.send_to[isend],
1373 mpi_tag, mpi_comm, &send_reqs[isend]));
1374 p_send += count;
1375 }
1376 }
1377
1378 r.resize(m_comm_mv.total_counts_recv);
1379 if (nrecvs > 0) {
1380 Vector<MPI_Status> mpi_statuses(nrecvs);
1381 BL_MPI_REQUIRE(MPI_Waitall(nrecvs, recv_reqs.data(), mpi_statuses.data()));
1382 // The buffer may be pinned host memory, so copy with a kernel.
1383 auto* AMREX_RESTRICT pr = r.data();
1384 auto const* AMREX_RESTRICT pb = recv_buffer;
1385 ParallelForOMP(m_comm_mv.total_counts_recv, [=] AMREX_GPU_DEVICE (Long i)
1386 {
1387 pr[i] = pb[i];
1388 });
1389 }
1390 if (nsends > 0) {
1391 Vector<MPI_Status> mpi_statuses(nsends);
1392 BL_MPI_REQUIRE(MPI_Waitall(nsends, send_reqs.data(), mpi_statuses.data()));
1393 }
1394
1395 Gpu::streamSynchronize();
1396 The_Comms_Arena()->free(send_buffer);
1397 The_Comms_Arena()->free(recv_buffer);
1398#endif
1399 return r;
1400}

1402template <typename T, template<typename> class Allocator>
1403void SpMatrix<T,Allocator>::startComm_tr (AlgPartition const& col_partition)
1404{
1405 BL_PROFILE("SpMatrix::startComm_tr");
1406#ifdef AMREX_USE_MPI
1407 if (detail::spmat_comm_is_local(this->partition(), col_partition)) { return; }
1408
1409 this->split_csr(col_partition);
1410
1411 int const nprocs = ParallelContext::NProcsSub();
1412 auto const mpi_tag = ParallelDescriptor::SeqNum();
1413 auto const mpi_long = ParallelDescriptor::Mpi_typemap<Long>::type();
1414 auto const mpi_t = ParallelDescriptor::Mpi_typemap<T>::type();
1415 auto const mpi_comm = ParallelContext::CommunicatorSub();
1416
1417 // transpose the off-diagonal part
1418 if (m_csr_remote.nnz > 0) {
1419 m_comm_tr.csrt.nnz = m_csr_remote.nnz;
1420 m_comm_tr.csrt.nrows = m_remote_cols_v.size();
1421 m_comm_tr.csrt.mat = (T*)The_Pinned_Arena()->alloc
1422 (sizeof(T)*m_comm_tr.csrt.nnz);
1423 m_comm_tr.csrt.col_index = (Long*)The_Pinned_Arena()->alloc
1424 (sizeof(Long)*m_comm_tr.csrt.nnz);
1425 m_comm_tr.csrt.row_offset = (Long*)The_Pinned_Arena()->alloc
1426 (sizeof(Long)*(m_comm_tr.csrt.nrows+1));
1427#ifdef AMREX_USE_GPU
1428 csr_type csr_comm;
1429 csr_comm.resize(m_comm_tr.csrt.nrows, m_comm_tr.csrt.nnz);
1430 auto const& csrv_comm = csr_comm.view();
1431#else
1432 auto const& csrv_comm = m_comm_tr.csrt;
1433#endif
1434 detail::transpose(csrv_comm, m_csr_remote.const_view());
1435 auto row_begin = m_row_begin;
1436 auto ri_rtol = m_ri_rtol.data();
1437 auto* col_index = csrv_comm.col_index;
1438 ParallelForOMP(csrv_comm.nnz, [=] AMREX_GPU_DEVICE (Long idx)
1439 {
1440 auto gjt =ri_rtol[col_index[idx]] + row_begin;
1441 col_index[idx] = gjt; // global index
1442 });
1443#ifdef AMREX_USE_GPU
1444 Gpu::copyAsync(Gpu::deviceToHost,
1445 csrv_comm. mat,
1446 csrv_comm. mat + csrv_comm.nnz,
1447 m_comm_tr.csrt.mat);
1448 Gpu::copyAsync(Gpu::deviceToHost,
1449 csrv_comm. col_index,
1450 csrv_comm. col_index + csrv_comm.nnz,
1451 m_comm_tr.csrt.col_index);
1452 Gpu::copyAsync(Gpu::deviceToHost,
1453 csrv_comm. row_offset,
1454 csrv_comm. row_offset + csrv_comm.nrows+1,
1455 m_comm_tr.csrt.row_offset);
1456 Gpu::streamSynchronize();
1457#endif
1458 }
1459
1460 if (m_num_neighbors < 0) { set_num_neighbors(); }
1461
1462 // As a sender, I need to let other processes know that how many
1463 // elements I will send them.
1464
1465 Vector<MPI_Request> mpi_requests;
1466 mpi_requests.reserve(nprocs);
1467 if (m_csr_remote.nnz > 0) {
1468 Long it = 0;
1469 for (int iproc = 0; iproc < nprocs; ++iproc) {
1470 Long n = 0;
1471 for (Long i = 0; i < Long(m_remote_cols_vv[iproc].size()); ++i) {
1472 n += m_comm_tr.csrt.row_offset[it+1] - m_comm_tr.csrt.row_offset[it];
1473 ++it;
1474 }
1475 if (n > 0) {
1476 mpi_requests.push_back(MPI_REQUEST_NULL);
1477 AMREX_ALWAYS_ASSERT(n < std::numeric_limits<int>::max());
1478 std::array<int,2> nn{int(n), int(m_remote_cols_vv[iproc].size())};
1479 BL_MPI_REQUIRE(MPI_Isend(nn.data(), 2, MPI_INT, iproc, mpi_tag,
1480 mpi_comm, &(mpi_requests.back())));
1481 m_comm_tr.send_to.push_back(iproc);
1482 m_comm_tr.send_counts.push_back(nn);
1483 }
1484 }
1485 }
1486
1487 // As a receiver, m_num_neighbors is the number of processes from which
1488 // I will receive data.
1489
1490 for (int irecv = 0; irecv < m_num_neighbors; ++irecv) {
1491 MPI_Status mpi_status;
1492 BL_MPI_REQUIRE(MPI_Probe(MPI_ANY_SOURCE, mpi_tag, mpi_comm, &mpi_status));
1493 int sender = mpi_status.MPI_SOURCE;
1494 std::array<int,2> nn;
1495 BL_MPI_REQUIRE(MPI_Recv(nn.data(), 2, MPI_INT, sender, mpi_tag,
1496 mpi_comm, &mpi_status));
1497 m_comm_tr.recv_from.push_back(sender);
1498 m_comm_tr.recv_counts.push_back(nn);
1499 m_comm_tr.total_counts_recv[0] += nn[0];
1500 m_comm_tr.total_counts_recv[1] += nn[1];
1501 }
1502
1503 if (! mpi_requests.empty()) {
1504 Vector<MPI_Status> mpi_statuses(mpi_requests.size());
1505 BL_MPI_REQUIRE(MPI_Waitall(int(mpi_requests.size()), mpi_requests.data(),
1506 mpi_statuses.data()));
1507 }
1508
1509 auto const mpi_tag_m = ParallelDescriptor::SeqNum();
1510 auto const mpi_tag_c = ParallelDescriptor::SeqNum();
1511 auto const mpi_tag_r = ParallelDescriptor::SeqNum();
1512 auto const mpi_tag_p = ParallelDescriptor::SeqNum();
1513
1514 // We need to send m_comm_tr.csrt.mat, col_index & row_offset. We also
1515 // need to send m_remote_cols_vv, which maps row index (in transposed
1516 // matrix) form local to global.
1517
1518 auto const nrecvs = int(m_comm_tr.recv_from.size());
1519 if (nrecvs > 0) {
1520 m_comm_tr.recv_buffer_mat = (T*) The_Pinned_Arena()->alloc
1521 (sizeof(T) * m_comm_tr.total_counts_recv[0]);
1522 m_comm_tr.recv_buffer_col_index = (Long*) The_Pinned_Arena()->alloc
1523 (sizeof(Long) * m_comm_tr.total_counts_recv[0]);
1524 m_comm_tr.recv_buffer_row_offset = (Long*) The_Pinned_Arena()->alloc
1525 (sizeof(Long) * (m_comm_tr.total_counts_recv[1]+nrecvs));
1526 m_comm_tr.recv_buffer_idx_map = (Long*) The_Pinned_Arena()->alloc
1527 (sizeof(Long) * m_comm_tr.total_counts_recv[1]);
1528 m_comm_tr.recv_buffer_offset.push_back({0,0,0,0});
1529 m_comm_tr.recv_reqs.resize(4*nrecvs, MPI_REQUEST_NULL);
1530 for (int irecv = 0; irecv < nrecvs; ++irecv) {
1531 auto [os0, os1, os2, os3] = m_comm_tr.recv_buffer_offset.back();
1532 auto [n0, n1] = m_comm_tr.recv_counts[irecv];
1533 auto recv_from_rank = m_comm_tr.recv_from[irecv];
1534 BL_MPI_REQUIRE(MPI_Irecv(m_comm_tr.recv_buffer_mat + os0,
1535 n0,
1536 mpi_t,
1537 recv_from_rank,
1538 mpi_tag_m,
1539 mpi_comm,
1540 &(m_comm_tr.recv_reqs[irecv*4])));
1541 BL_MPI_REQUIRE(MPI_Irecv(m_comm_tr.recv_buffer_col_index + os1,
1542 n0,
1543 mpi_long,
1544 recv_from_rank,
1545 mpi_tag_c,
1546 mpi_comm,
1547 &(m_comm_tr.recv_reqs[irecv*4+1])));
1548 BL_MPI_REQUIRE(MPI_Irecv(m_comm_tr.recv_buffer_row_offset + os2,
1549 n1+1,
1550 mpi_long,
1551 recv_from_rank,
1552 mpi_tag_r,
1553 mpi_comm,
1554 &(m_comm_tr.recv_reqs[irecv*4+2])));
1555 BL_MPI_REQUIRE(MPI_Irecv(m_comm_tr.recv_buffer_idx_map + os3,
1556 n1,
1557 mpi_long,
1558 recv_from_rank,
1559 mpi_tag_p,
1560 mpi_comm,
1561 &(m_comm_tr.recv_reqs[irecv*4+3])));
1562 m_comm_tr.recv_buffer_offset.push_back({os0 + n0,
1563 os1 + n0,
1564 os2 + n1+1,
1565 os3 + n1});
1566 }
1567 }
1568
1569 auto const nsends = int(m_comm_tr.send_to.size());
1570 if (nsends > 0) {
1571 m_comm_tr.send_reqs.resize(4*nsends, MPI_REQUEST_NULL);
1572 Long os0 = 0, os1 = 0;
1573 for (int isend = 0; isend < nsends; ++isend) {
1574 auto [n0, n1] = m_comm_tr.send_counts[isend];
1575 auto send_to_rank = m_comm_tr.send_to[isend];
1576 BL_MPI_REQUIRE(MPI_Isend(m_comm_tr.csrt.mat + os0,
1577 n0,
1578 mpi_t,
1579 send_to_rank,
1580 mpi_tag_m,
1581 mpi_comm,
1582 &(m_comm_tr.send_reqs[isend*4])));
1583 BL_MPI_REQUIRE(MPI_Isend(m_comm_tr.csrt.col_index + os0,
1584 n0,
1585 mpi_long,
1586 send_to_rank,
1587 mpi_tag_c,
1588 mpi_comm,
1589 &(m_comm_tr.send_reqs[isend*4+1])));
1590 BL_MPI_REQUIRE(MPI_Isend(m_comm_tr.csrt.row_offset + os1,
1591 n1+1,
1592 mpi_long,
1593 send_to_rank,
1594 mpi_tag_r,
1595 mpi_comm,
1596 &(m_comm_tr.send_reqs[isend*4+2])));
1597 BL_MPI_REQUIRE(MPI_Isend(m_remote_cols_vv[send_to_rank].data(),
1598 n1,
1599 mpi_long,
1600 send_to_rank,
1601 mpi_tag_p,
1602 mpi_comm,
1603 &(m_comm_tr.send_reqs[isend*4+3])));
1604 os0 += n0;
1605 os1 += n1;
1606 }
1607 }
1608#else
1609 amrex::ignore_unused(col_partition);
1610#endif
1611}

1669template <typename T, template<typename> class Allocator>
1670void SpMatrix<T,Allocator>::define_split (AlgPartition partition,
1671 AlgPartition const& col_partition,
1672 local_csr_type&& compact, Long nlocal,
1673 Long const* remote_cols, Long nremote)
1674{
1675 BL_PROFILE("SpMatrix::define_split");
1676 AMREX_ALWAYS_ASSERT(m_split == false && m_col_partition.empty());
1677 amrex::ignore_unused(nremote);
1678
1679 m_partition = std::move(partition);
1680 m_row_begin = m_partition[ParallelDescriptor::MyProc()];
1681 m_row_end = m_partition[ParallelDescriptor::MyProc()+1];
1682 AMREX_ASSERT(compact.nrows() == this->numLocalRows());
1683 AMREX_ASSERT(nremote >= 0 && (nremote == 0 || remote_cols != nullptr));
1684 m_diagonal = AlgVector<T,AllocT>{};
1685 m_col_partition = col_partition;
1686 m_col_begin = col_partition[ParallelDescriptor::MyProc()];
1687 m_col_end = col_partition[ParallelDescriptor::MyProc()+1];
1688 AMREX_ALWAYS_ASSERT(m_col_end - m_col_begin == nlocal);
1689 m_nnz = compact.nnz;
1690 m_csr = csr_type{};
1691
1692 Long const nlocalrows = this->numLocalRows();
1693 Long const nnz = compact.nnz;
1694
1695 // Entries with a compact column below nlocal are local; within a row
1696 // they come first.
1697 Gpu::DeviceVector<Long> pfsum(nnz);
1698 auto* p_pfsum = pfsum.data();
1699 auto const* pcol = compact.col_index.data();
1700 AMREX_ALWAYS_ASSERT(nlocal < Long(std::numeric_limits<int>::max()));
1701 int const nl = int(nlocal);
1702 Long const local_nnz = Scan::PrefixSum<Long>(nnz,
1703 [=] AMREX_GPU_DEVICE (Long i) -> Long { return pcol[i] < nl; },
1704 [=] AMREX_GPU_DEVICE (Long i, Long const& x) { p_pfsum[i] = x; },
1705 Scan::Type::exclusive, Scan::retSum);
1706 Long const remote_nnz = nnz - local_nnz;
1707
1708#ifndef AMREX_USE_MPI
1709 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(remote_nnz == 0,
1710 "SpMatrix::define_split: remote columns without MPI");
1711#endif
1712
1713 if (remote_nnz == 0) {
1714 m_csr_local = std::move(compact);
1715 } else {
1716 m_csr_local.resize(nlocalrows, local_nnz);
1717 container_type<Long> remote_gcols(remote_nnz);
1718 container_type<T> remote_mat(remote_nnz);
1719 container_type<int> remote_row_offset(nlocalrows+1);
1720 auto const* pmat = compact.mat.data();
1721 auto* pmat_l = m_csr_local.mat.data();
1722 auto* pcol_l = m_csr_local.col_index.data();
1723 auto* pmat_r = remote_mat.data();
1724 auto* pcol_r = remote_gcols.data();
1725 ParallelForOMP(nnz, [=] AMREX_GPU_DEVICE (Long i)
1726 {
1727 auto ps = p_pfsum[i];
1728 if (pcol[i] < nl) {
1729 pmat_l[ps] = pmat[i];
1730 pcol_l[ps] = pcol[i];
1731 } else {
1732 pmat_r[i-ps] = pmat[i];
1733 pcol_r[i-ps] = remote_cols[pcol[i] - nl];
1734 }
1735 });
1736 auto const noffset = nlocalrows+1;
1737 auto const* pro = compact.row_offset.data();
1738 auto* pro_l = m_csr_local.row_offset.data();
1739 auto* pro_r = remote_row_offset.data();
1740 ParallelForOMP(noffset, [=] AMREX_GPU_DEVICE (Long i)
1741 {
1742 Long ro_l = (i < noffset-1)
1743 ? ((pro[i] < nnz) ? p_pfsum[pro[i]] : local_nnz) : local_nnz;
1744 pro_l[i] = int(ro_l);
1745 pro_r[i] = int(pro[i] - ro_l);
1746 });
1747 Gpu::streamSynchronize();
1748 compact = local_csr_type{};
1749#ifdef AMREX_USE_MPI
1750 m_csr_remote.mat = std::move(remote_mat);
1751 m_csr_remote.col_index.resize(remote_nnz);
1752 m_csr_remote.row_offset = std::move(remote_row_offset);
1753 m_csr_remote.nnz = remote_nnz;
1754 trim_remote_rows();
1755 update_remote_col_index(remote_gcols, m_csr_remote.col_index, true);
1756#endif
1757 }
1758#ifdef AMREX_USE_MPI
1759 if (remote_nnz == 0) {
1760 // Keep the remote bookkeeping consistent with an empty block.
1761 update_remote_col_index(container_type<Long>{}, m_csr_remote.col_index, true);
1762 }
1763#endif
1764
1765 m_split = true;
1766}
1767
1768template <typename T, template<typename> class Allocator>
1769void SpMatrix<T,Allocator>::split_csr (AlgPartition const& col_partition)
1770{
1771 BL_PROFILE("SpMatrix::split_csr");
1772 if (m_split) {
1773 AMREX_ALWAYS_ASSERT
1774 (m_col_begin == col_partition[ParallelDescriptor::MyProc()] &&
1775 m_col_end == col_partition[ParallelDescriptor::MyProc()+1]);
1776 return;
1777 }
1778
1779 AMREX_ALWAYS_ASSERT(m_col_partition.empty());
1780
1781 m_col_partition = col_partition;
1782 m_col_begin = col_partition[ParallelDescriptor::MyProc()];
1783 m_col_end = col_partition[ParallelDescriptor::MyProc()+1];
1784 AMREX_ALWAYS_ASSERT_WITH_MESSAGE
1785 (m_col_end - m_col_begin < Long(std::numeric_limits<int>::max()),
1786 "SpMatrix: the local column block is too large for 32-bit indices");
1787
1788 // This function needs to be safe when nnz is zero.
1789
1790 // Split into a diagonal block with 32-bit local column indices and,
1791 // with MPI, an off-diagonal block with 32-bit indices into the sorted
1792 // list of remote columns.
1793
1794 Long const nlocalrows = this->numLocalRows();
1795 Long local_nnz;
1796 Gpu::DeviceVector<Long> pfsum(m_nnz);
1797 auto* p_pfsum = pfsum.data();
1798 auto col_begin = m_col_begin;
1799 auto col_end = m_col_end;
1800 auto const* pcol = m_csr.col_index.data();
1801 if (m_csr.nnz < Long(std::numeric_limits<int>::max())) {
1802 local_nnz = Scan::PrefixSum<int>(int(m_nnz),
1803 [=] AMREX_GPU_DEVICE (int i) -> int {
1804 return (pcol[i] >= col_begin &&
1805 pcol[i] < col_end); },
1806 [=] AMREX_GPU_DEVICE (int i, int const& x) {
1807 p_pfsum[i] = x; },
1808 Scan::Type::exclusive, Scan::retSum);
1809 } else {
1810 local_nnz = Scan::PrefixSum<Long>(m_nnz,
1811 [=] AMREX_GPU_DEVICE (Long i) -> Long {
1812 return (pcol[i] >= col_begin &&
1813 pcol[i] < col_end); },
1814 [=] AMREX_GPU_DEVICE (Long i, Long const& x) {
1815 p_pfsum[i] = x; },
1816 Scan::Type::exclusive, Scan::retSum);
1817 }
1818 Long const remote_nnz = m_nnz - local_nnz;
1819 AMREX_ALWAYS_ASSERT_WITH_MESSAGE
1820 (local_nnz < Long(std::numeric_limits<int>::max()) &&
1821 remote_nnz < Long(std::numeric_limits<int>::max()),
1822 "SpMatrix: too many local nonzeros for 32-bit row offsets");
1823
1824 m_csr_local.resize(nlocalrows, local_nnz);
1825 container_type<Long> remote_gcols;
1826 container_type<T> remote_mat;
1827 container_type<int> remote_row_offset;
1828#ifdef AMREX_USE_MPI
1829 remote_gcols.resize(remote_nnz);
1830 remote_mat.resize(remote_nnz);
1831 remote_row_offset.resize(nlocalrows+1);
1832#else
1833 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(remote_nnz == 0,
1834 "SpMatrix: column index outside the column partition");
1835#endif
1836 {
1837 auto const* pmat = m_csr.mat.data();
1838 auto* pmat_l = m_csr_local.mat.data();
1839 auto* pcol_l = m_csr_local.col_index.data();
1840 auto* pmat_r = remote_mat.data();
1841 auto* pcol_r = remote_gcols.data();
1842 ParallelForOMP(m_nnz, [=] AMREX_GPU_DEVICE (Long i)
1843 {
1844 auto ps = p_pfsum[i];

1846 if (local) {
1847 pmat_l[ps] = pmat[i];
1848 pcol_l[ps] = int(pcol[i] - col_begin);
1849 } else if (pmat_r) {
1850 pmat_r[i-ps] = pmat[i];
1851 pcol_r[i-ps] = pcol[i];
1852 }
1853 });
1854 auto const noffset = Long(m_csr.row_offset.size());
1855 auto const* pro = m_csr.row_offset.data();
1856 auto* pro_l = m_csr_local.row_offset.data();
1857 auto* pro_r = remote_row_offset.data();

1859 ParallelForOMP(noffset, [=] AMREX_GPU_DEVICE (Long i)
1860 {
1861 Long ro_l;
1862 if (i < noffset-1) {
1863 ro_l = (pro[i] < total_nnz) ? p_pfsum[pro[i]] : local_nnz;
1864 } else {
1865 ro_l = local_nnz;
1866 }
1867 pro_l[i] = int(ro_l);
1868 if (pro_r) { pro_r[i] = int(pro[i] - ro_l); }
1869 });
1870 Gpu::streamSynchronize();
1871 }
1872 m_csr = csr_type{}; // the unsplit form is no longer needed
1873
1874#ifdef AMREX_USE_MPI
1875 // Without remote entries the remote block stays empty (no offsets).
1876 if (remote_nnz > 0) {
1877 m_csr_remote.mat = std::move(remote_mat);
1878 m_csr_remote.col_index.resize(remote_nnz);
1879 m_csr_remote.row_offset = std::move(remote_row_offset);
1880 m_csr_remote.nnz = remote_nnz;
1881 trim_remote_rows();
1882 }
1883 update_remote_col_index(remote_gcols, m_csr_remote.col_index, true);
1884#endif
1885
1886 m_split = true;
1887}

1891template <typename T, template<typename> class Allocator>
1892void SpMatrix<T,Allocator>::trim_remote_rows ()
1893{
1894 BL_PROFILE("SpMatrix::trim_remote_rows");
1895 // In the remote part, it's expected that some rows don't have
1896 // nonzeros. So we trim them off, and also save a full copy of the
1897 // row offsets.
1898 Long old_size = m_csr_remote.row_offset.size();
1899 m_ri_ltor.resize(old_size-1);
1900 m_ri_rtol.resize(old_size-1);
1901 auto* p_ltor = m_ri_ltor.data();
1902 auto* p_rtol = m_ri_rtol.data();
1903 container_type<int> trimmed_row_offset(old_size);
1904 auto const* p_ro = m_csr_remote.row_offset.data();
1905 auto* p_tro = trimmed_row_offset.data();
1906 // This is basically std::unique.
1907 Long new_size = Scan::PrefixSum<Long>(old_size,
1908 [=] AMREX_GPU_DEVICE (Long i) -> Long {
1909 if (i+1 < old_size) {
1910 return (p_ro[i+1] > p_ro[i]);
1911 } else {
1912 return 1;
1913 }
1914 },
1915 [=] AMREX_GPU_DEVICE (Long i, Long const& x) {
1916 if (i == 0) {
1917 p_tro[0] = 0;
1918 } else if (p_ro[i] > p_ro[i-1]) {
1919 p_tro[x] = p_ro[i];
1920 }
1921 if (i+1 < old_size) {
1922 if (p_ro[i+1] > p_ro[i]) {
1923 p_rtol[x] = i;
1924 p_ltor[i] = x;
1925 } else {
1926 p_ltor[i] = -1;
1927 }
1928 }
1929 },
1930 Scan::Type::exclusive, Scan::retSum);
1931 m_ri_rtol.resize(new_size-1);
1932 trimmed_row_offset.resize(new_size);
1933#ifdef AMREX_USE_GPU
1934 m_ri_rtol.shrink_to_fit();
1935 trimmed_row_offset.shrink_to_fit();
1936#endif
1937 m_remote_row_offset = std::move(trimmed_row_offset);
1938 std::swap(m_csr_remote.row_offset, m_remote_row_offset);
1939}

2045template <typename T, template<typename> class Allocator>
2046void SpMatrix<T,Allocator>::prepare_comm_mv (AlgPartition const& col_partition)
2047{
2048 BL_PROFILE("SpMatrix::prepare_comm_mv");
2049 if (m_comm_mv.prepared) { return; }
2050
2051 // This function needs to be safe when nnz is zero.
2052
2053 this->split_csr(col_partition);
2054
2055 int const nprocs = ParallelContext::NProcsSub();
2056 auto const mpi_tag = ParallelDescriptor::SeqNum();
2057 auto const mpi_long = ParallelDescriptor::Mpi_typemap<Long>::type();
2058 auto const mpi_comm = ParallelContext::CommunicatorSub();
2059
2060 if (m_num_neighbors < 0) { set_num_neighbors(); }
2061
2062 Vector<MPI_Request> mpi_requests;
2063 mpi_requests.reserve(nprocs);
2064 for (int iproc = 0; iproc < nprocs; ++iproc) {
2065 if ( ! m_remote_cols_vv[iproc].empty()) {

2069 amrex::Abort("SpMatrix::prepare_comm_mv: remote column payload exceeds MPI int count range.");
2070 }

2072 // I need to let other processes know what I need from them.
2073 BL_MPI_REQUIRE(MPI_Isend(m_remote_cols_vv[iproc].data(),
2074 msg_count,
2075 mpi_long, iproc, mpi_tag, mpi_comm,
2076 &(mpi_requests.back())));

2078 m_comm_mv.recv_counts.push_back(msg_count);
2079 }
2080 }
2081
2082 m_comm_mv.total_counts_recv = Long(m_remote_cols_v.size());
2083
2084 Vector<Vector<Long>> send_indices(m_num_neighbors);
2085 m_comm_mv.total_counts_send = 0;
2086 for (int isend = 0; isend < m_num_neighbors; ++isend) {
2087 MPI_Status mpi_status;
2088 BL_MPI_REQUIRE(MPI_Probe(MPI_ANY_SOURCE, mpi_tag, mpi_comm, &mpi_status));
2089 int receiver = mpi_status.MPI_SOURCE;
2090 int count;
2091 BL_MPI_REQUIRE(MPI_Get_count(&mpi_status, mpi_long, &count));
2092 m_comm_mv.send_to.push_back(receiver);
2093 m_comm_mv.send_counts.push_back(count);
2094 send_indices[isend].resize(count);
2095 BL_MPI_REQUIRE(MPI_Recv(send_indices[isend].data(), count, mpi_long,
2096 receiver, mpi_tag, mpi_comm, &mpi_status));
2097 m_comm_mv.total_counts_send += count;
2098 }
2099
2100 m_comm_mv.send_indices.resize(m_comm_mv.total_counts_send);
2101 Gpu::PinnedVector<Long> send_indices_all;
2102 send_indices_all.reserve(m_comm_mv.total_counts_send);
2103 for (auto const& vl : send_indices) {
2104 for (auto x : vl) {
2105 send_indices_all.push_back(x);
2106 }
2107 }
2108 Gpu::copyAsync(Gpu::hostToDevice, send_indices_all.begin(), send_indices_all.end(),
2109 m_comm_mv.send_indices.begin());
2110 Gpu::streamSynchronize();
2111
2112 if (! mpi_requests.empty()) {
2113 Vector<MPI_Status> mpi_statuses(mpi_requests.size());
2114 BL_MPI_REQUIRE(MPI_Waitall(int(mpi_requests.size()), mpi_requests.data(),
2115 mpi_statuses.data()));
2116 }
2117
2118 m_comm_mv.prepared = true;
2119}

2166 m_col_begin = m_col_partition[ParallelDescriptor::MyProc() ];
2167 m_col_end = m_col_partition[ParallelDescriptor::MyProc()+1];

2169 m_ri_ltor.resize(numLocalRows(), -1);

2175 "SpMatrix: too many received nonzeros for 32-bit row offsets");
2176
2177 m_nnz += nnz;
2178 auto nb = int(ctr.recv_from.size()); // # of blocked CSRs to be merged
2179 auto total_local_rows = ctr.total_counts_recv[1];
2180
2181 // Build compressed row index map

2183 ctr.recv_buffer_idx_map + total_local_rows);

2186
2187 // Assembled on the host with global columns, then converted to 32-bit
2188 // remote indices.

2191#else

2193#endif

2195 csrr.mat.resize(nnz);
2196 csrr.col_index.resize(nnz);
2197 csrr.row_offset.resize(nrows+1);

2199
2200 // Merge blocks in sender-rank order so that rows stay sorted. Each
2201 // block covers a contiguous global column range.
2202 Vector<int> order(nb);
2203 std::iota(order.begin(), order.end(), 0);
2204 std::sort(order.begin(), order.end(), [&] (int a, int b) {
2205 return ctr.recv_from[a] < ctr.recv_from[b]; });

2208 Vector<int> row_nnz(nrows, 0);
2209 for (int i : order) {
2210 auto nrow_i = ctr.recv_counts[i][1];
2211 Long const* row_offset = ctr.recv_buffer_row_offset
2212 + ctr.recv_buffer_offset[i][2];

2214 + ctr.recv_buffer_offset[i][3];

2221 AMREX_ASSERT(p < nrows && ri_map[p] == gr);
2222 // p is now compressed row index

2224 }
2225 }
2226
2227 std::exclusive_scan(row_nnz.begin(), row_nnz.end(), csrr.row_offset.begin(), 0);
2228 csrr.row_offset.back() = int(csrr.nnz);
2229
2230 auto rowpos = csrr.row_offset; // make a copy to keep track of offset
2231
2232 for (int i : order) {
2233 auto nrow_i = ctr.recv_counts[i][1];
2234 T const* mat = ctr.recv_buffer_mat
2235 + ctr.recv_buffer_offset[i][0];
2236 Long const* col_index = ctr.recv_buffer_col_index
2237 + ctr.recv_buffer_offset[i][1];
2238 Long const* row_offset = ctr.recv_buffer_row_offset
2239 + ctr.recv_buffer_offset[i][2];
2240 Long const* idx_map = ctr.recv_buffer_idx_map
2241 + ctr.recv_buffer_offset[i][3];

2244 for (int lr = 0; lr < nrow_i; ++lr) {
2245 Long const gr = idx_map[lr];

2247 AMREX_ASSERT(p < nrows && ri_map[p] == gr);
2248 // p is now compressed row index
2249 auto os_src = row_offset[lr] - row_offset[0];
2250 auto nvals = row_offset[lr+1] - row_offset[lr];
2251 auto os_dst = rowpos[p];
2252 std::memcpy(csrr.mat.data()+os_dst, mat+os_src, sizeof(T)*nvals);
2253 std::memcpy(gcols.data()+os_dst, col_index+os_src, sizeof(Long)*nvals);
2254 rowpos[p] += int(nvals);
2255 }
2256 }
2257
2258 m_ri_rtol.resize(nrows);
2259 Gpu::copyAsync(Gpu::hostToDevice, ri_map.begin(), ri_map.end(), m_ri_rtol.begin());
2260 {

2262 auto* AMREX_RESTRICT ltor = m_ri_ltor.data();

2264 ParallelForOMP(nrows, [=] AMREX_GPU_DEVICE (Long i) {
2265 rtol[i] -= row_begin;
2266 ltor[rtol[i]] = i;
2267 });
2268 }

2271
2272#ifdef AMREX_USE_GPU
2273 amrex::duplicateCSR(Gpu::hostToDevice, m_csr_remote, csrr);
2274 Gpu::streamSynchronize();
2275#endif
2276}

2278template <typename T, template<typename> class Allocator>
2279auto SpMatrix<T,Allocator>::fetch_remote_rows_mm (SpMatrix<T,Allocator> const& B)
2280 -> RemoteRowsMM
2281{

2283 // other ranks need from us and which we need from them.
2284 this->prepare_comm_mv(B.partition());
2285
2286 auto const& cm = m_comm_mv;
2287 auto const nrecvs = int(cm.recv_from.size());
2288 auto const nsends = int(cm.send_to.size());
2289 auto const mpi_long = ParallelDescriptor::Mpi_typemap<Long>::type();
2290 auto const mpi_t = ParallelDescriptor::Mpi_typemap<T>::type();
2291 auto const mpi_comm = ParallelContext::CommunicatorSub();
2292
2293 RemoteRowsMM ext;
2294 ext.nrows = cm.total_counts_recv; // == m_remote_cols_v.size()
2295 ext.row_offset.resize(ext.nrows+1);
2296
2297 auto const b0 = B.m_csr_local.const_view();
2298 auto const b1 = B.remote_full_const_view();
2299 Long const b_row_begin = B.m_row_begin;
2300 Long const nsend_rows = cm.total_counts_send;
2301 auto const* AMREX_RESTRICT send_idx = cm.send_indices.data();
2302
2303 // Round 1: number of nonzeros in each requested row.
2304 Gpu::PinnedVector<Long> h_send(nsend_rows+1);
2305 {
2306 Gpu::DeviceVector<Long> d_cnt(nsend_rows);
2307 auto* pcnt = d_cnt.data();
2308 ParallelForOMP(nsend_rows, [=] AMREX_GPU_DEVICE (Long i)
2309 {
2310 Long const lr = send_idx[i] - b_row_begin;
2311 pcnt[i] = (b0.row_offset[lr+1] - b0.row_offset[lr])
2312 + (b1.row_offset[lr+1] - b1.row_offset[lr]);
2313 });
2314 Gpu::copyAsync(Gpu::deviceToHost, d_cnt.begin(), d_cnt.end(), h_send.begin());
2315 Gpu::streamSynchronize();
2316 }
2317
2318 Gpu::PinnedVector<Long> h_recv(ext.nrows+1);
2319 {
2320 auto const tag = ParallelDescriptor::SeqNum();
2321 Vector<MPI_Request> reqs;
2322 reqs.reserve(nrecvs+nsends);
2323 Long os = 0;
2324 for (int i = 0; i < nrecvs; ++i) {
2325 reqs.push_back(MPI_REQUEST_NULL);
2326 BL_MPI_REQUIRE(MPI_Irecv(h_recv.data()+os, cm.recv_counts[i], mpi_long,
2327 cm.recv_from[i], tag, mpi_comm, &reqs.back()));
2328 os += cm.recv_counts[i];
2329 }
2330 os = 0;
2331 for (int i = 0; i < nsends; ++i) {
2332 reqs.push_back(MPI_REQUEST_NULL);
2333 BL_MPI_REQUIRE(MPI_Isend(h_send.data()+os, cm.send_counts[i], mpi_long,
2334 cm.send_to[i], tag, mpi_comm, &reqs.back()));

2336 }
2337 if (! reqs.empty()) {
2338 Vector<MPI_Status> stats(reqs.size());
2339 BL_MPI_REQUIRE(MPI_Waitall(int(reqs.size()), reqs.data(), stats.data()));
2340 }
2341 }
2342
2343 // Counts to offsets, in place; per-rank totals.
2344 auto to_offsets = [] (Gpu::PinnedVector<Long>& v, Long n) {
2345 Long s = 0;
2346 for (Long i = 0; i < n; ++i) {
2347 Long const c = v[i];
2348 v[i] = s;
2349 s += c;
2350 }
2351 v[n] = s;
2352 };

2354 to_offsets(h_send, nsend_rows);
2355 ext.nnz = h_recv[ext.nrows];

2359 for (int i = 0, r = 0; i < nrecvs; ++i) {

2361 r += cm.recv_counts[i];

2363 amrex::Abort("SpMatrix::fetch_remote_rows_mm: message exceeds MPI int count range.");
2364 }
2365 }
