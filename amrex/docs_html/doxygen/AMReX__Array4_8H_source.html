198 template<std::size_t... idx>
199 constexpr auto make_oob_message_impl (std::index_sequence<idx...>) {
200 constexpr std::size_t N = sizeof...(idx);
201 constexpr char prefix[] = " (";
202 constexpr char middle[] = ") is out of bound (";
203 constexpr char suffix[] = ")\n";
204
205 constexpr std::size_t size =
206 sizeof(prefix) - 1 +
207 (2*N + (N-1)) +
208 sizeof(middle) - 1 +
209 (5*N + (N-1)) +
210 sizeof(suffix); // include null terminator
211
212 std::array<char, size> buf{};
213
214 std::size_t pos = 0;
215
216 // prefix
217 for (char c : prefix) {
218 if (c != '\0') {
219 buf[pos++] = c;
220 }
221 }
222 // %d,%d,...
223 ((buf[pos++] = '%', buf[pos++] = 'd', idx + 1 < N ? buf[pos++] = ',' : 0), ...);
224
225 // ") is out of bound ("
226 for (char c : middle) {
227 if (c != '\0') {
228 buf[pos++] = c;
229 }
230 }
231
232 // %d:%d,%d:%d,...
233 ((buf[pos++] = '%', buf[pos++] = 'd', buf[pos++] = ':', buf[pos++] = '%', buf[pos++] = 'd',
234 idx + 1 < N ? buf[pos++] = ',' : 0), ...);
235
236 // suffix
237 for (char c : suffix) {
238 buf[pos++] = c;
239 }
240 return buf;
241 }

799 [[nodiscard]] AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
800 constexpr Long get_offset (IntVectND<M> const& iv) const noexcept
801 {
802 Long offset = iv.vect[0] - begin.vect[0];
803 // If M == N and we have a component at the end, we only loop up to N-1 for spatial.
804 // Otherwise, we loop up to M.
805 constexpr int idx = (last_dim_component && M == N) ? N - 1 : M;
806 if constexpr (N > 1) {
807 constexpr_for<1, idx>([&](int d) {
808 offset += Long(iv.vect[d] - begin.vect[d]) * stride.a[d-1];
809 });
810 }
811
812 // Handle the component offset if the input is the full N dimensions
813 // and the last dimension is a component.
814 if constexpr (last_dim_component && M == N) {
815 offset += iv.vect[N-1] * get_stride<N-1>();
816 }
817 return offset;
818 }
